Feedback form with validation

A star rating, category and comment are required. The comment needs at least 10 characters.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="f" novalidate style="max-width:640px">
<div class="fg"><label for="name">Name</label><div class="line"><input type="text" id="name" name="name"><small class="msg" id="e_name"></small></div></div>
<div class="fg"><label for="email">Email</label><div class="line"><input type="email" id="email" name="email"><small class="msg" id="e_email"></small></div></div>
<div class="fg"><label for="rating">Rating</label><div class="line"><label><input type="radio" name="rating" value="1 - Poor"> 1 - Poor</label> <label><input type="radio" name="rating" value="2 - Fair"> 2 - Fair</label> <label><input type="radio" name="rating" value="3 - Good"> 3 - Good</label> <label><input type="radio" name="rating" value="4 - Very good"> 4 - Very good</label> <label><input type="radio" name="rating" value="5 - Excellent"> 5 - Excellent</label> <small class="msg" id="e_rating"></small></div></div>
<div class="fg"><label for="category">Feedback about</label><div class="line"><select id="category" name="category"><option value="">-- Select --</option><option>Teaching</option><option>Facilities</option><option>Website</option><option>Other</option></select><small class="msg" id="e_category"></small></div></div>
<div class="fg"><label for="comment">Comments</label><div class="line"><textarea id="comment" name="comment" rows="3"></textarea><small class="msg" id="e_comment"></small></div></div>
<button type="submit">Send feedback</button> <button type="reset" class="secondary">Reset</button>
</form>
<div id="banner" role="status"></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const form = $('f');
const kinds = {"name": "text", "email": "email", "rating": "radio", "category": "select", "comment": "textarea"};
const rules = {
  name: (v, get) => !v ? 'Name is required' : /^[A-Za-z ]+$/.test(v) ? '' : 'Only alphabets and spaces are allowed',
  email: (v, get) => !v ? 'Email is required' : /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v) ? '' : 'Enter a valid email like name@example.com',
  rating: (v, get) => v ? '' : 'Please give a rating',
  category: (v, get) => v ? '' : 'Please select an option',
  comment: (v, get) => !v ? 'This field is required' : v.length < 10 ? 'Write at least 10 characters (' + v.length + '/10)' : ''
};
const ageFrom = (iso) => {
  const d = new Date(iso), now = new Date();
  let age = now.getFullYear() - d.getFullYear();
  if (now < new Date(now.getFullYear(), d.getMonth(), d.getDate())) age--;
  return age;
};
function getVal(name) {
  const kind = kinds[name];
  if (kind === 'radio') { const c = form.querySelector(`input[name="${name}"]:checked`); return c ? c.value : ''; }
  if (kind === 'checks') return [...form.querySelectorAll(`input[name="${name}"]:checked`)].map((c) => c.value);
  if (kind === 'check') return form.elements[name].checked;
  return form.elements[name].value.trim();
}
function showState(name, msg) {
  const box = $('e_' + name);
  box.textContent = msg;
  box.classList.remove('good');
  const el = form.elements[name];
  if (el && el.classList) { el.classList.toggle('invalid', msg !== ''); el.classList.toggle('valid', msg === '' && getVal(name) !== ''); }
}
function validateField(name) {
  const msg = rules[name](getVal(name), getVal);
  showState(name, msg);
  return msg === '';
}

form.addEventListener('submit', (e) => {
  e.preventDefault();                       // stop the page from reloading / submitting
  let ok = true, firstBad = null;
  Object.keys(rules).forEach((name) => { if (!validateField(name)) { ok = false; firstBad = firstBad || name; } });
  const banner = $('banner');
  if (ok) {
    banner.className = 'banner good';
    banner.textContent = 'Thank you for your feedback!';
    form.reset();
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
  } else {
    banner.className = 'banner bad';
    banner.textContent = 'Please correct the highlighted fields.';
    const el = form.elements[firstBad];
    if (el && el.focus) el.focus();
  }
});
form.addEventListener('reset', () => {
  setTimeout(() => {
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
    $('banner').className = ''; $('banner').textContent = '';
  });
});
