Create a strong password validator

A live checklist shows which password rules are met. The button unlocks only when all five are satisfied.

Working demonstration

    View source code of this program 
    <!-- HTML -->
<label class="field" style="max-width:340px">Password<input id="pw" type="text" autocomplete="off" placeholder="Type a password"></label>
<ul class="rules" id="rules"></ul><button id="go" disabled>Set password</button><p id="msg" class="ok"></p>

/* CSS */
.demo .rules { list-style: none; padding: 0; margin: .6rem 0 1rem; } .demo .rules li { padding: .15rem 0; color: #c62828; } .demo .rules li.ok { color: #178a4a; } .demo .rules li::before { content: '\2716  '; } .demo .rules li.ok::before { content: '\2714  '; }

// JavaScript
const $ = (id) => document.getElementById(id);
const rules = [
  ['At least 8 characters', (p) => p.length >= 8],
  ['An uppercase letter (A-Z)', (p) => /[A-Z]/.test(p)],
  ['A lowercase letter (a-z)', (p) => /[a-z]/.test(p)],
  ['A digit (0-9)', (p) => /\d/.test(p)],
  ['A special character (!@#$%...)', (p) => /[^A-Za-z0-9]/.test(p)],
];
const list = $('rules');
rules.forEach(([text]) => { const li = document.createElement('li'); li.textContent = text; list.appendChild(li); });
$('pw').addEventListener('input', () => {
  const p = $('pw').value;
  let passed = 0;
  rules.forEach(([, test], i) => { const ok = test(p); list.children[i].classList.toggle('ok', ok); if (ok) passed++; });
  $('go').disabled = passed < rules.length;
  $('msg').textContent = '';
});
$('go').addEventListener('click', () => { $('msg').textContent = 'Strong password accepted.'; });
