Validate input and display errors without a page reload

The form is validated and "sent" entirely in JavaScript. The page-load time at the bottom never changes, proving that the page did not reload.

<!-- HTML -->
<form id="f" novalidate style="max-width:460px"><div class="fg"><label>Name</label><input id="name" type="text"><small class="msg" id="e_name"></small></div>
<div class="fg"><label>Email</label><input id="email" type="email"><small class="msg" id="e_email"></small></div>
<div class="fg"><label>Message</label><textarea id="msg" rows="3" style="width:100%"></textarea><small class="msg" id="e_msg"></small></div><button id="go">Send</button></form>
<div id="banner" role="status"></div><p class="muted">This page was loaded at <strong id="loaded"></strong>. Submissions so far: <strong id="n">0</strong></p>

// JavaScript
const $ = (id) => document.getElementById(id);
$('loaded').textContent = new Date().toLocaleTimeString();
let sent = 0;
const rules = { name: (v) => v.length >= 3 ? '' : 'Name needs at least 3 characters', email: (v) => /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v) ? '' : 'Enter a valid email', msg: (v) => v.length >= 10 ? '' : 'Message needs at least 10 characters' };
const fakeServer = (data) => new Promise((resolve) => setTimeout(() => resolve({ ok: true, id: Math.floor(Math.random() * 9000) + 1000 }), 900));   // pretend network request
$('f').addEventListener('submit', async (e) => {
  e.preventDefault();
  let ok = true;
  Object.entries(rules).forEach(([id, fn]) => { const m = fn($(id).value.trim()); $('e_' + id).textContent = m; $(id).classList.toggle('invalid', !!m); if (m) ok = false; });
  if (!ok) { $('banner').className = 'banner bad'; $('banner').textContent = 'Please fix the errors above.'; return; }
  $('go').disabled = true; $('go').textContent = 'Sending...';
  $('banner').className = 'banner'; $('banner').textContent = '';
  const res = await fakeServer({ name: $('name').value });
  $('go').disabled = false; $('go').textContent = 'Send';
  $('banner').className = 'banner good';
  $('banner').textContent = `Message sent (reference #${res.id}). The page did not reload.`;
  $('n').textContent = ++sent;
  $('f').reset();
});
