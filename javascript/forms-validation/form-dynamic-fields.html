Form with dynamic fields based on user selection

Choose who you are (student, employee or business) and the form builds different fields. You can also add extra phone numbers.

Working demonstration






View source code of this program

<!-- HTML -->
<form id="f" novalidate style="max-width:520px"><div class="fg"><label>I am a</label><select id="type"><option value="">-- Select --</option><option value="student">Student</option><option value="employee">Employee</option><option value="business">Business owner</option></select></div>
<div id="dyn"></div><div id="phones"></div><button type="button" id="addPhone" class="secondary small" hidden>+ Add another phone</button><p class="err" id="err"></p><button type="submit" id="go" hidden>Submit</button></form><pre class="out" id="out" hidden></pre>

// JavaScript
const $ = (id) => document.getElementById(id);
const config = {
  student: [['college', 'College name', 'text'], ['course', 'Course', 'text'], ['year', 'Year of study', 'number']],
  employee: [['company', 'Company name', 'text'], ['role', 'Job title', 'text'], ['exp', 'Years of experience', 'number']],
  business: [['biz', 'Business name', 'text'], ['gst', 'GST number', 'text'], ['staff', 'Number of employees', 'number']],
};
function addPhone() {
  const n = $('phones').children.length + 1;
  const div = document.createElement('div');
  div.className = 'fg';
  div.innerHTML = `<label>Phone ${n}</label><input type="tel" class="phone" placeholder="10 digits">`;
  $('phones').appendChild(div);
}
$('type').addEventListener('change', () => {
  const fields = config[$('type').value] || [];
  $('dyn').innerHTML = '';
  fields.forEach(([id, label, type]) => {
    const div = document.createElement('div');
    div.className = 'fg';
    div.innerHTML = `<label></label><input id="${id}" type="${type}" data-label="${label}">`;
    div.firstChild.textContent = label;
    $('dyn').appendChild(div);
  });
  $('phones').innerHTML = '';
  if (fields.length) addPhone();
  $('addPhone').hidden = $('go').hidden = !fields.length;
});
$('addPhone').addEventListener('click', addPhone);
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  const data = { type: $('type').value };
  let bad = null;
  $('dyn').querySelectorAll('input').forEach((i) => { if (!i.value.trim()) bad = bad || i.dataset.label; data[i.id] = i.value.trim(); i.classList.toggle('invalid', !i.value.trim()); });
  const phones = [...document.querySelectorAll('.phone')].map((p) => p.value.trim());
  if (phones.some((p) => !/^[6-9]\d{9}$/.test(p))) bad = bad || 'a valid 10-digit phone number';
  data.phones = phones;
  $('err').textContent = bad ? `Please fill in: ${bad}` : '';
  $('out').hidden = !!bad;
  if (!bad) $('out').textContent = 'Collected data:\n' + JSON.stringify(data, null, 2);
});
