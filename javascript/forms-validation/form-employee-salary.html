Employee salary form: calculate salary details

Enter the basic salary and percentages. The form calculates HRA, DA, gross salary, deductions (PF and tax) and net salary.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="f" novalidate><div class="row"><label class="field">Employee name<input id="name" type="text" value="Ravi Kumar"></label><label class="field">Basic salary<input id="basic" type="number" value="40000" min="0"></label><label class="field">HRA %<input id="hra" type="number" value="20" min="0" max="100"></label><label class="field">DA %<input id="da" type="number" value="10" min="0" max="100"></label><label class="field">PF %<input id="pf" type="number" value="12" min="0" max="30"></label></div><button>Calculate salary</button></form><p class="err" id="err"></p><div id="res"></div>

/* CSS */
.demo .payslip td:last-child, .demo .payslip th:last-child { text-align: right; }

// JavaScript
const $ = (id) => document.getElementById(id);
const val = (id) => parseFloat($(id).value);
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  const [basic, hraP, daP, pfP] = ['basic', 'hra', 'da', 'pf'].map(val);
  if ([basic, hraP, daP, pfP].some((x) => Number.isNaN(x) || x < 0) || basic <= 0) { $('err').textContent = 'Enter a basic salary above 0 and non-negative percentages.'; $('res').innerHTML = ''; return; }
  $('err').textContent = '';
  const hra = basic * hraP / 100, da = basic * daP / 100, gross = basic + hra + da, pf = basic * pfP / 100;
  const annual = gross * 12;
  const tax = annual <= 300000 ? 0 : annual <= 700000 ? (annual - 300000) * 0.05 / 12 : (20000 + (annual - 700000) * 0.10) / 12;   // simple demo slabs
  const net = gross - pf - tax;
  const fmt = (n) => n.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 });
  const rows = [['Basic salary', basic], ['HRA', hra], ['DA', da], ['Gross salary', gross], ['PF deduction', -pf], ['Income tax (monthly, demo slabs)', -tax], ['Net salary', net]];
  $('res').innerHTML = '<table class="data payslip" style="max-width:480px"><tr><th colspan="2" id="who"></th></tr>' + rows.map(([l, v]) => `<tr><td>${l}</td><td>${v < 0 ? '- ' : ''}${fmt(Math.abs(v))}</td></tr>`).join('') + '</table>';
  $('who').textContent = 'Pay slip: ' + ($('name').value || 'Employee');
});
