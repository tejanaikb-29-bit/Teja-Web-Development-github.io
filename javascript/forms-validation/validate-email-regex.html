Validate an email address using a regular expression

Checks the email against the pattern /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/ and explains each part.

The result will appear here.

<!-- HTML -->
<div class="row">
  <label class="field">Enter an email address
    <input id="email" type="text" value="student@example.com">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  const email = txt('email').trim();
  const pattern = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/;
  const ok = pattern.test(email);
  show(`${email || '(empty)'}\n${ok ? 'VALID email address' : 'INVALID email address'}\n
  Pattern parts:
  ^[^\\s@]+   text before @ (no spaces, no @)
  @          the @ symbol
  [^\\s@]+    the domain name
  \\.         a dot
  [^\\s@]{2,}$ at least 2 characters at the end (com, in, org...)`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
