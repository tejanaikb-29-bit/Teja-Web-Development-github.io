Show / hide password option

A login form where an eye button reveals or hides each password field, with validation on submit.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="f" novalidate style="max-width:420px"><label class="field">Password<span class="pwbox"><input id="p1" type="password" placeholder="min 8 characters"><button type="button" class="secondary" data-for="p1">Show</button></span></label>
<label class="field" style="margin-top:.8rem">Confirm password<span class="pwbox"><input id="p2" type="password"><button type="button" class="secondary" data-for="p2">Show</button></span></label>
<p id="msg" class="msg" style="margin:.6rem 0"></p><button>Save password</button></form>

/* CSS */
.demo .pwbox { display: flex; gap: .4rem; } .demo .pwbox input { flex: 1; min-width: 0; }

// JavaScript
const $ = (id) => document.getElementById(id);
document.querySelectorAll('button[data-for]').forEach((b) => {
  b.addEventListener('click', () => {
    const input = $(b.dataset.for);
    const show = input.type === 'password';
    input.type = show ? 'text' : 'password';
    b.textContent = show ? 'Hide' : 'Show';
  });
});
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  const a = $('p1').value, b = $('p2').value;
  $('msg').className = 'msg';
  if (a.length < 8) $('msg').textContent = 'Password needs at least 8 characters.';
  else if (a !== b) $('msg').textContent = 'The two passwords do not match.';
  else { $('msg').className = 'msg good'; $('msg').textContent = 'Password saved.'; }
});
