<!-- HTML -->
<div class="prog" id="prog"><div>1. Account</div><div>2. Personal</div><div>3. Review</div></div>
<form id="f" novalidate class="step">
  <section data-step="0"><div class="fg"><label>Username</label><input id="user" type="text"><small class="msg" id="e_user"></small></div>

// JavaScript
const $ = (id) => document.getElementById(id);
let step = 0;
const checks = [
  { user: (v) => /^\w{4,}$/.test(v) ? '' : 'At least 4 letters, digits or _', email: (v) => /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v) ? '' : 'Enter a valid email', pw: (v) => v.length >= 8 ? '' : 'At least 8 characters' },
  { name: (v) => /^[A-Za-z ]{3,}$/.test(v) ? '' : 'Letters only, min 3 characters', phone: (v) => /^[6-9]\d{9}$/.test(v) ? '' : 'Enter a valid 10-digit number', dob: (v) => v && new Date(v) < new Date() ? '' : 'Choose a past date' },
];
function validate(s) {
  let ok = true;
  Object.entries(checks[s]).forEach(([id, fn]) => { const msg = fn($(id).value.trim()); $('e_' + id).textContent = msg; $(id).classList.toggle('invalid', !!msg); if (msg) ok = false; });
  return ok;
}
function render() {
  document.querySelectorAll('section[data-step]').forEach((s) => { s.hidden = Number(s.dataset.step) !== step; });
  [...$('prog').children].forEach((d, i) => { d.className = i < step ? 'done' : i === step ? 'on' : ''; });
  $('back').hidden = step === 0;
  $('next').textContent = step === 2 ? 'Submit' : 'Next';
  if (step === 2) {
    $('review').innerHTML = '';
    [['Username', 'user'], ['Email', 'email'], ['Full name', 'name'], ['Phone', 'phone'], ['Date of birth', 'dob']].forEach(([label, id]) => {
      const tr = $('review').insertRow();
      tr.insertCell().textContent = label; tr.insertCell().textContent = $(id).value;
    });
  }
}
$('next').addEventListener('click', () => {
  if (step < 2) { if (validate(step)) { step++; render(); } }
  else { $('f').hidden = true; $('prog').hidden = true; $('done').hidden = false; }
});
$('back').addEventListener('click', () => { step--; render(); });
render();
