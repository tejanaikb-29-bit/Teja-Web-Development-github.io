Create a password strength meter

A coloured bar and label rate the password from Very weak to Very strong and suggest improvements.

Working demonstration

Enter a password

    View source code of this program 
    <!-- HTML -->
<label class="field" style="max-width:340px">Password<input id="pw" type="password" autocomplete="off" placeholder="Start typing..."></label><div class="bar"><i id="fill"></i></div><p><strong id="label">Enter a password</strong></p><ul id="tips" class="muted"></ul>

/* CSS */
.demo .bar { height: 12px; background: #e2e8f4; border-radius: 6px; overflow: hidden; margin: .6rem 0 .3rem; max-width: 340px; } .demo .bar i { display: block; height: 100%; width: 0; transition: width .25s, background .25s; }

// JavaScript
const $ = (id) => document.getElementById(id);
const levels = [['Very weak', '#c62828'], ['Weak', '#e67e22'], ['Fair', '#d98a00'], ['Strong', '#5aa02c'], ['Very strong', '#178a4a']];
$('pw').addEventListener('input', () => {
  const p = $('pw').value;
  $('tips').innerHTML = '';
  if (!p) { $('fill').style.width = '0'; $('label').textContent = 'Enter a password'; $('label').style.color = ''; return; }
  const checks = [[p.length >= 8, 'Use at least 8 characters'], [/[A-Z]/.test(p), 'Add an uppercase letter'], [/[a-z]/.test(p), 'Add a lowercase letter'], [/\d/.test(p), 'Add a number'], [/[^A-Za-z0-9]/.test(p), 'Add a symbol'], [p.length >= 12, 'Make it 12+ characters for extra strength']];
  let score = checks.filter(([ok]) => ok).length - 1;          // 0..5
  if (/^(.)\1+$/.test(p) || /password|123456|qwerty/i.test(p)) score = 0;   // obvious patterns
  const level = levels[Math.max(0, Math.min(4, score - (score > 4 ? 1 : 0)))];
  $('fill').style.width = ((levels.indexOf(level) + 1) / levels.length) * 100 + '%';
  $('fill').style.background = level[1];
  $('label').textContent = level[0];
  $('label').style.color = level[1];
  checks.filter(([ok]) => !ok).forEach(([, tip]) => { const li = document.createElement('li'); li.textContent = tip; $('tips').appendChild(li); });
});
