Student registration form with full validation

Every field type is validated: text, email, phone, date, radio, select, checkboxes, password, confirm password, textarea and terms.

<!-- HTML -->
<form id="f" novalidate style="max-width:640px">
<div class="fg"><label for="name">Full name</label><div class="line"><input type="text" id="name" name="name" placeholder="Your full name"><small class="msg" id="e_name"></small></div></div>
<div class="fg"><label for="email">Email</label><div class="line"><input type="email" id="email" name="email" placeholder="name@example.com"><small class="msg" id="e_email"></small></div></div>
<div class="fg"><label for="phone">Mobile number</label><div class="line"><input type="tel" id="phone" name="phone" placeholder="10 digits"><small class="msg" id="e_phone"></small></div></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const form = $('f');
const kinds = {"name": "text", "email": "email", "phone": "tel", "dob": "date", "gender": "radio", "course": "select", "hobbies": "checks", "password": "password", "confirm": "password", "address": "textarea", "agree": "check"};
const rules = {
  name: (v, get) => !v ? 'Name is required' : v.length < 3 ? 'Name must have at least 3 characters' : /^[A-Za-z ]+$/.test(v) ? '' : 'Only alphabets and spaces are allowed',
  email: (v, get) => !v ? 'Email is required' : /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v) ? '' : 'Enter a valid email like name@example.com',
  phone: (v, get) => !v ? 'Mobile number is required' : /^[6-9]\d{9}$/.test(v) ? '' : 'Enter 10 digits starting with 6, 7, 8 or 9',
  dob: (v, get) => !v ? 'Date of birth is required' : new Date(v) > new Date() ? 'Date of birth cannot be in the future' : ageFrom(v) < 16 ? 'You must be at least 16 years old' : '',
  gender: (v, get) => v ? '' : 'Please choose one option',
  course: (v, get) => v ? '' : 'Please select an option',
  hobbies: (v, get) => v.length ? '' : 'Select at least one option',
  password: (v, get) => !v ? 'Password is required' : v.length < 8 ? 'At least 8 characters' : !/[A-Z]/.test(v) ? 'Add an uppercase letter' : !/[a-z]/.test(v) ? 'Add a lowercase letter' : !/\d/.test(v) ? 'Add a digit' : !/[^A-Za-z0-9]/.test(v) ? 'Add a special character (!@#$...)' : '',
  confirm: (v, get) => !v ? 'Please confirm the password' : v === get('password') ? '' : 'Passwords do not match',
  address: (v, get) => !v ? 'This field is required' : v.length < 10 ? 'Write at least 10 characters (' + v.length + '/10)' : '',
  agree: (v, get) => v ? '' : 'You must accept the terms'
};
const ageFrom = (iso) => {
  const d = new Date(iso), now = new Date();
  let age = now.getFullYear() - d.getFullYear();
  if (now < new Date(now.getFullYear(), d.getMonth(), d.getDate())) age--;
  return age;
};
function getVal(name) {
  const kind = kinds[name];
  if (kind === 'radio') { const c = form.querySelector(`input[name="${name}"]:checked`); return c ? c.value : ''; }
  if (kind === 'checks') return [...form.querySelectorAll(`input[name="${name}"]:checked`)].map((c) => c.value);
  if (kind === 'check') return form.elements[name].checked;
  return form.elements[name].value.trim();
}
function showState(name, msg) {
  const box = $('e_' + name);
  box.textContent = msg;
  box.classList.remove('good');
  const el = form.elements[name];
  if (el && el.classList) { el.classList.toggle('invalid', msg !== ''); el.classList.toggle('valid', msg === '' && getVal(name) !== ''); }
}
function validateField(name) {
  const msg = rules[name](getVal(name), getVal);
  showState(name, msg);
  return msg === '';
}

form.addEventListener('submit', (e) => {
  e.preventDefault();                       // stop the page from reloading / submitting
  let ok = true, firstBad = null;
  Object.keys(rules).forEach((name) => { if (!validateField(name)) { ok = false; firstBad = firstBad || name; } });
  const banner = $('banner');
  if (ok) {
    banner.className = 'banner good';
    banner.textContent = 'Registration successful! All fields are valid.';
    form.reset();
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
  } else {
    banner.className = 'banner bad';
    banner.textContent = 'Please correct the highlighted fields.';
    const el = form.elements[firstBad];
    if (el && el.focus) el.focus();
  }
});
form.addEventListener('reset', () => {
  setTimeout(() => {
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
    $('banner').className = ''; $('banner').textContent = '';
  });
});
