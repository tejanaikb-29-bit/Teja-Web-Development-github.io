Marks entry form: total, average and grade

Enter marks for six subjects. The form validates each mark, then shows the total, average, grade and result.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="f" novalidate><label class="field" style="max-width:280px">Student name<input id="name" type="text" value="Priya"></label><div class="grid6" id="grid"></div><button type="submit">Calculate</button> <button type="reset" class="secondary">Clear</button></form><p class="err" id="err"></p><div id="res"></div>

/* CSS */
.demo .grid6 { display: grid; grid-template-columns: repeat(auto-fit, minmax(150px, 1fr)); gap: .8rem; margin: .8rem 0; }

// JavaScript
const $ = (id) => document.getElementById(id);
const subs = ['Tamil', 'English', 'Maths', 'Physics', 'Chemistry', 'Computer Science'];
subs.forEach((s, i) => { const l = document.createElement('label'); l.className = 'field'; l.innerHTML = `${s}<input type="number" class="m" min="0" max="100" value="${[78, 82, 91, 74, 69, 95][i]}">`; $('grid').appendChild(l); });
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  const marks = [...document.querySelectorAll('.m')].map((i) => (i.value === '' ? NaN : Number(i.value)));
  const bad = subs.filter((s, i) => Number.isNaN(marks[i]) || marks[i] < 0 || marks[i] > 100);
  document.querySelectorAll('.m').forEach((i, k) => i.classList.toggle('invalid', bad.includes(subs[k])));
  if (bad.length) { $('err').textContent = 'Enter marks from 0 to 100 for: ' + bad.join(', '); $('res').innerHTML = ''; return; }
  $('err').textContent = '';
  const total = marks.reduce((a, b) => a + b, 0), avg = total / marks.length, failed = marks.some((m) => m < 35);
  const grade = failed ? 'F' : avg >= 90 ? 'O' : avg >= 80 ? 'A+' : avg >= 70 ? 'A' : avg >= 60 ? 'B' : avg >= 50 ? 'C' : 'D';
  $('res').innerHTML = `<table class="data" style="max-width:420px"><tr><th>Student</th><td></td></tr><tr><th>Total</th><td>${total} / ${marks.length * 100}</td></tr><tr><th>Average</th><td>${avg.toFixed(2)}</td></tr><tr><th>Grade</th><td><strong>${grade}</strong></td></tr><tr><th>Result</th><td class="${failed ? 'err' : 'ok'}"><strong>${failed ? 'FAIL' : 'PASS'}</strong></td></tr></table>`;
  $('res').querySelector('td').textContent = $('name').value || 'Student';
});
