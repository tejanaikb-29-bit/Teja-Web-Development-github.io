Validate checkbox selection

At least two interests must be ticked. A single terms checkbox must also be checked.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="f" novalidate style="max-width:640px">
<div class="fg"><label for="interests">Interests (choose at least 2)</label><div class="line"><label><input type="checkbox" name="interests" value="HTML"> HTML</label> <label><input type="checkbox" name="interests" value="CSS"> CSS</label> <label><input type="checkbox" name="interests" value="JavaScript"> JavaScript</label> <label><input type="checkbox" name="interests" value="Python"> Python</label> <label><input type="checkbox" name="interests" value="Databases"> Databases</label> <small class="msg" id="e_interests"></small></div></div>
<div class="fg"><label for="agree"></label><div class="line"><label><input type="checkbox" id="agree" name="agree"> I accept the terms</label><small class="msg" id="e_agree"></small></div></div>
<button type="submit">Subscribe</button> <button type="reset" class="secondary">Reset</button>
</form>
<div id="banner" role="status"></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const form = $('f');
const kinds = {"interests": "checks", "agree": "check"};
const rules = {
  interests: (v, get) => v.length >= 2 ? '' : 'Select at least two interests (' + v.length + ' selected)',
  agree: (v, get) => v ? '' : 'You must accept the terms'
};
const ageFrom = (iso) => {
  const d = new Date(iso), now = new Date();
  let age = now.getFullYear() - d.getFullYear();
  if (now < new Date(now.getFullYear(), d.getMonth(), d.getDate())) age--;
  return age;
};
function getVal(name) {
  const kind = kinds[name];
  if (kind === 'radio') { const c = form.querySelector(`input[name="${name}"]:checked`); return c ? c.value : ''; }
  if (kind === 'checks') return [...form.querySelectorAll(`input[name="${name}"]:checked`)].map((c) => c.value);
  if (kind === 'check') return form.elements[name].checked;
  return form.elements[name].value.trim();
}
function showState(name, msg) {
  const box = $('e_' + name);
  box.textContent = msg;
  box.classList.remove('good');
  const el = form.elements[name];
  if (el && el.classList) { el.classList.toggle('invalid', msg !== ''); el.classList.toggle('valid', msg === '' && getVal(name) !== ''); }
}
function validateField(name) {
  const msg = rules[name](getVal(name), getVal);
  showState(name, msg);
  return msg === '';
}

const EVENTS = ["change"];
const logBox = $('log');
function log(text) { if (logBox) logBox.textContent = text + '\n' + logBox.textContent.split('\n').slice(0, 5).join('\n'); }
Object.keys(rules).forEach((name) => {
  document.querySelectorAll(`[name="${name}"]`).forEach((el) => {
    EVENTS.forEach((type) => {
      el.addEventListener(type, () => {
        log(`${type} on "${name}"`);
        if (type === 'focus') { showState(name, ''); el.classList && el.classList.add('valid'); if (el.classList) el.classList.remove('invalid'); }
        else validateField(name);
      });
    });
  });
});

form.addEventListener('submit', (e) => {
  e.preventDefault();                       // stop the page from reloading / submitting
  let ok = true, firstBad = null;
  Object.keys(rules).forEach((name) => { if (!validateField(name)) { ok = false; firstBad = firstBad || name; } });
  const banner = $('banner');
  if (ok) {
    banner.className = 'banner good';
    banner.textContent = 'Subscription saved.';
    form.reset();
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
  } else {
    banner.className = 'banner bad';
    banner.textContent = 'Please correct the highlighted fields.';
    const el = form.elements[firstBad];
    if (el && el.focus) el.focus();
  }
});
form.addEventListener('reset', () => {
  setTimeout(() => {
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
    $('banner').className = ''; $('banner').textContent = '';
  });
});
