<!-- HTML -->
<form id="f" novalidate style="max-width:520px">
  <div class="fg"><label>Name</label><input id="name" type="text"><small class="msg" id="e_name"></small></div>
  <div class="fg"><label>Email</label><input id="email" type="email"><small class="msg" id="e_email"></small></div>
  <div class="fg"><label>Course</label><select id="course"><option value="">-- Select --</option><option>BCA</option><option>B.Sc CS</option><option>B.Com</option></select><small class="msg" id="e_course"></small></div>
  <div class="fg"><label>Message</label><textarea id="msg" rows="3" style="width:100%"></textarea><small class="msg" id="e_msg"></small></div>
  <button type="submit">Preview</button>
</form>
<div class="preview" id="pv" hidden><h3 style="margin-top:0">Preview your details</h3><dl id="dl"></dl><div class="row"><button id="edit" class="secondary">Edit</button><button id="ok" class="success">Confirm &amp; submit</button></div></div>
<div class="banner good" id="done" hidden>Submitted successfully. Thank you!</div>

/* CSS */
.demo .preview { background: #f7f9fe; border: 2px solid #1f419a; border-radius: 12px; padding: 1rem 1.2rem; } .demo .preview dl { display: grid; grid-template-columns: 120px 1fr; gap: .4rem .8rem; } .demo .preview dt { font-weight: 700; color: #55637d; } .demo .preview dd { margin: 0; }

// JavaScript
const $ = (id) => document.getElementById(id);
const rules = { name: (v) => /^[A-Za-z ]{3,}$/.test(v) ? '' : 'Enter your name (letters, min 3)', email: (v) => /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v) ? '' : 'Enter a valid email', course: (v) => v ? '' : 'Select a course', msg: (v) => v.length >= 10 ? '' : 'Message: at least 10 characters' };
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  let ok = true;
  Object.entries(rules).forEach(([id, fn]) => { const m = fn($(id).value.trim()); $('e_' + id).textContent = m; $(id).classList.toggle('invalid', !!m); if (m) ok = false; });
  if (!ok) return;
  $('dl').innerHTML = '';
  [['Name', 'name'], ['Email', 'email'], ['Course', 'course'], ['Message', 'msg']].forEach(([label, id]) => {
    const dt = document.createElement('dt'), dd = document.createElement('dd');
    dt.textContent = label; dd.textContent = $(id).value.trim();
    $('dl').append(dt, dd);
  });
  $('f').hidden = true; $('pv').hidden = false;
});
$('edit').addEventListener('click', () => { $('pv').hidden = true; $('f').hidden = false; });
$('ok').addEventListener('click', () => { $('pv').hidden = true; $('done').hidden = false; $('f').reset(); });
