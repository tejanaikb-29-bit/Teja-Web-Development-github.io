<!-- HTML -->
<table class="data"><thead><tr><th>Product</th><th>Price</th><th>Qty</th><th>Amount</th></tr></thead><tbody id="items"></tbody></table>
<div class="row"><label class="field">Coupon code<input id="coupon" type="text" placeholder="SAVE10"></label><button id="apply" class="secondary">Apply</button><span id="cmsg" class="muted"></span></div>
<table class="data totals" style="max-width:420px;margin-top:.6rem"><tr><td>Subtotal</td><td id="sub"></td></tr><tr><td>Discount</td><td id="disc"></td></tr><tr><td>GST (18%)</td><td id="gst"></td></tr><tr><td>Delivery</td><td id="ship"></td></tr><tr class="grand"><td>Grand total</td><td id="total"></td></tr></table>

/* CSS */

// JavaScript
const $ = (id) => document.getElementById(id);
const products = [['Notebook', 40], ['Pen pack', 60], ['Backpack', 899], ['Water bottle', 350], ['Headphones', 1299]];
let discountRate = 0;
products.forEach(([name, price], i) => {
  const tr = $('items').insertRow();
  tr.insertCell().textContent = name;
  tr.insertCell().textContent = price.toFixed(2);
  const q = tr.insertCell();
  q.innerHTML = `<input type="number" class="qtyin" min="0" max="20" value="${i < 2 ? 1 : 0}">`;
  tr.insertCell().className = 'amt';
  q.firstChild.addEventListener('input', update);
});
function update() {
  let subtotal = 0;
  [...$('items').rows].forEach((tr, i) => {
    const qty = Math.max(0, Math.min(20, parseInt(tr.querySelector('input').value) || 0));
    const amount = qty * products[i][1];
    tr.querySelector('.amt').textContent = amount.toFixed(2);
    subtotal += amount;
  });
  const discount = subtotal * discountRate;
  const taxable = subtotal - discount;
  const gst = taxable * 0.18;
  const ship = subtotal === 0 || taxable >= 500 ? 0 : 49;
  const f = (n) => n.toFixed(2);
  $('sub').textContent = f(subtotal); $('disc').textContent = '- ' + f(discount); $('gst').textContent = f(gst);
  $('ship').textContent = ship ? f(ship) : 'FREE (orders of 500 or more)'; $('total').textContent = f(taxable + gst + ship);
}
$('apply').addEventListener('click', () => {
  const code = $('coupon').value.trim().toUpperCase();
  discountRate = code === 'SAVE10' ? 0.10 : code === 'SAVE20' ? 0.20 : 0;
  $('cmsg').textContent = discountRate ? `Coupon applied: ${discountRate * 100}% off` : 'Invalid coupon (try SAVE10 or SAVE20)';
  $('cmsg').style.color = discountRate ? '#178a4a' : '#c62828';
  update();
});
update();
