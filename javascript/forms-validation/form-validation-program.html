Form validation program

A basic form checked with JavaScript before submission: name, email and age are validated.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="f" novalidate style="max-width:640px">
<div class="fg"><label for="name">Name</label><div class="line"><input type="text" id="name" name="name" placeholder="Your name"><small class="msg" id="e_name"></small></div></div>
<div class="fg"><label for="email">Email</label><div class="line"><input type="email" id="email" name="email" placeholder="name@example.com"><small class="msg" id="e_email"></small></div></div>
<div class="fg"><label for="age">Age</label><div class="line"><input type="number" id="age" name="age" placeholder="18 - 60"><small class="msg" id="e_age"></small></div></div>
<button type="submit">Validate & submit</button> <button type="reset" class="secondary">Reset</button>
</form>
<div id="banner" role="status"></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const form = $('f');
const kinds = {"name": "text", "email": "email", "age": "number"};
const rules = {
  name: (v, get) => !v ? 'Name is required' : /^[A-Za-z ]+$/.test(v) ? '' : 'Only alphabets and spaces are allowed',
  email: (v, get) => !v ? 'Email is required' : !v.includes('@') ? 'Email must contain @' : v.indexOf('@') < 1 ? 'Add text before @' : !v.split('@')[1].includes('.') ? 'Add a domain like example.com' : '',
  age: (v, get) => !v ? 'Age is required' : !Number.isInteger(Number(v)) ? 'Age must be a whole number' : Number(v) < 18 || Number(v) > 60 ? 'Age must be between 18 and 60' : ''
};
const ageFrom = (iso) => {
  const d = new Date(iso), now = new Date();
  let age = now.getFullYear() - d.getFullYear();
  if (now < new Date(now.getFullYear(), d.getMonth(), d.getDate())) age--;
  return age;
};
function getVal(name) {
  const kind = kinds[name];
  if (kind === 'radio') { const c = form.querySelector(`input[name="${name}"]:checked`); return c ? c.value : ''; }
  if (kind === 'checks') return [...form.querySelectorAll(`input[name="${name}"]:checked`)].map((c) => c.value);
  if (kind === 'check') return form.elements[name].checked;
  return form.elements[name].value.trim();
}
function showState(name, msg) {
  const box = $('e_' + name);
  box.textContent = msg;
  box.classList.remove('good');
  const el = form.elements[name];
  if (el && el.classList) { el.classList.toggle('invalid', msg !== ''); el.classList.toggle('valid', msg === '' && getVal(name) !== ''); }
}
function validateField(name) {
  const msg = rules[name](getVal(name), getVal);
  showState(name, msg);
  return msg === '';
}

form.addEventListener('submit', (e) => {
  e.preventDefault();                       // stop the page from reloading / submitting
  let ok = true, firstBad = null;
  Object.keys(rules).forEach((name) => { if (!validateField(name)) { ok = false; firstBad = firstBad || name; } });
  const banner = $('banner');
  if (ok) {
    banner.className = 'banner good';
    banner.textContent = 'All fields are valid. Form accepted!';
    form.reset();
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
  } else {
    banner.className = 'banner bad';
    banner.textContent = 'Please correct the highlighted fields.';
    const el = form.elements[firstBad];
    if (el && el.focus) el.focus();
  }
});
form.addEventListener('reset', () => {
  setTimeout(() => {
    Object.keys(rules).forEach((name) => showState(name, ''));
    document.querySelectorAll('.valid').forEach((el) => el.classList.remove('valid'));
    $('banner').className = ''; $('banner').textContent = '';
  });
});
