<!-- HTML -->
<div class="row">

    <input id="s" type="text" value="Contact Asha at asha@example.com or ravi.k@mail.org before 12 May 2026.">

<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  const s = txt('s');
  const digits = s.match(/\d+/g);
  const capitals = s.match(/\b[A-Z][a-z]+\b/g);
  const emails = s.match(/[\w.]+@[\w.]+\.[a-z]{2,}/gi);
  const yearMatch = /(\d{4})/.exec(s);
  show(`Text: ${s}

  /\\d+/g            -> ${digits ? digits.join(', ') : 'no match'}
  /\\b[A-Z][a-z]+\\b/g -> ${capitals ? capitals.join(', ') : 'no match'}
  e-mail pattern     -> ${emails ? emails.join(', ') : 'no match'}
  /(\\d{4})/.exec()   -> ${yearMatch ? yearMatch[1] + ' at index ' + yearMatch.index : 'no match'}
  /^Contact/.test()  -> ${/^Contact/.test(s)}
  replace(/\\d/g,'#') -> ${s.replace(/\d/g, '#')}
  replace(/\\s+/g,'_')-> ${s.replace(/\s+/g, '_')}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
