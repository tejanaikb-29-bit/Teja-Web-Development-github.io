Date and Math objects

Reads the current date and time with the Date object and shows common Math functions for a number you enter.

Working demonstration

The result will appear here.
View source code of this program 
<!-- HTML -->
<div class="row">
  <label class="field">Enter a number
    <input id="n" type="number" value="17.6" step="any">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('n'))) { show('Please enter valid input.'); return; }
  const n = num('n');
  const d = new Date();
  show(`--- Date object ---
  Full date   : ${d.toString()}
  Year        : ${d.getFullYear()}
  Month       : ${d.getMonth() + 1}  (getMonth() starts at 0)
  Day of month: ${d.getDate()}
  Weekday     : ${['Sun','Mon','Tue','Wed','Thu','Fri','Sat'][d.getDay()]}
  Time        : ${d.toLocaleTimeString()}
  Timestamp   : ${d.getTime()} ms since 1970

  --- Math object for ${n} ---
  Math.round  : ${Math.round(n)}
  Math.floor  : ${Math.floor(n)}
  Math.ceil   : ${Math.ceil(n)}
  Math.abs    : ${Math.abs(n)}
  Math.sqrt   : ${Math.sqrt(n).toFixed(4)}
  Math.pow(n,2): ${Math.pow(n, 2)}
  Math.max(n,100): ${Math.max(n, 100)}
  Math.PI     : ${Math.PI}
  Math.random : ${Math.random()}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
