// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  let count = 1;
  count = 2;                                   // let can be reassigned
  const PI = 3.14159;
  let constMessage;
  try { PI = 3; } catch (e) { constMessage = e.name + ': ' + e.message; }   // const cannot
  {
    let blockScoped = 'only visible inside this block';
  }
  const student = { name: 'Asha', age: 20, city: 'Chennai' };
  const { name, age, country = 'India' } = student;       // object destructuring with a default value
  const [first, second, ...others] = ['HTML', 'CSS', 'JavaScript', 'Git'];   // array destructuring
  show(`let count       -> ${count}
  const PI        -> ${PI}
  Assign to const -> ${constMessage}
  Template literal: ${name} is ${age} years old and lives in ${country}.
  Destructured array: first=${first}, second=${second}, others=[${others}]
  Expression in template: 2 + 3 = ${2 + 3}`);
}
$('run').addEventListener('click', run);
