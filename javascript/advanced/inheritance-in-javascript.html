Inheritance in JavaScript

Dog and Cat classes extend Animal, reuse its code with super and override the speak() method.

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };
class Animal {
  constructor(name) { this.name = name; }
  speak() { return `${this.name} makes a sound.`; }
  describe() { return `${this.constructor.name}: ${this.name}`; }
}
class Dog extends Animal {
  constructor(name, breed) { super(name); this.breed = breed; }     // call the parent constructor
  speak() { return `${this.name} the ${this.breed} barks: Woof!`; }   // method overriding
}
class Cat extends Animal {
  speak() { return `${super.speak()} It purrs softly.`; }          // extend the parent method
}

function run() {
  const animals = [new Animal('Generic'), new Dog('Buddy', 'Labrador'), new Cat('Misty')];
  show(animals.map((a) => `${a.describe()}\n   -> ${a.speak()}   (instanceof Animal: ${a instanceof Animal})`).join('\n\n'));
}
$('run').addEventListener('click', run);
