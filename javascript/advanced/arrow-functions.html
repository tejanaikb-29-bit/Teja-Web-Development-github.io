Program using arrow functions

Arrow functions give a short syntax: (params) => expression. Used here with map, filter and reduce.

<!-- HTML -->
<div class="row">
  <label class="field">Numbers (comma separated)
    <input id="nums" type="text" value="1, 2, 3, 4, 5, 6" placeholder="e.g. 5, 12, 8, 3">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };
const add = (a, b) => a + b;             // two parameters
const square = (n) => n * n;             // one parameter, implicit return
const isEven = (n) => n % 2 === 0;
const greet = (name) => `Hello, ${name}!`;

function run() {
  if (list('nums').length === 0 || list('nums').some(Number.isNaN)) { show('Please enter valid input.'); return; }
  const nums = list('nums');
  show(`Numbers        : ${nums.join(', ')}
  Squares        : ${nums.map(square).join(', ')}
  Even numbers   : ${nums.filter(isEven).join(', ')}
  Sum (reduce)   : ${nums.reduce(add, 0)}
  greet('Ravi')  : ${greet('Ravi')}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
