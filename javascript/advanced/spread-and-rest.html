Spread and rest operators

... expands an array or object (spread) or collects many arguments into an array (rest).

Working demonstration

<!-- HTML -->
<div class="row">
  <label class="field">Numbers (comma separated)
    <input id="nums" type="text" value="4, 9, 2, 7" placeholder="e.g. 5, 12, 8, 3">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };
function sum(...values) { return values.reduce((t, v) => t + v, 0); }        // rest: collect arguments
function describe(label, ...rest) { return `${label} has ${rest.length} extra values: [${rest}]`; }

function run() {
  if (list('nums').length === 0 || list('nums').some(Number.isNaN)) { show('Please enter valid input.'); return; }
  const nums = list('nums');
  const copy = [...nums];                              // spread: copy an array
  const merged = [...nums, 100, ...[1, 2]];            // spread: merge arrays
  const person = { name: 'Ravi', age: 21 };
  const updated = { ...person, age: 22, city: 'Madurai' };   // spread objects
  show(`Numbers          : [${nums}]
  Copy (spread)    : [${copy}]
  Merged           : [${merged}]
  Math.max(...nums): ${Math.max(...nums)}
  Object spread    : ${JSON.stringify(updated)}
  Rest parameters  : sum(${nums.join(', ')}) = ${sum(...nums)}
  Rest + first arg : ${describe('Total', 1, 2, 3)}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
