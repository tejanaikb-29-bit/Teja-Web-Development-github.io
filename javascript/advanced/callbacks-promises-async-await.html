Callbacks, promises and async / await

Four ways to handle asynchronous code. Each step waits for a timer, and the log shows the order of events.

Working demonstration

Press the button.
View source code of this program 
<!-- HTML -->
<button id="run">Run demo</button>
<div class="out" id="out" style="font-family:Consolas,monospace">Press the button.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const log = (text) => { $('out').textContent += text + '\n'; };
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

// 1. Callback: a function passed to another function and called later
function loadData(callback) { setTimeout(() => callback('Data loaded with a callback'), 600); }

// 2. Promise: an object that resolves or rejects in the future
function checkNumber(n) {
  return new Promise((resolve, reject) => {
    setTimeout(() => (n > 0 ? resolve(`${n} is positive`) : reject(new Error(`${n} is not positive`))), 600);
  });
}

// 3. async / await: promise code that reads like normal code
async function main() {
  $('run').disabled = true;
  $('out').textContent = '';
  log('1) Callback...');
  await new Promise((done) => loadData((msg) => { log('   ' + msg); done(); }));

  log('2) Promise with .then() / .catch()...');
  await checkNumber(5).then((m) => log('   resolved: ' + m));
  await checkNumber(-2).catch((e) => log('   rejected: ' + e.message));

  log('3) async / await with try / catch...');
  try {
    log('   ' + await checkNumber(10));
    log('   ' + await checkNumber(0));
  } catch (error) {
    log('   caught: ' + error.message);
  } finally {
    log('   (finally always runs)');
  }

  log('4) Promise.all - run two timers together...');
  const results = await Promise.all([wait(300).then(() => 'A done'), wait(600).then(() => 'B done')]);
  log('   ' + results.join(', '));
  log('All finished.');
  $('run').disabled = false;
}
$('run').addEventListener('click', main);
