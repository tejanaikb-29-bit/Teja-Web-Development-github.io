A button that changes the font size dynamically

Increase, decrease or reset the font size of the paragraph. Limits are 10px to 60px.

Working demonstration

The quick brown fox jumps over the lazy dog.

Size: 20px
View source code of this program 
<!-- HTML -->
<p id="p" style="font-size:20px;transition:font-size .15s">The quick brown fox jumps over the lazy dog.</p><div class="row"><button id="inc">A+</button><button id="dec" class="secondary">A-</button><button id="rst" class="danger">Reset</button><span class="muted">Size: <strong id="sz">20</strong>px</span></div>

// JavaScript
const $ = (id) => document.getElementById(id);
let size = 20;
const apply = () => { $('p').style.fontSize = size + 'px'; $('sz').textContent = size; };
$('inc').addEventListener('click', () => { size = Math.min(60, size + 4); apply(); });
$('dec').addEventListener('click', () => { size = Math.max(10, size - 4); apply(); });
$('rst').addEventListener('click', () => { size = 20; apply(); });
