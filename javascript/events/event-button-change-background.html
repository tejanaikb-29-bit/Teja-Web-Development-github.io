A button that changes the background colour when clicked

Each click gives the box a new random background colour.

Working demonstration

Background colour:
#ffffff
View source code of this program 
<!-- HTML -->
<div class="pad" id="box" style="min-height:140px">Background colour: <br>#ffffff</div><button id="b">Change background colour</button>

/* CSS */
.demo .pad { display: grid; place-items: center; min-height: 120px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; font-weight: 700; text-align: center; user-select: none; padding: 1rem; transition: background .15s, transform .1s; }

// JavaScript
const $ = (id) => document.getElementById(id);
$('b').addEventListener('click', () => {
  const hue = Math.floor(Math.random() * 360);
  const color = `hsl(${hue}, 75%, 80%)`;
  $('box').style.background = color;
  $('box').innerHTML = `Background colour:<br>${color}`;
});
