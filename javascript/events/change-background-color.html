Change the background colour of a webpage

Buttons and a colour picker change document.body.style.backgroundColor.

Working demonstration

The whole page background changes when you click.

View source code of this program 
<!-- HTML -->
<div class="row">
  <button id="red" class="danger">Red</button><button id="green" class="success">Green</button><button id="blue">Blue</button>
  <button id="random" class="secondary">Random</button>
  <label class="field">Pick a colour<input id="picker" type="color" value="#ffffff"></label>
  <button id="reset" class="secondary">Reset</button>
</div>
<p>The whole page background changes when you click.</p>

// JavaScript
const $ = (id) => document.getElementById(id);
const setColor = (c) => { document.body.style.backgroundColor = c; };
$('red').onclick = () => setColor('#ffb3b3');
$('green').onclick = () => setColor('#b8e6c4');
$('blue').onclick = () => setColor('#b3d4ff');
$('random').onclick = () => setColor(`hsl(${Math.floor(Math.random() * 360)}, 70%, 82%)`);
$('picker').oninput = (e) => setColor(e.target.value);
$('reset').onclick = () => { document.body.style.backgroundColor = ''; };
