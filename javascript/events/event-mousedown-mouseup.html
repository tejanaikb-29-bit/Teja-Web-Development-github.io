onmousedown and onmouseup

mousedown fires when the button is pressed and mouseup when it is released, with the button number.

Working demonstration

Press and hold a mouse button here
Event log (newest first)
View source code of this program 
<!-- HTML -->
<div class="pad" id="box">Press and hold a mouse button here</div><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem;min-height:5rem">Event log (newest first)</div>

/* CSS */
.demo .pad { display: grid; place-items: center; min-height: 120px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; font-weight: 700; text-align: center; user-select: none; padding: 1rem; transition: background .15s, transform .1s; }

// JavaScript
const $ = (id) => document.getElementById(id);
const logBox = $('log');
function log(text) {
  const line = document.createElement('div');
  line.textContent = new Date().toLocaleTimeString() + '  ' + text;
  if (logBox.textContent.startsWith('Event log')) logBox.textContent = '';
  logBox.prepend(line);
  while (logBox.children.length > 8) logBox.lastElementChild.remove();
}

const names = ['left', 'middle', 'right'];
$('box').onmousedown = (e) => { $('box').style.background = '#ffb3b3'; $('box').textContent = 'Button DOWN'; log(`mousedown (${names[e.button]} button)`); };
$('box').onmouseup = (e) => { $('box').style.background = '#b8e6c4'; $('box').textContent = 'Button UP'; log(`mouseup (${names[e.button]} button)`); };
$('box').oncontextmenu = (e) => e.preventDefault();      // so right-click does not open a menu
