The onmousemove event

A dot follows the pointer inside the area and the coordinates relative to the area are displayed.

Working demonstration

Position inside the area: move the mouse

View source code of this program 
<!-- HTML -->
<div class="area" id="area"><div class="dot" id="dot" hidden></div></div><p>Position inside the area: <strong id="pos">move the mouse</strong></p>

/* CSS */
.demo .area { position: relative; height: 220px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; overflow: hidden; cursor: crosshair; } .demo .dot { position: absolute; width: 22px; height: 22px; border-radius: 50%; background: #d9481f; transform: translate(-50%, -50%); pointer-events: none; }

// JavaScript
const $ = (id) => document.getElementById(id);
$('area').addEventListener('mousemove', (e) => {
  const r = $('area').getBoundingClientRect();
  const x = Math.round(e.clientX - r.left), y = Math.round(e.clientY - r.top);
  $('dot').hidden = false;
  $('dot').style.left = x + 'px';
  $('dot').style.top = y + 'px';
  $('pos').textContent = `x = ${x}, y = ${y}`;
});
$('area').addEventListener('mouseleave', () => { $('dot').hidden = true; });
