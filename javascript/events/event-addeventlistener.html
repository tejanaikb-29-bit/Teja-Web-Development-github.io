Event listeners using addEventListener()

addEventListener(type, handler) attaches a handler. The same button reacts to click, mouseenter and mouseleave.

Working demonstration

Event log (newest first)
View source code of this program 
<!-- HTML -->
<button id="b" style="padding:1.2rem 2rem">Interact with me</button><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem;min-height:5rem">Event log (newest first)</div>

/* CSS */
.demo .pad { display: grid; place-items: center; min-height: 120px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; font-weight: 700; text-align: center; user-select: none; padding: 1rem; transition: background .15s, transform .1s; }

// JavaScript
const $ = (id) => document.getElementById(id);
const logBox = $('log');
function log(text) {
  const line = document.createElement('div');
  line.textContent = new Date().toLocaleTimeString() + '  ' + text;
  if (logBox.textContent.startsWith('Event log')) logBox.textContent = '';
  logBox.prepend(line);
  while (logBox.children.length > 8) logBox.lastElementChild.remove();
}

const b = $('b');
b.addEventListener('click', () => log('click'));
b.addEventListener('mouseenter', () => { b.style.background = '#178a4a'; log('mouseenter'); });
b.addEventListener('mouseleave', () => { b.style.background = ''; log('mouseleave'); });
