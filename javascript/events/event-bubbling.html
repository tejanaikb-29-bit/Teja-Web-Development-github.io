Event bubbling

A click on the inner element bubbles up: inner, then middle, then outer. Tick the box to stop it with stopPropagation().

Working demonstration

<!-- HTML -->
<div class="lv l1" id="outer"><span>Outer div</span><div class="lv l2" id="middle"><span>Middle div</span><div class="lv l3" id="inner">Inner div (click me)</div></div></div>
<label><input type="checkbox" id="stop"> stopPropagation() in the middle div</label><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem;min-height:5rem">Event log (newest first)</div>

/* CSS */
.demo .lv { padding: 1rem; border: 3px solid #14213d; border-radius: 10px; cursor: pointer; }
.demo .l1 { background: #dbe9ff; } .demo .l2 { background: #c4f0d2; } .demo .l3 { background: #ffe3a3; text-align: center; font-weight: 700; }
.demo .lv > span { display: block; font-weight: 700; margin-bottom: .5rem; }
.demo .flash { outline: 4px solid #d9481f; }

// JavaScript
const $ = (id) => document.getElementById(id);
const logBox = $('log');
function log(text) {
  const line = document.createElement('div');
  line.textContent = new Date().toLocaleTimeString() + '  ' + text;
  if (logBox.textContent.startsWith('Event log')) logBox.textContent = '';
  logBox.prepend(line);
  while (logBox.children.length > 8) logBox.lastElementChild.remove();
}

['inner', 'middle', 'outer'].forEach((id) => {
  $(id).addEventListener('click', (e) => {          // bubbling phase (default)
    log(`${id} handler (target was "${e.target.id}")`);
    $(id).classList.add('flash'); setTimeout(() => $(id).classList.remove('flash'), 400);
    if (id === 'middle' && $('stop').checked) { e.stopPropagation(); log('   stopPropagation() called - outer will not fire'); }
  });
});
