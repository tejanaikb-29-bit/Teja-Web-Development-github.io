Add multiple event listeners to a single element

One button has three separate click listeners. All three run, in the order they were added.

Working demonstration

    Clicks: 0

    View source code of this program 
    <!-- HTML -->
<button id="b">Click me (3 listeners)</button><ol id="steps" class="out" style="padding-left:2.2rem"></ol><p class="muted">Clicks: <strong id="n">0</strong></p>

// JavaScript
const $ = (id) => document.getElementById(id);
const b = $('b');
let n = 0;
b.addEventListener('click', () => { $('steps').innerHTML = ''; add('Listener 1: shows this list'); });
b.addEventListener('click', () => { n++; $('n').textContent = n; add('Listener 2: counts the click'); });
b.addEventListener('click', () => { b.style.transform = 'scale(1.1)'; setTimeout(() => (b.style.transform = ''), 150); add('Listener 3: animates the button'); });
function add(text) { const li = document.createElement('li'); li.textContent = text; $('steps').appendChild(li); }
