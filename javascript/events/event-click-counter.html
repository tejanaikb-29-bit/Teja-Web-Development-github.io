Click counter

Counts clicks on a big button and remembers the best clicks-in-10-seconds score.

Working demonstration

0

View source code of this program 
<!-- HTML -->
<div class="big" id="n">0</div><button id="hit">Click!</button><div class="row" style="justify-content:center"><button id="challenge" class="secondary">10 second challenge</button><button id="reset" class="danger">Reset</button></div><p class="center" id="msg"></p>

/* CSS */
.demo .big { font: 800 4rem 'Courier New', monospace; text-align: center; } .demo #hit { display: block; margin: 0 auto; padding: 1.4rem 3rem; font-size: 1.4rem; border-radius: 50px; }

// JavaScript
const $ = (id) => document.getElementById(id);
let count = 0, timerId = null, best = 0;
$('hit').addEventListener('click', () => { count++; $('n').textContent = count; });
$('reset').addEventListener('click', () => { count = 0; $('n').textContent = 0; $('msg').textContent = ''; clearInterval(timerId); timerId = null; });
$('challenge').addEventListener('click', () => {
  if (timerId) return;
  count = 0; $('n').textContent = 0;
  let left = 10;
  $('msg').textContent = `Go! ${left}s left`;
  timerId = setInterval(() => {
    left--;
    if (left > 0) { $('msg').textContent = `Go! ${left}s left`; return; }
    clearInterval(timerId); timerId = null;
    best = Math.max(best, count);
    $('msg').textContent = `Time! You clicked ${count} times (${(count / 10).toFixed(1)} per second). Best: ${best}`;
  }, 1000);
});
