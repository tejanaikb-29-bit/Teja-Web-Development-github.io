Keyboard-controlled moving object

Move the square with the arrow keys or W A S D. It stays inside the arena and leaves a step counter.

Working demonstration

Click the arena, then use arrow keys or W A S D. Steps: 0

View source code of this program 
<!-- HTML -->
<div class="arena" id="arena" tabindex="0"><div class="player" id="p" style="left:20px;top:20px"></div></div><p>Click the arena, then use <strong>arrow keys</strong> or <strong>W A S D</strong>. Steps: <strong id="steps">0</strong></p>

/* CSS */
.demo .arena { position: relative; height: 280px; border: 3px solid #14213d; border-radius: 12px; background: repeating-linear-gradient(0deg, #f1f4fa 0 20px, #e8edf8 20px 40px); overflow: hidden; outline: 0; } .demo .arena:focus { border-color: #d9481f; } .demo .player { position: absolute; width: 40px; height: 40px; background: #d9481f; border-radius: 8px; transition: left .06s linear, top .06s linear; }

// JavaScript
const $ = (id) => document.getElementById(id);
const STEP = 20;
let x = 20, y = 20, steps = 0;
const keys = { ArrowUp: [0, -1], ArrowDown: [0, 1], ArrowLeft: [-1, 0], ArrowRight: [1, 0], w: [0, -1], s: [0, 1], a: [-1, 0], d: [1, 0] };
$('arena').focus();
$('arena').addEventListener('keydown', (e) => {
  const move = keys[e.key];
  if (!move) return;
  e.preventDefault();
  const maxX = $('arena').clientWidth - 40, maxY = $('arena').clientHeight - 40;
  x = Math.min(maxX, Math.max(0, x + move[0] * STEP));
  y = Math.min(maxY, Math.max(0, y + move[1] * STEP));
  $('p').style.left = x + 'px';
  $('p').style.top = y + 'px';
  $('steps').textContent = ++steps;
});
