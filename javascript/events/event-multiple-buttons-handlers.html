Multiple buttons with different event handlers

Five buttons each have their own handler: greeting, colour change, counter, date and clearing.

Working demonstration

Output area
View source code of this program 
<!-- HTML -->
<div class="row"><button id="b1">Greet</button><button id="b2" class="secondary">Colour</button><button id="b3" class="success">Count</button><button id="b4">Show date</button><button id="b5" class="danger">Clear</button></div><div class="pad" id="box">Output area</div>

/* CSS */
.demo .pad { display: grid; place-items: center; min-height: 120px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; font-weight: 700; text-align: center; user-select: none; padding: 1rem; transition: background .15s, transform .1s; }

// JavaScript
const $ = (id) => document.getElementById(id);
let count = 0;
const box = $('box');
$('b1').onclick = () => { box.textContent = 'Hello, welcome to JavaScript events!'; };
$('b2').onclick = () => { box.style.background = `hsl(${Math.random() * 360}, 70%, 80%)`; };
$('b3').onclick = () => { box.textContent = 'Count: ' + ++count; };
$('b4').onclick = () => { box.textContent = new Date().toDateString(); };
$('b5').onclick = () => { box.textContent = 'Output area'; box.style.background = ''; count = 0; };
