Event delegation

One listener on the parent list handles clicks for every item, including items added later.

Working demonstration

  • Apples
  • Bananas
  • Cherries

Click an item to select it.

View source code of this program 
<!-- HTML -->
<div class="row"><input id="txt" type="text" placeholder="New item"><button id="add">Add item</button></div><ul id="list"><li>Apples</li><li>Bananas</li><li>Cherries</li></ul><p class="muted" id="msg">Click an item to select it.</p>

/* CSS */
.demo #list li { cursor: pointer; padding: .3rem .5rem; border-radius: 6px; } .demo #list li:hover { background: #eef2fb; } .demo #list li.sel { background: #fff2a8; font-weight: 700; }

// JavaScript
const $ = (id) => document.getElementById(id);
// ONE listener on the <ul> instead of one per <li>
$('list').addEventListener('click', (e) => {
  const li = e.target.closest('li');
  if (!li) return;
  li.classList.toggle('sel');
  $('msg').textContent = `Clicked: "${li.textContent}" (${document.querySelectorAll('#list li.sel').length} selected)`;
});
$('add').addEventListener('click', () => {
  const v = $('txt').value.trim();
  if (!v) return;
  const li = document.createElement('li');
  li.textContent = v;
  $('list').appendChild(li);          // works immediately, no new listener needed
  $('txt').value = '';
});
