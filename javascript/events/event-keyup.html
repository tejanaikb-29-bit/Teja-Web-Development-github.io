Keyboard events using keyup

keyup fires once when a key is released. The text below is updated only after each key is released.

<!-- HTML -->
<input id="inp" type="text" placeholder="Type here" style="width:100%;max-width:420px"><p>Length after last keyup: <strong id="len">0</strong> &middot; upper-case copy: <strong id="up"></strong></p><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem;min-height:5rem">Event log (newest first)</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const logBox = $('log');
function log(text) {
  const line = document.createElement('div');
  line.textContent = new Date().toLocaleTimeString() + '  ' + text;
  if (logBox.textContent.startsWith('Event log')) logBox.textContent = '';
  logBox.prepend(line);
  while (logBox.children.length > 8) logBox.lastElementChild.remove();
}

$('inp').addEventListener('keyup', (e) => {
  $('len').textContent = $('inp').value.length;
  $('up').textContent = $('inp').value.toUpperCase();
  log(`keyup -> key "${e.key}" released`);
});
