Mouse coordinate tracker

Shows the pointer position relative to the viewport, the page and the tracking box, updated on every mouse move.

Working demonstration

View source code of this program 
<!-- HTML -->
<div class="track" id="track">Move the mouse anywhere; click here to log a point</div><table class="data" style="margin-top:.8rem"><tr><th>clientX / clientY (viewport)</th><td id="c">-</td></tr><tr><th>pageX / pageY (whole page)</th><td id="p">-</td></tr><tr><th>Inside the box</th><td id="b">-</td></tr><tr><th>Last click</th><td id="k">-</td></tr></table>

/* CSS */
.demo .track { height: 200px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; display: grid; place-items: center; text-align: center; }

// JavaScript
const $ = (id) => document.getElementById(id);
document.addEventListener('mousemove', (e) => {
  $('c').textContent = `${e.clientX}, ${e.clientY}`;
  $('p').textContent = `${e.pageX}, ${e.pageY}`;
  const r = $('track').getBoundingClientRect();
  const inside = e.clientX >= r.left && e.clientX <= r.right && e.clientY >= r.top && e.clientY <= r.bottom;
  $('b').textContent = inside ? `${Math.round(e.clientX - r.left)}, ${Math.round(e.clientY - r.top)}` : 'outside';
});
$('track').addEventListener('click', (e) => {
  const r = $('track').getBoundingClientRect();
  $('k').textContent = `${Math.round(e.clientX - r.left)}, ${Math.round(e.clientY - r.top)}`;
});
