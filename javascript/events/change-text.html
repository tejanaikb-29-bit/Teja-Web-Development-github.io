Change text using JavaScript

Reads text from an input box and replaces the heading text with textContent.

Working demonstration

Original heading text

View source code of this program 
<!-- HTML -->
<h2 id="heading">Original heading text</h2>
<div class="row"><input id="newText" type="text" placeholder="Type new text" style="flex:1;min-width:200px">
<button id="change">Change text</button><button id="reset" class="secondary">Reset</button></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const original = $('heading').textContent;
$('change').addEventListener('click', () => {
  const value = $('newText').value.trim();
  if (value) $('heading').textContent = value;
});
$('reset').addEventListener('click', () => { $('heading').textContent = original; $('newText').value = ''; });
