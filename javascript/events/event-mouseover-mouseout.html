onmouseover and onmouseout

mouseover fires when the pointer enters the box and mouseout when it leaves.

Working demonstration

Move the mouse over me

Entered 0 times · left 0 times

View source code of this program 
<!-- HTML -->
<div class="pad" id="box">Move the mouse over me</div><p>Entered <strong id="in">0</strong> times &middot; left <strong id="outc">0</strong> times</p>

/* CSS */
.demo .pad { display: grid; place-items: center; min-height: 120px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; font-weight: 700; text-align: center; user-select: none; padding: 1rem; transition: background .15s, transform .1s; }

// JavaScript
const $ = (id) => document.getElementById(id);
let entered = 0, left = 0;
$('box').onmouseover = () => { $('box').style.background = '#b8e6c4'; $('box').textContent = 'Mouse is OVER the box'; $('in').textContent = ++entered; };
$('box').onmouseout = () => { $('box').style.background = ''; $('box').textContent = 'Mouse left the box'; $('outc').textContent = ++left; };
