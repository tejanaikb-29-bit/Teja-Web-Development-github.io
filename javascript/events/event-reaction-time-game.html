Reaction-time game

Wait for the box to turn green, then click as fast as you can. Clicking too early is a false start.

<!-- HTML -->
<div class="rt" id="rt">Click to start</div><p class="center">Last: <strong id="last">-</strong> &middot; Best: <strong id="best">-</strong> &middot; Average: <strong id="avg">-</strong></p>

/* CSS */
.demo .rt { display: grid; place-items: center; min-height: 200px; border-radius: 14px; font: 700 1.5rem 'Segoe UI', sans-serif; color: #fff; cursor: pointer; user-select: none; text-align: center; padding: 1rem; background: #1f419a; }

// JavaScript
const $ = (id) => document.getElementById(id);
let state = 'idle', startAt = 0, timeout = null, times = [];
const set = (text, color) => { $('rt').textContent = text; $('rt').style.background = color; };
$('rt').addEventListener('click', () => {
  if (state === 'idle') {
    state = 'waiting';
    set('Wait for green...', '#c62828');
    timeout = setTimeout(() => { state = 'go'; startAt = performance.now(); set('CLICK NOW!', '#178a4a'); }, 1000 + Math.random() * 2500);
  } else if (state === 'waiting') {
    clearTimeout(timeout);
    state = 'idle';
    set('Too soon! Click to try again.', '#d98a00');
  } else if (state === 'go') {
    const ms = Math.round(performance.now() - startAt);
    times.push(ms);
    state = 'idle';
    $('last').textContent = ms + ' ms';
    $('best').textContent = Math.min(...times) + ' ms';
    $('avg').textContent = Math.round(times.reduce((a, b) => a + b, 0) / times.length) + ' ms';
    set(`${ms} ms - click to play again`, '#1f419a');
  }
});
