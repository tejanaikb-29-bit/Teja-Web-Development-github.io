The change event

change fires when a value is committed: after leaving a text box, choosing a select option or ticking a checkbox.

Working demonstration

Event log (newest first)

<!-- HTML -->
<div class="row"><label class="field">Text (fires on blur)<input id="t" type="text"></label><label class="field">Select<select id="s"><option>Red</option><option>Green</option><option>Blue</option></select></label><label><input id="c" type="checkbox"> Subscribe</label><label class="field">Colour<input id="col" type="color"></label></div><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem;min-height:5rem">Event log (newest first)</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const logBox = $('log');
function log(text) {
  const line = document.createElement('div');
  line.textContent = new Date().toLocaleTimeString() + '  ' + text;
  if (logBox.textContent.startsWith('Event log')) logBox.textContent = '';
  logBox.prepend(line);
  while (logBox.children.length > 8) logBox.lastElementChild.remove();
}

$('t').addEventListener('change', (e) => log(`text changed to "${e.target.value}"`));
$('s').addEventListener('change', (e) => log(`select changed to "${e.target.value}"`));
$('c').addEventListener('change', (e) => log(`checkbox is now ${e.target.checked ? 'checked' : 'unchecked'}`));
$('col').addEventListener('change', (e) => log(`colour chosen: ${e.target.value}`));
