Display the key pressed by the user

Press any key on the keyboard and the key, its code and the key code are shown in large text.

Working demonstration

Press any key

Keys are captured anywhere on this page.

View source code of this program 
<!-- HTML -->
<div class="key" id="key"><span>Press any key</span></div><p class="muted center">Keys are captured anywhere on this page.</p>

/* CSS */
.demo .key { font: 800 4.5rem/1 'Courier New', monospace; text-align: center; padding: 1.2rem; background: #101828; color: #5eead4; border-radius: 14px; min-height: 8rem; display: grid; place-items: center; } .demo .key small { font-size: 1rem; color: #94a3b8; display: block; }

// JavaScript
const $ = (id) => document.getElementById(id);
document.addEventListener('keydown', (e) => {
  const shown = e.key === ' ' ? 'Space' : e.key;
  $('key').innerHTML = `<span></span>`;
  $('key').firstChild.textContent = shown.length > 8 ? shown.slice(0, 8) : shown;
  const small = document.createElement('small');
  small.textContent = `key: "${e.key}"  |  code: ${e.code}  |  keyCode: ${e.keyCode}`;
  $('key').firstChild.appendChild(small);
  if (['Space', 'ArrowUp', 'ArrowDown'].includes(e.code)) e.preventDefault();     // stop page scrolling
});
