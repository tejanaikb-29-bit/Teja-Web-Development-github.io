Remove an event listener using removeEventListener()

The button has a listener that can be removed and re-added. removeEventListener needs the same function reference.

Working demonstration

Listener is ATTACHED. Click the target.
View source code of this program 
<!-- HTML -->
<button id="target" style="padding:1rem 1.6rem">Target button</button><div class="row"><button id="rm" class="danger">removeEventListener</button><button id="ad" class="success">addEventListener</button></div><div class="out" id="out">Listener is ATTACHED. Click the target.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
let clicks = 0;
function onTargetClick() {                       // a named function, so it can be removed later
  clicks++;
  $('out').textContent = `Listener ran. Target clicked ${clicks} time(s).`;
}
$('target').addEventListener('click', onTargetClick);
$('rm').addEventListener('click', () => { $('target').removeEventListener('click', onTargetClick); $('out').textContent = 'Listener REMOVED. Clicking the target now does nothing.'; });
$('ad').addEventListener('click', () => { $('target').removeEventListener('click', onTargetClick); $('target').addEventListener('click', onTargetClick); $('out').textContent = 'Listener ATTACHED again.'; });
