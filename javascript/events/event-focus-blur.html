The focus and blur events

focus fires when a field is selected and blur when the user leaves it. The field colour and log show both events.

Working demonstration

Event log (newest first)
View source code of this program 
<!-- HTML -->
<div class="row"><input class="fld" id="a" type="text" placeholder="First field"><input class="fld" id="b" type="text" placeholder="Second field"><input class="fld" id="c" type="text" placeholder="Third field"></div><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem;min-height:5rem">Event log (newest first)</div>

/* CSS */
.demo input.fld:focus { outline: 0; border-color: #178a4a; background: #f0fbf4; }

// JavaScript
const $ = (id) => document.getElementById(id);
const logBox = $('log');
function log(text) {
  const line = document.createElement('div');
  line.textContent = new Date().toLocaleTimeString() + '  ' + text;
  if (logBox.textContent.startsWith('Event log')) logBox.textContent = '';
  logBox.prepend(line);
  while (logBox.children.length > 8) logBox.lastElementChild.remove();
}

['a', 'b', 'c'].forEach((id) => {
  $(id).addEventListener('focus', () => log(`focus  -> field "${id}" received focus`));
  $(id).addEventListener('blur', () => log(`blur   -> field "${id}" lost focus (value: "${$(id).value}")`));
});
