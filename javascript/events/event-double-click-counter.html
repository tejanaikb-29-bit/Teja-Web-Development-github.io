Double-click counter

Counts double-clicks and single clicks separately, so you can see that a double-click also fires two single clicks.

Working demonstration

Double-click here
click events0

View source code of this program 
<!-- HTML -->
<div class="pad" id="box">Double-click here</div><table class="data" style="margin-top:.8rem;max-width:360px"><tr><th>click events</th><td id="c">0</td></tr><tr><th>dblclick events</th><td id="d">0</td></tr></table><button id="reset" class="danger" style="margin-top:.6rem">Reset</button>

/* CSS */
.demo .pad { display: grid; place-items: center; min-height: 120px; border: 3px dashed #8a97b3; border-radius: 12px; background: #f7f9fe; font-weight: 700; text-align: center; user-select: none; padding: 1rem; transition: background .15s, transform .1s; }

// JavaScript
const $ = (id) => document.getElementById(id);
let clicks = 0, doubles = 0;
$('box').addEventListener('click', () => { $('c').textContent = ++clicks; });
$('box').addEventListener('dblclick', () => { $('d').textContent = ++doubles; $('box').style.background = '#fff2a8'; setTimeout(() => ($('box').style.background = ''), 200); });
$('reset').addEventListener('click', () => { clicks = doubles = 0; $('c').textContent = 0; $('d').textContent = 0; });
