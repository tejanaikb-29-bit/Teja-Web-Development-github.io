Show / hide an HTML element

Toggles the display style of a box with a button.

Working demonstration

I am a box

Click the button to hide or show me.

View source code of this program 
<!-- HTML -->
<button id="toggle">Hide the box</button>
<div class="box" id="box"><h3 style="margin-top:0">I am a box</h3><p>Click the button to hide or show me.</p></div>

/* CSS */
.demo .box { padding: 1.2rem; background: #fff2c2; border: 2px solid #f2c200; border-radius: 10px; margin-top: 1rem; }

// JavaScript
const $ = (id) => document.getElementById(id);
$('toggle').addEventListener('click', () => {
  const box = $('box');
  const hidden = box.style.display === 'none';
  box.style.display = hidden ? 'block' : 'none';
  $('toggle').textContent = hidden ? 'Hide the box' : 'Show the box';
});
