Simple keyboard typing test

Type the sentence shown. Correct letters turn green, mistakes turn red. The timer starts on your first key press and results show WPM and accuracy.

Working demonstration

Time: 0.0s
Results appear when you finish.
View source code of this program 
<!-- HTML -->
<div class="sentence" id="sentence"></div><input id="inp" type="text" placeholder="Start typing here..." style="width:100%" autocomplete="off"><div class="row"><button id="new" class="secondary">New sentence</button><span class="muted">Time: <strong id="time">0.0</strong>s</span></div><div class="out" id="out">Results appear when you finish.</div>

/* CSS */
.demo .sentence { font: 500 1.4rem/1.8 'Courier New', monospace; padding: 1rem; background: #f7f9fe; border-radius: 10px; margin-bottom: .8rem; } .demo .sentence .ok { color: #178a4a; background: #e3f6ea; } .demo .sentence .bad { color: #c62828; background: #fdeaea; } .demo .sentence .cur { border-bottom: 3px solid #14213d; }

// JavaScript
const $ = (id) => document.getElementById(id);
const texts = ['the quick brown fox jumps over the lazy dog', 'practice makes a programmer better every day', 'javascript brings web pages to life', 'small steps every day lead to big results'];
let target = '', start = null, tick = null, finished = false;
function newTest() {
  target = texts[Math.floor(Math.random() * texts.length)];
  start = null; finished = false; clearInterval(tick);
  $('inp').value = ''; $('inp').disabled = false; $('inp').focus();
  $('time').textContent = '0.0'; $('out').textContent = 'Results appear when you finish.';
  render('');
}
function render(typed) {
  $('sentence').innerHTML = '';
  [...target].forEach((ch, i) => {
    const span = document.createElement('span');
    span.textContent = ch;
    if (i < typed.length) span.className = typed[i] === ch ? 'ok' : 'bad';
    else if (i === typed.length) span.className = 'cur';
    $('sentence').appendChild(span);
  });
}
$('inp').addEventListener('input', () => {
  if (finished) return;
  const typed = $('inp').value;
  if (!start) { start = performance.now(); tick = setInterval(() => { $('time').textContent = ((performance.now() - start) / 1000).toFixed(1); }, 100); }
  render(typed);
  if (typed.length >= target.length) {
    finished = true; clearInterval(tick); $('inp').disabled = true;
    const seconds = (performance.now() - start) / 1000;
    let correct = 0;
    [...target].forEach((ch, i) => { if (typed[i] === ch) correct++; });
    const wpm = Math.round((target.length / 5) / (seconds / 60));    // one "word" = 5 characters
    $('time').textContent = seconds.toFixed(1);
    $('out').textContent = `Finished in ${seconds.toFixed(1)} seconds\nSpeed: ${wpm} WPM\nAccuracy: ${Math.round((correct / target.length) * 100)}%`;
  }
});
$('new').addEventListener('click', newTest);
newTest();
