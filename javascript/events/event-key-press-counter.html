Count the number of keyboard key presses

Counts every key press on the page, and shows how often each key was pressed.

<!-- HTML -->
<div class="cnt" id="cnt">0</div><p class="center muted">Total key presses (click this page and start typing)</p><div class="out" id="detail">No keys yet.</div><button id="reset" class="danger">Reset</button>

/* CSS */
.demo .cnt { font: 800 4rem 'Courier New', monospace; text-align: center; }

// JavaScript
const $ = (id) => document.getElementById(id);
let total = 0, perKey = {};
document.addEventListener('keydown', (e) => {
  if (e.repeat) return;                                // count a held key only once
  if (e.target.tagName === 'BUTTON' && (e.key === 'Enter' || e.key === ' ')) return;
  total++;
  const k = e.key === ' ' ? 'Space' : e.key;
  perKey[k] = (perKey[k] || 0) + 1;
  $('cnt').textContent = total;
  $('detail').textContent = Object.entries(perKey).sort((a, b) => b[1] - a[1]).slice(0, 10).map(([k, n]) => `${k}: ${n}`).join('   ');
});
$('reset').addEventListener('click', () => { total = 0; perKey = {}; $('cnt').textContent = 0; $('detail').textContent = 'No keys yet.'; });
