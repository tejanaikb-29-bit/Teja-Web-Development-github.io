Keyboard events using keydown

keydown fires when a key is pressed down (and repeats while it is held). Type in the box to see key details.

Working demonstration

Event log (newest first)
View source code of this program 
<!-- HTML -->
<input id="inp" type="text" placeholder="Click here and press keys" style="width:100%;max-width:420px"><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem;min-height:5rem">Event log (newest first)</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const logBox = $('log');
function log(text) {
  const line = document.createElement('div');
  line.textContent = new Date().toLocaleTimeString() + '  ' + text;
  if (logBox.textContent.startsWith('Event log')) logBox.textContent = '';
  logBox.prepend(line);
  while (logBox.children.length > 8) logBox.lastElementChild.remove();
}

$('inp').addEventListener('keydown', (e) => {
  log(`keydown -> key: "${e.key}"  code: ${e.code}  repeat: ${e.repeat}${e.shiftKey ? '  +Shift' : ''}${e.ctrlKey ? '  +Ctrl' : ''}`);
});
