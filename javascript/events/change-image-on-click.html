Change an image when a button is clicked

Changes the src attribute of an <img> element each time the button is clicked.

Working demonstration

Image 1 of 4
View source code of this program 
<!-- HTML -->
<img id="pic" src="../assets/images/img1.svg" alt="Mountains" style="max-width:420px;border-radius:10px">
<div class="row"><button id="next">Next image</button><span class="muted" id="label">Image 1 of 4</span></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const images = [
  { src: '../assets/images/img1.svg', alt: 'Mountains' },
  { src: '../assets/images/img2.svg', alt: 'Sunset' },
  { src: '../assets/images/img3.svg', alt: 'Ocean' },
  { src: '../assets/images/img4.svg', alt: 'Forest' },
];
let i = 0;
$('next').addEventListener('click', () => {
  i = (i + 1) % images.length;
  $('pic').src = images[i].src;
  $('pic').alt = images[i].alt;
  $('label').textContent = `Image ${i + 1} of ${images.length}: ${images[i].alt}`;
});
