Drag-and-drop application

A small task board: drag task cards between the To Do, Doing and Done columns using the HTML5 drag-and-drop events.

<!-- HTML -->
<div class="board"><div class="col" data-col="todo"><h4>To Do</h4></div><div class="col" data-col="doing"><h4>Doing</h4></div><div class="col" data-col="done"><h4>Done</h4></div></div><p class="muted" id="msg">Drag a card to another column.</p>

/* CSS */
.demo .board { display: grid; grid-template-columns: repeat(3, 1fr); gap: .8rem; }
.demo .col { background: #eef2fb; border-radius: 10px; padding: .6rem; min-height: 200px; border: 2px dashed transparent; }
.demo .col.over { border-color: #1f419a; background: #dfe8fb; } .demo .col h4 { margin: 0 0 .5rem; }
.demo .task { background: #fff; border: 1px solid #ccd5e6; border-left: 5px solid #1f419a; border-radius: 8px; padding: .55rem .7rem; margin-bottom: .5rem; cursor: grab; }
.demo .task.dragging { opacity: .4; }
@media (max-width: 600px) { .demo .board { grid-template-columns: 1fr; } }

// JavaScript
const $ = (id) => document.getElementById(id);
const tasks = [['Write HTML page', 'todo'], ['Style with CSS', 'todo'], ['Add JavaScript', 'doing'], ['Test in browser', 'done']];
tasks.forEach(([t, col], i) => {
  const d = document.createElement('div');
  d.className = 'task'; d.draggable = true; d.id = 'task' + i; d.textContent = t;
  d.addEventListener('dragstart', (e) => { e.dataTransfer.setData('text/plain', d.id); d.classList.add('dragging'); });
  d.addEventListener('dragend', () => d.classList.remove('dragging'));
  document.querySelector(`[data-col=${col}]`).appendChild(d);
});
document.querySelectorAll('.col').forEach((col) => {
  col.addEventListener('dragover', (e) => { e.preventDefault(); col.classList.add('over'); });   // preventDefault allows dropping
  col.addEventListener('dragleave', () => col.classList.remove('over'));
  col.addEventListener('drop', (e) => {
    e.preventDefault();
    col.classList.remove('over');
    const card = $(e.dataTransfer.getData('text/plain'));
    col.appendChild(card);
    $('msg').textContent = `"${card.textContent}" moved to ${col.querySelector('h4').textContent}`;
  });
});
