Student marks / grade calculator

Enter marks for five subjects to get the total, percentage, grade and pass / fail result.

Working demonstration

Enter marks (0 - 100) and press Calculate.

<!-- HTML -->
<label class="field" style="max-width:300px">Student name<input id="name" type="text" value="Asha"></label>
<div class="marks" id="marks"></div>
<div class="row"><button id="calc">Calculate result</button><button id="clear" class="secondary">Clear</button></div>
<div class="out" id="out">Enter marks (0 - 100) and press Calculate.</div>
<table class="data" style="margin-top:1rem;max-width:420px"><tr><th>Percentage</th><th>Grade</th></tr>
<tr><td>90 and above</td><td>O</td></tr><tr><td>80 - 89</td><td>A+</td></tr><tr><td>70 - 79</td><td>A</td></tr><tr><td>60 - 69</td><td>B</td></tr><tr><td>50 - 59</td><td>C</td></tr><tr><td>Below 50</td><td>F</td></tr></table>

/* CSS */
.demo .marks { display: grid; grid-template-columns: repeat(auto-fit, minmax(150px, 1fr)); gap: .8rem; margin: .8rem 0; } .demo .grade { font-size: 3rem; font-weight: 800; }

// JavaScript
const $ = (id) => document.getElementById(id);
const subjects = ['Tamil / Language', 'English', 'Mathematics', 'Science', 'Computer Science'];
const defaults = [82, 76, 91, 68, 88];
subjects.forEach((s, i) => {
  const label = document.createElement('label');
  label.className = 'field';
  label.innerHTML = `${s}<input type="number" min="0" max="100" class="mark" value="${defaults[i]}">`;
  $('marks').appendChild(label);
});
function gradeOf(p) { return p >= 90 ? 'O' : p >= 80 ? 'A+' : p >= 70 ? 'A' : p >= 60 ? 'B' : p >= 50 ? 'C' : 'F'; }

$('calc').addEventListener('click', () => {
  const values = [...document.querySelectorAll('.mark')].map((i) => i.value.trim() === '' ? NaN : Number(i.value));
  if (values.some((v) => Number.isNaN(v) || v < 0 || v > 100)) { $('out').textContent = 'Every mark must be a number from 0 to 100.'; return; }
  const total = values.reduce((a, b) => a + b, 0);
  const percent = total / values.length;
  const failed = subjects.filter((s, i) => values[i] < 35);
  $('out').innerHTML = `<strong>${$('name').value || 'Student'}</strong><br>Total: ${total} / ${values.length * 100}<br>Percentage: ${percent.toFixed(2)}%<br>
    Grade: <span class="grade">${failed.length ? 'F' : gradeOf(percent)}</span><br>
    Result: <strong class="${failed.length ? 'err' : 'ok'}">${failed.length ? 'FAIL (below 35 in ' + failed.join(', ') + ')' : 'PASS'}</strong>`;
});
$('clear').addEventListener('click', () => {
  document.querySelectorAll('.mark').forEach((i) => { i.value = ''; });
  $('out').textContent = 'Enter marks (0 - 100) and press Calculate.';
});
