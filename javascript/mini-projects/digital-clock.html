Digital clock using JavaScript

Shows the current time and date, refreshed every second with setInterval().

Working demonstration

--:--:--

View source code of this program 
<!-- HTML -->
<div class="clock" id="clock">--:--:--</div>
<p class="center muted" id="date"></p>
<div class="center"><button id="fmt" class="secondary">Switch 12 / 24 hour</button></div>

// JavaScript
const $ = (id) => document.getElementById(id);
let use24 = true;

function tick() {
  const now = new Date();
  let hours = now.getHours();
  const minutes = String(now.getMinutes()).padStart(2, '0');
  const seconds = String(now.getSeconds()).padStart(2, '0');
  let suffix = '';
  if (!use24) {
    suffix = hours >= 12 ? ' PM' : ' AM';
    hours = hours % 12 || 12;
  }
  $('clock').textContent = `${String(hours).padStart(2, '0')}:${minutes}:${seconds}${suffix}`;
  $('date').textContent = now.toDateString();
}

tick();                       // show the time immediately
setInterval(tick, 1000);      // then update every 1000 ms
$('fmt').addEventListener('click', () => { use24 = !use24; tick(); });
