// JavaScript
const $ = (id) => document.getElementById(id);
const pf = $('pf');
// theme (saved)
if (localStorage.getItem('pf-theme') === 'dark') pf.classList.add('dark');
const setThemeLabel = () => { $('theme').textContent = pf.classList.contains('dark') ? 'Light mode' : 'Dark mode'; };
setThemeLabel();
$('theme').addEventListener('click', () => {
  pf.classList.toggle('dark');
  localStorage.setItem('pf-theme', pf.classList.contains('dark') ? 'dark' : 'light');
  setThemeLabel();
});
// smooth scroll inside the page
$('nav').addEventListener('click', (e) => {
  const b = e.target.closest('button[data-go]');
  if (!b) return;
  document.querySelectorAll('#nav button[data-go]').forEach((x) => x.classList.toggle('on', x === b));
  $(b.dataset.go).scrollIntoView({ behavior: 'smooth', block: 'start' });
});
// typing effect
const phrases = ['Front-end developer', 'HTML, CSS & JavaScript learner', 'Problem solver'];
let p = 0, c = 0, deleting = false;
function type() {
  const word = phrases[p];
  c += deleting ? -1 : 1;
  $('typed').textContent = word.slice(0, c);
  let delay = deleting ? 40 : 90;
  if (!deleting && c === word.length) { deleting = true; delay = 1200; }
  else if (deleting && c === 0) { deleting = false; p = (p + 1) % phrases.length; delay = 400; }
  setTimeout(type, delay);
}
type();
// skill bars animate when scrolled into view
const skills = [['HTML5', 92], ['CSS3', 85], ['JavaScript', 78], ['Git', 65]];
skills.forEach(([n, v]) => {
  const row = document.createElement('div');
  row.className = 'b';
  row.innerHTML = `<span>${n}</span><div class="track"><i data-w="${v}"></i></div><b>${v}%</b>`;
  $('bars').appendChild(row);
});
new IntersectionObserver((entries, obs) => {
  entries.forEach((en) => { if (en.isIntersecting) { en.target.querySelectorAll('i').forEach((i) => { i.style.width = i.dataset.w + '%'; }); obs.unobserve(en.target); } });
}, { threshold: 0.3 }).observe($('bars'));
// filterable projects
const projects = [
  { t: 'Digital Clock', k: 'JavaScript', img: 1 }, { t: 'Product Cards', k: 'CSS', img: 3 },
  { t: 'Landing Page', k: 'HTML', img: 4 }, { t: 'Quiz App', k: 'JavaScript', img: 5 }, { t: 'Photo Gallery', k: 'CSS', img: 7 },
];
let filter = 'All';
function renderProjects() {
  $('filters').innerHTML = '';
  ['All', 'HTML', 'CSS', 'JavaScript'].forEach((f) => {
    const b = document.createElement('button');
    b.textContent = f; if (f === filter) b.classList.add('on');
    b.addEventListener('click', () => { filter = f; renderProjects(); });
    $('filters').appendChild(b);
  });
  $('proj').innerHTML = '';
  projects.filter((x) => filter === 'All' || x.k === filter).forEach((x) => {
    const a = document.createElement('article');
    a.innerHTML = `<img src="../assets/images/img${x.img}.svg" alt=""><h4>${x.t}</h4><small>${x.k}</small>`;
    $('proj').appendChild(a);
  });
}
renderProjects();
// contact form validation
$('form').addEventListener('submit', (e) => {
  e.preventDefault();
  const okName = $('cn').value.trim().length >= 2;
  const okMail = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test($('ce').value.trim());
  const okMsg = $('cm').value.trim().length >= 10;
  $('cnE').textContent = okName ? '' : 'Please enter your name.';
  $('ceE').textContent = okMail ? '' : 'Enter a valid email.';
  $('cmE').textContent = okMsg ? '' : 'Message needs at least 10 characters.';
  $('ok').textContent = okName && okMail && okMsg ? 'Thank you! Your message is valid (demo - nothing is sent).' : '';
});
