Currency converter using an API

Converts between currencies using live rates from the free open.er-api.com service. If you are offline, sample rates are used.

Working demonstration

View source code of this program 
<!-- HTML -->
<div class="row">
  <label class="field">Amount<input id="amount" type="number" value="100" min="0" step="any" style="width:140px"></label>
  <label class="field">From<select id="from"></select></label>
  <button id="swap" class="secondary" title="Swap currencies">&#8646;</button>
  <label class="field">To<select id="to"></select></label>
</div>
<div class="out"><div class="big" id="result"></div><div class="muted" id="rate"></div><div class="muted" id="source"></div></div>

/* CSS */
.demo .big { font-size: 1.8rem; font-weight: 800; }

// JavaScript
const $ = (id) => document.getElementById(id);
// Sample rates (per 1 USD) used until / unless live data loads
let rates = { USD: 1, INR: 83.2, EUR: 0.92, GBP: 0.79, JPY: 150.5, AUD: 1.52, CAD: 1.36, CNY: 7.2, AED: 3.67, SGD: 1.35 };
let live = false;

function fillSelects() {
  const codes = Object.keys(rates).sort();
  [['from', 'USD'], ['to', 'INR']].forEach(([id, def]) => {
    const sel = $(id), keep = sel.value || def;
    sel.innerHTML = codes.map((c) => `<option value="${c}">${c}</option>`).join('');
    sel.value = codes.includes(keep) ? keep : def;
  });
}
function convert() {
  const amount = parseFloat($('amount').value);
  if (Number.isNaN(amount) || amount < 0) { $('result').textContent = 'Enter a valid amount'; return; }
  const from = $('from').value, to = $('to').value;
  const rate = rates[to] / rates[from];               // go through USD as the common base
  $('result').textContent = `${amount.toLocaleString()} ${from} = ${(amount * rate).toLocaleString(undefined, { maximumFractionDigits: 2 })} ${to}`;
  $('rate').textContent = `1 ${from} = ${rate.toFixed(4)} ${to}`;
  $('source').textContent = live ? 'Live rates loaded from open.er-api.com' : 'Using sample rates (could not reach the API)';
}
['amount', 'from', 'to'].forEach((id) => $(id).addEventListener('input', convert));
$('swap').addEventListener('click', () => { const f = $('from').value; $('from').value = $('to').value; $('to').value = f; convert(); });
fillSelects();
convert();
fetch('https://open.er-api.com/v6/latest/USD')
  .then((r) => r.json())
  .then((d) => { if (d.result === 'success') { rates = d.rates; live = true; fillSelects(); convert(); } })
  .catch(() => { /* keep the sample rates */ });
