Digital calculator

// JavaScript
const $ = (id) => document.getElementById(id);
let current = '0', previous = null, op = null, fresh = false;
const display = $('display');

const update = () => { display.textContent = current; };
function compute(a, b, o) {
  switch (o) {
    case '+': return a + b;
    case '-': return a - b;
    case '×': return a * b;
    case '÷': return b === 0 ? NaN : a / b;
  }
  return b;
}
function inputDigit(d) {
  if (fresh || current === 'Error') { current = '0'; fresh = false; }
  current = current === '0' ? d : current + d;
}
function inputDot() {
  if (fresh || current === 'Error') { current = '0'; fresh = false; }
  if (!current.includes('.')) current += '.';
}
function calculate() {
  if (op === null) return;
  const result = compute(previous, parseFloat(current), op);
  current = Number.isNaN(result) ? 'Error' : String(parseFloat(result.toFixed(10)));
  op = null;
  fresh = true;
}
function chooseOp(o) {
  if (current === 'Error') return;
  if (op !== null && !fresh) calculate();      // chain: 2 + 3 + 4
  previous = parseFloat(current);
  op = o;
  fresh = true;
}
function handle(action, value) {
  switch (action) {
    case 'digit': inputDigit(value); break;
    case 'dot': inputDot(); break;
    case 'op': chooseOp(value); break;
    case 'eq': calculate(); break;
    case 'clear': current = '0'; previous = null; op = null; fresh = false; break;
    case 'back': current = current.length > 1 && current !== 'Error' ? current.slice(0, -1) : '0'; break;
    case 'sign': if (current !== 'Error') current = String(-parseFloat(current)); break;
    case 'percent': if (current !== 'Error') current = String(parseFloat(current) / 100); break;
  }
  update();
}
// One listener on the keypad handles every button (event delegation)
$('pad').addEventListener('click', (e) => {
  const btn = e.target.closest('button');
  if (btn) handle(btn.dataset.a, btn.dataset.v);
});
// Keyboard support
document.addEventListener('keydown', (e) => {
  if (/^[0-9]$/.test(e.key)) handle('digit', e.key);
  else if (e.key === '.') handle('dot');
  else if (e.key === '+' || e.key === '-') handle('op', e.key);
  else if (e.key === '*') handle('op', '×');
  else if (e.key === '/') { e.preventDefault(); handle('op', '÷'); }
  else if (e.key === 'Enter' || e.key === '=') handle('eq');
  else if (e.key === 'Backspace') handle('back');
  else if (e.key === 'Escape') handle('clear');
});
update();
