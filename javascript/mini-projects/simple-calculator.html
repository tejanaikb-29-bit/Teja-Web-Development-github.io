Simple calculator using HTML, CSS and JavaScript

Enter two numbers, choose an operator and see the result.

Working demonstration

Result appears here
View source code of this program 
<!-- HTML -->
<div class="row">
  <label class="field">First number<input id="a" type="number" value="12" step="any"></label>
  <label class="field">Operator<select id="op"><option value="+">+ Add</option><option value="-">- Subtract</option><option value="*">x Multiply</option><option value="/">/ Divide</option><option value="%">% Modulus</option></select></label>
  <label class="field">Second number<input id="b" type="number" value="4" step="any"></label>
  <button id="calc">Calculate</button>
</div>
<div class="out"><span class="res" id="res">Result appears here</span></div>

/* CSS */
.demo .res { font-size: 1.6rem; font-weight: 700; color: #1f419a; }

// JavaScript
const $ = (id) => document.getElementById(id);
$('calc').addEventListener('click', () => {
  const a = parseFloat($('a').value), b = parseFloat($('b').value), op = $('op').value;
  if (Number.isNaN(a) || Number.isNaN(b)) { $('res').textContent = 'Enter both numbers'; return; }
  if ((op === '/' || op === '%') && b === 0) { $('res').textContent = 'Cannot divide by zero'; return; }
  let result;
  switch (op) {
    case '+': result = a + b; break;
    case '-': result = a - b; break;
    case '*': result = a * b; break;
    case '/': result = a / b; break;
    case '%': result = a % b; break;
  }
  $('res').textContent = `${a} ${op} ${b} = ${parseFloat(result.toFixed(10))}`;
});
