Simple quiz application

A multiple-choice quiz with instant feedback, a running score and a final result.

Working demonstration

View source code of this program 
<!-- HTML -->
<div id="quiz">
  <p class="muted" id="progress"></p>
  <h3 id="question"></h3>
  <div id="options"></div>
  <p id="feedback" style="min-height:1.6em;font-weight:700"></p>
  <button id="next" style="display:none">Next question</button>
</div>
<div id="result" style="display:none"></div>

/* CSS */
#options button { display: block; width: 100%; text-align: left; margin: .4rem 0; background: #fff; color: #1c2437; border: 2px solid #ccd5e6; }
#options button:hover:not(:disabled) { border-color: #2b59c3; background: #f1f5ff; }
#options button.right { background: #e6f6ec; border-color: #178a4a; }
#options button.wrong { background: #fdeaea; border-color: #c62828; }

// JavaScript
const $ = (id) => document.getElementById(id);
const questions = [
  { q: 'Which tag is used for the largest heading?', options: ['<h6>', '<h1>', '<head>', '<title>'], answer: 1 },
  { q: 'Which CSS property changes the text colour?', options: ['font-color', 'text-style', 'color', 'background'], answer: 2 },
  { q: 'Which keyword declares a constant in JavaScript?', options: ['var', 'let', 'const', 'static'], answer: 2 },
  { q: 'What does DOM stand for?', options: ['Document Object Model', 'Data Object Mode', 'Digital Ordered Map', 'Document Order Method'], answer: 0 },
  { q: 'Which method adds an item to the end of an array?', options: ['shift()', 'push()', 'pop()', 'unshift()'], answer: 1 },
  { q: 'Which symbol is used for an id selector in CSS?', options: ['.', '*', '#', '&'], answer: 2 }
];
let index = 0, score = 0;

function showQuestion() {
  const item = questions[index];
  $('progress').textContent = `Question ${index + 1} of ${questions.length}   |   Score: ${score}`;
  $('question').textContent = item.q;
  $('feedback').textContent = '';
  $('next').style.display = 'none';
  $('options').innerHTML = '';
  item.options.forEach((text, i) => {
    const btn = document.createElement('button');
    btn.textContent = text;
    btn.addEventListener('click', () => choose(i));
    $('options').appendChild(btn);
  });
}
function choose(i) {
  const item = questions[index];
  const buttons = $('options').querySelectorAll('button');
  buttons.forEach((b) => (b.disabled = true));
  buttons[item.answer].classList.add('right');
  if (i === item.answer) {
    score++;
    $('feedback').textContent = 'Correct!';
    $('feedback').className = 'ok';
  } else {
    buttons[i].classList.add('wrong');
    $('feedback').textContent = 'Wrong answer.';
    $('feedback').className = 'err';
  }
  $('next').textContent = index === questions.length - 1 ? 'See result' : 'Next question';
  $('next').style.display = 'inline-block';
}
function showResult() {
  $('quiz').style.display = 'none';
  const pct = Math.round((score / questions.length) * 100);
  $('result').style.display = 'block';
  $('result').innerHTML = `<h3>Quiz finished!</h3><p>You scored <strong>${score} / ${questions.length}</strong> (${pct}%).</p>
    <p>${pct >= 80 ? 'Excellent work!' : pct >= 50 ? 'Good effort, keep practising.' : 'Review the topics and try again.'}</p>
    <button id="again">Try again</button>`;
  $('again').addEventListener('click', () => {
    index = 0; score = 0;
    $('result').style.display = 'none';
    $('quiz').style.display = 'block';
    showQuestion();
  });
}
$('next').addEventListener('click', () => {
  index++;
  if (index < questions.length) showQuestion(); else showResult();
});
showQuestion();
