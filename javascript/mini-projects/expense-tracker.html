// JavaScript
const $ = (id) => document.getElementById(id);
let items = [];
try { items = JSON.parse(localStorage.getItem('expenses')) || []; } catch (e) { items = []; }
const money = (n) => n.toFixed(2);

function render() {
  const list = $('list');
  list.innerHTML = '';
  let inc = 0, exp = 0;
  items.forEach((it) => {
    if (it.type === 'income') inc += it.amount; else exp += it.amount;
    const li = document.createElement('li');
    li.className = it.type;
    li.innerHTML = '<div class="grow"><strong></strong><small></small></div><b></b>';
    li.querySelector('strong').textContent = it.desc;
    li.querySelector('small').textContent = it.date;
    li.querySelector('b').textContent = (it.type === 'income' ? '+' : '-') + money(it.amount);
    const del = document.createElement('button');
    del.textContent = 'Delete'; del.className = 'danger small';
    del.addEventListener('click', () => { items = items.filter((x) => x.id !== it.id); save(); render(); });
    li.appendChild(del);
    list.appendChild(li);
  });
  if (!items.length) list.innerHTML = '<li class="muted">No transactions yet.</li>';
  $('inc').textContent = money(inc);
  $('exp').textContent = money(exp);
  $('bal').textContent = money(inc - exp);
  $('bal').style.color = inc - exp < 0 ? '#c62828' : '#1c2437';
}
function save() { localStorage.setItem('expenses', JSON.stringify(items)); }

$('form').addEventListener('submit', (e) => {
  e.preventDefault();
  const desc = $('desc').value.trim(), amount = parseFloat($('amount').value);
  if (!desc || Number.isNaN(amount) || amount <= 0) { $('error').textContent = 'Enter a description and an amount greater than 0.'; return; }
  $('error').textContent = '';
  items.unshift({ id: Date.now(), desc, amount, type: $('type').value, date: new Date().toLocaleDateString() });
  save(); render();
  $('form').reset();
});
render();
