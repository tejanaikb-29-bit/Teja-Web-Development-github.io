// JavaScript
const $ = (id) => document.getElementById(id);
let cart = {};              // { productId: quantity }
function saveCart() { /* memory only */ }

const products = [
  { id: 1, name: 'Notebook', price: 3.5, icon: '📓' },
  { id: 2, name: 'Pen set', price: 2.25, icon: '🖊️' },
  { id: 3, name: 'Backpack', price: 24, icon: '🎒' },
  { id: 4, name: 'Headphones', price: 32.99, icon: '🎧' },
  { id: 5, name: 'Water bottle', price: 9.5, icon: '🧴' },
  { id: 6, name: 'Calculator', price: 12, icon: '🧮' }
];
const money = (n) => '$' + n.toFixed(2);

function renderProducts() {
  $('products').innerHTML = '';
  products.forEach((p) => {
    const card = document.createElement('div');
    card.className = 'product';
    card.innerHTML = `<div class="emoji">${p.icon}</div><h4>${p.name}</h4><div class="price">${money(p.price)}</div>`;
    const btn = document.createElement('button');
    btn.textContent = 'Add to cart';
    btn.addEventListener('click', () => { cart[p.id] = (cart[p.id] || 0) + 1; saveCart(); renderCart(); });
    card.appendChild(btn);
    $('products').appendChild(card);
  });
}
function renderCart() {
  const box = $('cart');
  box.innerHTML = '';
  let total = 0;
  const ids = Object.keys(cart);
  if (!ids.length) box.innerHTML = '<p class="muted">The cart is empty.</p>';
  ids.forEach((id) => {
    const p = products.find((x) => x.id === Number(id));
    const qty = cart[id];
    total += p.price * qty;
    const row = document.createElement('div');
    row.className = 'line';
    row.innerHTML = `<span class="name">${p.icon} ${p.name}</span>`;
    const minus = document.createElement('button');
    minus.textContent = '−';
    minus.addEventListener('click', () => { cart[id]--; if (cart[id] <= 0) delete cart[id]; saveCart(); renderCart(); });
    const q = document.createElement('b');
    q.textContent = qty;
    const plus = document.createElement('button');
    plus.textContent = '+';
    plus.addEventListener('click', () => { cart[id]++; saveCart(); renderCart(); });
    const sub = document.createElement('span');
    sub.textContent = money(p.price * qty);
    row.append(minus, q, plus, sub);
    box.appendChild(row);
  });
  $('total').textContent = money(total);
}
$('empty').addEventListener('click', () => { cart = {}; saveCart(); renderCart(); });
renderProducts();
renderCart();
