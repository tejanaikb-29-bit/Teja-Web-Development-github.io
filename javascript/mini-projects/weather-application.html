Weather application using an API

Type a city name to get the current weather and a 5-day forecast from the free Open-Meteo API (internet needed, no key required).

// JavaScript
const $ = (id) => document.getElementById(id);
const codes = {
  0: ['Clear sky', '\u2600\ufe0f'], 1: ['Mainly clear', '\ud83c\udf24\ufe0f'], 2: ['Partly cloudy', '\u26c5'], 3: ['Overcast', '\u2601\ufe0f'],
  45: ['Fog', '\ud83c\udf2b\ufe0f'], 48: ['Rime fog', '\ud83c\udf2b\ufe0f'], 51: ['Light drizzle', '\ud83c\udf26\ufe0f'], 53: ['Drizzle', '\ud83c\udf26\ufe0f'], 55: ['Dense drizzle', '\ud83c\udf27\ufe0f'],
  61: ['Light rain', '\ud83c\udf27\ufe0f'], 63: ['Rain', '\ud83c\udf27\ufe0f'], 65: ['Heavy rain', '\ud83c\udf27\ufe0f'], 71: ['Light snow', '\ud83c\udf28\ufe0f'], 73: ['Snow', '\u2744\ufe0f'], 75: ['Heavy snow', '\u2744\ufe0f'],
  80: ['Rain showers', '\ud83c\udf26\ufe0f'], 81: ['Rain showers', '\ud83c\udf27\ufe0f'], 82: ['Violent showers', '\u26c8\ufe0f'], 95: ['Thunderstorm', '\u26c8\ufe0f'], 96: ['Thunderstorm with hail', '\u26c8\ufe0f'], 99: ['Thunderstorm with hail', '\u26c8\ufe0f'],
};
const info = (code) => codes[code] || ['Unknown', '\ud83c\udf21\ufe0f'];

async function getWeather() {
  const city = $('city').value.trim();
  if (!city) { $('status').textContent = 'Please enter a city name.'; return; }
  $('status').textContent = 'Loading...';
  $('wx').style.display = 'none';
  try {
    // Step 1: convert the city name to latitude / longitude
    const geo = await (await fetch(`https://geocoding-api.open-meteo.com/v1/search?name=${encodeURIComponent(city)}&count=1`)).json();
    if (!geo.results || !geo.results.length) throw new Error('City not found');
    const { latitude, longitude, name, country } = geo.results[0];
    // Step 2: ask for the weather at that position
    const url = `https://api.open-meteo.com/v1/forecast?latitude=${latitude}&longitude=${longitude}&current_weather=true&daily=weathercode,temperature_2m_max,temperature_2m_min&timezone=auto`;
    const data = await (await fetch(url)).json();
    const cw = data.current_weather;
    const [text, icon] = info(cw.weathercode);
    $('icon').textContent = icon;
    $('temp').textContent = Math.round(cw.temperature) + '\u00b0C';
    $('desc').textContent = text;
    $('place').textContent = `${name}, ${country}`;
    $('wind').textContent = `Wind: ${cw.windspeed} km/h`;
    $('time').textContent = 'Updated: ' + cw.time.replace('T', ' ');
    $('days').innerHTML = '';
    data.daily.time.slice(1, 6).forEach((day, i) => {
      const d = document.createElement('div');
      d.className = 'day';
      const label = new Date(day).toLocaleDateString(undefined, { weekday: 'short', day: 'numeric' });
      d.innerHTML = `<strong>${label}</strong><div style="font-size:1.6rem">${info(data.daily.weathercode[i + 1])[1]}</div>${Math.round(data.daily.temperature_2m_max[i + 1])}\u00b0 / ${Math.round(data.daily.temperature_2m_min[i + 1])}\u00b0`;
      $('days').appendChild(d);
    });
    $('wx').style.display = 'block';
    $('status').textContent = '';
  } catch (error) {
    $('status').textContent = error.message === 'City not found' ? 'City not found. Check the spelling.' : 'Could not load the weather (' + error.message + '). Check your internet connection.';
  }
}
$('go').addEventListener('click', getWeather);
$('city').addEventListener('keydown', (e) => { if (e.key === 'Enter') getWeather(); });
