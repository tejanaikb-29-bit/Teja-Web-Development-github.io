Countdown timer

Set minutes and seconds, then start, pause or reset the countdown.

<!-- HTML -->
<div class="row">
  <label class="field">Minutes<input id="min" type="number" min="0" max="99" value="1" style="width:90px"></label>
  <label class="field">Seconds<input id="sec" type="number" min="0" max="59" value="30" style="width:90px"></label>
</div>
<div class="clock" id="display">00:00</div>
<div class="row" style="justify-content:center">
  <button id="start">Start</button><button id="pause" class="secondary">Pause</button><button id="reset" class="danger">Reset</button>
</div>
<p class="center" id="msg" style="font-weight:700;min-height:1.5em"></p>

// JavaScript
const $ = (id) => document.getElementById(id);
let remaining = 0, timer = null;
const pad = (n) => String(n).padStart(2, '0');

function render() {
  $('display').textContent = `${pad(Math.floor(remaining / 60))}:${pad(remaining % 60)}`;
}
$('start').addEventListener('click', () => {
  if (timer) return;
  if (remaining <= 0) {
    remaining = (parseInt($('min').value) || 0) * 60 + (parseInt($('sec').value) || 0);
  }
  if (remaining <= 0) { $('msg').textContent = 'Enter a time greater than zero.'; return; }
  $('msg').textContent = '';
  render();
  timer = setInterval(() => {
    remaining--;
    render();
    if (remaining <= 0) {
      clearInterval(timer);
      timer = null;
      $('msg').textContent = "Time's up!";
    }
  }, 1000);
});
$('pause').addEventListener('click', () => { clearInterval(timer); timer = null; });
$('reset').addEventListener('click', () => {
  clearInterval(timer);
  timer = null;
  remaining = 0;
  $('msg').textContent = '';
  render();
});
