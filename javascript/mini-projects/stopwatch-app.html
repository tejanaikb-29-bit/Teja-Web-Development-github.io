Stopwatch

A stopwatch with start, stop, reset and lap times.

Working demonstration

    <!-- HTML -->
<div class="clock" id="display">00:00:00.00</div>
<div class="row" style="justify-content:center">
  <button id="start">Start</button>
  <button id="stop" class="secondary">Stop</button>
  <button id="reset" class="danger">Reset</button>
  <button id="lap" class="success">Lap</button>
</div>
<ol id="laps"></ol>

// JavaScript
const $ = (id) => document.getElementById(id);
let startTime = 0, elapsed = 0, timer = null;

function format(ms) {
  const cs = Math.floor(ms / 10) % 100;
  const s = Math.floor(ms / 1000) % 60;
  const m = Math.floor(ms / 60000) % 60;
  const h = Math.floor(ms / 3600000);
  return [h, m, s].map((n) => String(n).padStart(2, '0')).join(':') + '.' + String(cs).padStart(2, '0');
}
function render() {
  $('display').textContent = format(elapsed + (timer ? Date.now() - startTime : 0));
}
$('start').addEventListener('click', () => {
  if (timer) return;                       // already running
  startTime = Date.now();
  timer = setInterval(render, 10);
});
$('stop').addEventListener('click', () => {
  if (!timer) return;
  elapsed += Date.now() - startTime;
  clearInterval(timer);
  timer = null;
  render();
});
$('reset').addEventListener('click', () => {
  clearInterval(timer);
  timer = null;
  elapsed = 0;
  render();
  $('laps').innerHTML = '';
});

$('lap').addEventListener('click', () => {
  if (!timer) return;
  const li = document.createElement('li');
  li.textContent = $('display').textContent;
  $('laps').appendChild(li);
});
