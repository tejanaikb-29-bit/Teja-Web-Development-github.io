Random password generator

Choose the length and character types to generate a secure random password, with a strength indicator and copy button.

<!-- HTML -->
<div class="pw" id="pw">Click Generate</div>
<div class="meter"><i id="bar"></i></div><p id="strength" class="muted"></p>
<div class="row"><label class="field">Length: <span id="lenVal">14</span><input id="len" type="range" min="6" max="40" value="14"></label></div>
<div class="opts"><label><input type="checkbox" id="upper" checked> Uppercase (A-Z)</label><label><input type="checkbox" id="lower" checked> Lowercase (a-z)</label>
<label><input type="checkbox" id="digits" checked> Numbers (0-9)</label><label><input type="checkbox" id="symbols" checked> Symbols (!@#$...)</label></div>

/* CSS */
.demo .pw { font: 700 1.4rem 'Courier New', monospace; padding: 1rem; background: #101828; color: #5eead4; border-radius: 10px; word-break: break-all; min-height: 3.2rem; } .demo .meter { height: 10px; border-radius: 6px; background: #e2e8f4; margin-top: .6rem; overflow: hidden; } .demo .meter i { display: block; height: 100%; width: 0; transition: width .3s, background .3s; } .demo .opts label { display: block; margin: .3rem 0; }

// JavaScript
const $ = (id) => document.getElementById(id);
const sets = { upper: 'ABCDEFGHIJKLMNOPQRSTUVWXYZ', lower: 'abcdefghijklmnopqrstuvwxyz', digits: '0123456789', symbols: '!@#$%^&*()-_=+[]{};:,.?' };

function randomInt(max) {
  const a = new Uint32Array(1);
  crypto.getRandomValues(a);            // cryptographically secure random number
  return a[0] % max;
}
function generate() {
  const chosen = Object.keys(sets).filter((k) => $(k).checked);
  if (!chosen.length) { $('pw').textContent = 'Select at least one option'; return; }
  const length = Number($('len').value);
  let pool = '';
  const chars = [];
  chosen.forEach((k) => { pool += sets[k]; chars.push(sets[k][randomInt(sets[k].length)]); });   // at least one of each type
  while (chars.length < length) chars.push(pool[randomInt(pool.length)]);
  for (let i = chars.length - 1; i > 0; i--) { const j = randomInt(i + 1); [chars[i], chars[j]] = [chars[j], chars[i]]; }   // shuffle
  $('pw').textContent = chars.join('');
  const score = Math.min(4, chosen.length) + (length >= 12 ? 2 : length >= 8 ? 1 : 0) + (length >= 20 ? 1 : 0);
  const levels = [['Weak', '#c62828', 25], ['Fair', '#d98a00', 50], ['Good', '#3b82f6', 75], ['Strong', '#178a4a', 100]];
  const lv = score <= 3 ? levels[0] : score <= 4 ? levels[1] : score <= 5 ? levels[2] : levels[3];
  $('bar').style.width = lv[2] + '%'; $('bar').style.background = lv[1];
  $('strength').textContent = 'Strength: ' + lv[0];
}
$('len').addEventListener('input', () => { $('lenVal').textContent = $('len').value; });
$('gen').addEventListener('click', generate);
$('copy').addEventListener('click', async () => {
  try { await navigator.clipboard.writeText($('pw').textContent); $('copy').textContent = 'Copied!'; }
  catch (e) { $('copy').textContent = 'Press Ctrl+C'; }
  setTimeout(() => { $('copy').textContent = 'Copy'; }, 1500);
});
generate();
