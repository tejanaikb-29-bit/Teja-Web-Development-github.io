BMI calculator

Calculates Body Mass Index from weight and height and shows the category on a colour scale.

Working demonstration

--
Underweight <18.5Normal 18.5-24.9Overweight 25-29.9Obese 30+
View source code of this program 
<!-- HTML -->
<div class="row">
  <label class="field">Weight (kg)<input id="w" type="number" value="65" min="1" step="0.1"></label>
  <label class="field">Height (cm)<input id="h" type="number" value="170" min="30" step="0.1"></label>
  <button id="go">Calculate BMI</button>
</div>
<div class="out"><div class="bmi" id="bmi">--</div><div id="cat" style="font-weight:700"></div>
<div class="scale" id="scale"><span style="background:#3b82f6"></span><span style="background:#22c55e"></span><span style="background:#f59e0b"></span><span style="background:#ef4444"></span><i class="pin" id="pin"></i></div>
<div class="labels"><span>Underweight &lt;18.5</span><span>Normal 18.5-24.9</span><span>Overweight 25-29.9</span><span>Obese 30+</span></div></div>

/* CSS */
.demo .scale { display: flex; height: 16px; border-radius: 8px; overflow: hidden; margin: 1rem 0 .3rem; position: relative; }
.demo .scale span { flex: 1; } .demo .pin { position: absolute; top: -6px; width: 4px; height: 28px; background: #14213d; border-radius: 2px; transition: left .4s; left: 0; }
.demo .labels { display: flex; justify-content: space-between; font-size: .75rem; color: #55637d; } .demo .bmi { font-size: 2.6rem; font-weight: 800; }

// JavaScript
const $ = (id) => document.getElementById(id);
function calculate() {
  const w = parseFloat($('w').value), h = parseFloat($('h').value) / 100;
  if (!(w > 0) || !(h > 0)) { $('bmi').textContent = '--'; $('cat').textContent = 'Enter a valid weight and height.'; return; }
  const bmi = w / (h * h);                                 // BMI = kg / m^2
  $('bmi').textContent = bmi.toFixed(1);
  let cat, color;
  if (bmi < 18.5) { cat = 'Underweight'; color = '#3b82f6'; }
  else if (bmi < 25) { cat = 'Normal weight'; color = '#178a4a'; }
  else if (bmi < 30) { cat = 'Overweight'; color = '#d98a00'; }
  else { cat = 'Obese'; color = '#c62828'; }
  $('cat').textContent = cat; $('cat').style.color = color;
  const pos = Math.min(Math.max((bmi - 10) / 30, 0), 1);   // map BMI 10..40 to 0..100%
  $('pin').style.left = `calc(${pos * 100}% - 2px)`;
}
$('go').addEventListener('click', calculate);
document.querySelectorAll('input').forEach((i) => i.addEventListener('keydown', (e) => { if (e.key === 'Enter') calculate(); }));
