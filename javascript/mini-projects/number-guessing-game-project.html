Number guessing game

Guess the secret number between 1 and 100 with hints and a history of your guesses.

Working demonstration

I am thinking of a number between 1 and 100. Can you guess it?

Type a number and press Guess.

View source code of this program 
<!-- HTML -->
<p>I am thinking of a number between <strong>1</strong> and <strong>100</strong>. Can you guess it?</p>
<div class="row">
  <input id="guess" type="number" min="1" max="100" placeholder="Your guess">
  <button id="try">Guess</button>
  <button id="new" class="secondary">New game</button>
</div>
<div class="out" id="out">Type a number and press Guess.</div>
<p class="muted" id="history"></p>

// JavaScript
const $ = (id) => document.getElementById(id);
let secret, attempts, past;

function newGame() {
  secret = Math.floor(Math.random() * 100) + 1;
  attempts = 0;
  past = [];
  $('guess').disabled = false;
  $('guess').value = '';
  $('out').textContent = 'New game started. Type a number and press Guess.';
  $('history').textContent = '';
  $('guess').focus();
}
function check() {
  const g = parseInt($('guess').value);
  if (Number.isNaN(g) || g < 1 || g > 100) { $('out').textContent = 'Please enter a number from 1 to 100.'; return; }
  attempts++;
  past.push(g);
  if (g === secret) {
    $('out').textContent = `Correct! The number was ${secret}. You needed ${attempts} attempt(s).`;
    $('guess').disabled = true;
  } else {
    const diff = Math.abs(g - secret);
    const hint = diff <= 5 ? 'Very close! ' : diff <= 15 ? 'Warm. ' : 'Cold. ';
    $('out').textContent = hint + (g < secret ? 'Too low, go higher.' : 'Too high, go lower.');
  }
  $('history').textContent = `Attempts: ${attempts}   Guesses: ${past.join(', ')}`;
  $('guess').value = '';
  $('guess').focus();
}
$('try').addEventListener('click', check);
$('guess').addEventListener('keydown', (e) => { if (e.key === 'Enter') check(); });
$('new').addEventListener('click', newGame);
newGame();
