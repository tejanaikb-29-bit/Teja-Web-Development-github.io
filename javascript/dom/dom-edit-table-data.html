Edit table data dynamically

Double-click a cell to edit it. Press Enter or click away to save, Escape to cancel.

Working demonstration

View source code of this program 
<!-- HTML -->
<table class="data"><thead><tr><th>Name</th><th>Department</th><th>Salary</th></tr></thead>
<tbody id="tb"><tr><td>Asha</td><td>Design</td><td>52000</td></tr><tr><td>Ravi</td><td>Engineering</td><td>68000</td></tr><tr><td>Meena</td><td>Finance</td><td>61000</td></tr></tbody></table>
<p class="muted" id="msg">Double-click any cell.</p>

/* CSS */
.demo #tb td { cursor: text; } .demo #tb td:hover { background: #f4f8ff; } .demo td input { width: 100%; }

// JavaScript
const $ = (id) => document.getElementById(id);
$('tb').addEventListener('dblclick', (e) => {
  const td = e.target.closest('td');
  if (!td || td.querySelector('input')) return;
  const old = td.textContent;
  const input = document.createElement('input');
  input.value = old;
  td.textContent = '';
  td.appendChild(input);
  input.focus(); input.select();
  let done = false;
  const finish = (save) => {
    if (done) return; done = true;
    const value = save && input.value.trim() ? input.value.trim() : old;
    td.textContent = value;
    $('msg').textContent = value !== old ? `Changed "${old}" to "${value}"` : 'No change.';
  };
  input.addEventListener('keydown', (ev) => { if (ev.key === 'Enter') finish(true); if (ev.key === 'Escape') finish(false); });
  input.addEventListener('blur', () => finish(true));
});
