Add and remove CSS classes using JavaScript

classList.add() and classList.remove() apply or take away styling classes. The current class list is displayed.

<!-- HTML -->
<div id="t" class="tgt">Target element</div>
<div class="row" id="btns"><button data-add="bg-blue">+ bg-blue</button><button data-add="bg-green">+ bg-green</button><button data-add="round">+ round</button><button data-add="big">+ big</button><button data-add="bold-border">+ bold-border</button>
<button id="clear" class="danger">Remove all classes</button></div>
<p>className: <code id="cls"></code></p>

/* CSS */
.demo .tgt { padding: 1.2rem; border-radius: 10px; border: 3px solid #b9c3d6; text-align: center; font-weight: 700; transition: .3s; margin-bottom: .8rem; }
.demo .tgt.bg-blue { background: #b3d4ff; } .demo .tgt.bg-green { background: #b8e6c4; } .demo .tgt.round { border-radius: 40px; }
.demo .tgt.big { font-size: 1.8rem; } .demo .tgt.bold-border { border: 6px solid #14213d; }

// JavaScript
const $ = (id) => document.getElementById(id);
function show() { $('cls').textContent = $('t').className; }
document.querySelectorAll('button[data-add]').forEach((b) => {
  b.addEventListener('click', () => {
    const cls = b.dataset.add;
    if (cls === 'bg-blue') $('t').classList.remove('bg-green');
    if (cls === 'bg-green') $('t').classList.remove('bg-blue');
    $('t').classList.add(cls);
    show();
  });
});
$('clear').addEventListener('click', () => { $('t').className = 'tgt'; show(); });
show();
