Show / hide password feature

A button switches the input type between password and text. A checkbox version is included.

Working demonstration

View source code of this program 
<!-- HTML -->
<div class="row"><label class="field">Password<span style="display:flex;gap:.4rem"><input id="pw" type="password" value="MySecret123"><button id="eye" type="button">Show</button></span></label></div>
<label><input type="checkbox" id="chk"> Show password (checkbox version)</label>

// JavaScript
const $ = (id) => document.getElementById(id);
function setVisible(visible) {
  $('pw').type = visible ? 'text' : 'password';
  $('eye').textContent = visible ? 'Hide' : 'Show';
  $('chk').checked = visible;
}
$('eye').addEventListener('click', () => setVisible($('pw').type === 'password'));
$('chk').addEventListener('change', () => setVisible($('chk').checked));
