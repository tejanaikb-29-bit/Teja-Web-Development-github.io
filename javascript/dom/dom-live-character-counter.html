Live character counter for a textarea

Counts characters as you type and turns red when you approach the 140 character limit.

Working demonstration

0 / 140 characters · 140 left

View source code of this program 
<!-- HTML -->
<textarea id="t" rows="4" maxlength="140" placeholder="Write a short message (max 140 characters)..." style="width:100%"></textarea>
<div class="cc-bar"><i id="bar"></i></div><p><strong id="n">0</strong> / 140 characters &middot; <span id="left">140 left</span></p>

/* CSS */
.demo .cc-bar { height: 8px; background: #e2e8f4; border-radius: 4px; overflow: hidden; margin-top: .3rem; } .demo .cc-bar i { display: block; height: 100%; width: 0; background: #178a4a; transition: width .1s, background .2s; }

// JavaScript
const $ = (id) => document.getElementById(id);
const MAX = 140;
$('t').addEventListener('input', () => {
  const len = $('t').value.length;
  $('n').textContent = len;
  $('left').textContent = `${MAX - len} left`;
  $('bar').style.width = (len / MAX) * 100 + '%';
  const color = len >= MAX ? '#c62828' : len > MAX * 0.85 ? '#d98a00' : '#178a4a';
  $('bar').style.background = color;
  $('n').style.color = color;
});
