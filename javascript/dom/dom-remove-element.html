Remove an existing HTML element dynamically

Each card has a remove button that calls element.remove(). A second button removes the last child using removeChild().

Working demonstration

View source code of this program 
<!-- HTML -->
<div id="list"></div>
<div class="row"><button id="last" class="secondary">removeChild(last)</button><button id="restore">Restore all</button></div>

/* CSS */
.demo .chip { display: flex; justify-content: space-between; align-items: center; padding: .6rem .9rem; background: #f1f4fa; border-radius: 8px; margin: .4rem 0; }

// JavaScript
const $ = (id) => document.getElementById(id);
function build() {
  $('list').innerHTML = '';
  ['Apple', 'Banana', 'Cherry', 'Date'].forEach((name) => {
    const chip = document.createElement('div');
    chip.className = 'chip';
    chip.innerHTML = `<span>${name}</span>`;
    const b = document.createElement('button');
    b.textContent = 'Remove'; b.className = 'danger small';
    b.addEventListener('click', () => chip.remove());          // remove this element
    chip.appendChild(b);
    $('list').appendChild(chip);
  });
}
$('last').addEventListener('click', () => { if ($('list').lastElementChild) $('list').removeChild($('list').lastElementChild); });
$('restore').addEventListener('click', build);
build();
