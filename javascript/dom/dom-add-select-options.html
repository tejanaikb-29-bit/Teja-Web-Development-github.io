Add options dynamically to a <select> element

Type a new option and add it with new Option(); you can also remove the selected option.

Working demonstration

View source code of this program 
<!-- HTML -->
<div class="row"><input id="txt" type="text" placeholder="New option text"><button id="add">Add option</button></div>
<div class="row"><select id="sel" size="6" style="min-width:200px"><option>Chennai</option><option>Madurai</option></select>
<div><button id="rem" class="danger">Remove selected</button><p id="msg" class="muted"></p></div></div>

// JavaScript
const $ = (id) => document.getElementById(id);
$('add').addEventListener('click', () => {
  const t = $('txt').value.trim();
  if (!t) { $('msg').textContent = 'Type something first.'; return; }
  if ([...$('sel').options].some((o) => o.text.toLowerCase() === t.toLowerCase())) { $('msg').textContent = 'That option already exists.'; return; }
  $('sel').add(new Option(t, t.toLowerCase()));         // new Option(text, value)
  $('sel').selectedIndex = $('sel').options.length - 1;
  $('txt').value = '';
  $('msg').textContent = `Added "${t}". Total options: ${$('sel').options.length}`;
});
$('rem').addEventListener('click', () => {
  if ($('sel').selectedIndex < 0) { $('msg').textContent = 'Select an option to remove.'; return; }
  $('sel').remove($('sel').selectedIndex);
  $('msg').textContent = 'Option removed.';
});
