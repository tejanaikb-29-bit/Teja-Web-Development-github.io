Change the attributes of an HTML element dynamically

setAttribute(), getAttribute() and removeAttribute() change src, alt, href, title and disabled on real elements.

<!-- HTML -->
<img id="img" src="../assets/images/img1.svg" alt="Mountains" title="Mountains" width="260" style="border-radius:8px"><br>
<a id="lnk" href="index.html">A link to the JavaScript list</a> <input id="inp" type="text" value="Editable input">
<div class="row"><button id="a1">Change image</button><button id="a2" class="secondary">Change link href + text</button><button id="a3">Disable input</button><button id="a4" class="danger">Remove title attribute</button></div>
<pre class="out" id="out"></pre>

// JavaScript
const $ = (id) => document.getElementById(id);
const info = () => {
  $('out').textContent = `img.src   = ${$('img').getAttribute('src')}\nimg.alt   = ${$('img').getAttribute('alt')}\nimg.title = ${$('img').getAttribute('title')}\nlink.href = ${$('lnk').getAttribute('href')}\ninput disabled? ${$('inp').hasAttribute('disabled')}`;
};
$('a1').onclick = () => { $('img').setAttribute('src', '../assets/images/img3.svg'); $('img').setAttribute('alt', 'Ocean'); $('img').setAttribute('title', 'Ocean'); info(); };
$('a2').onclick = () => { $('lnk').setAttribute('href', '../index.html'); $('lnk').textContent = 'Now links to Home'; info(); };
$('a3').onclick = () => { const d = $('inp').hasAttribute('disabled'); d ? $('inp').removeAttribute('disabled') : $('inp').setAttribute('disabled', ''); $('a3').textContent = d ? 'Disable input' : 'Enable input'; info(); };
$('a4').onclick = () => { $('img').removeAttribute('title'); info(); };
info();
