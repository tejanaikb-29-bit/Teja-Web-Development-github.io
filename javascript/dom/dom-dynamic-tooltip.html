Create a tooltip dynamically

On mouseover a tooltip element is created with createElement(), positioned next to the mouse and removed on mouseout.

<!-- HTML -->
<span class="hasTip" data-tip="HyperText Markup Language">HTML</span><span class="hasTip" data-tip="Cascading Style Sheets">CSS</span>
<span class="hasTip" data-tip="The programming language of the web">JavaScript</span><span class="hasTip" data-tip="Document Object Model - the page as objects">DOM</span>

/* CSS */
.demo .tip { position: fixed; z-index: 2000; background: #14213d; color: #fff; padding: .35rem .7rem; border-radius: 6px; font-size: .85rem; pointer-events: none; box-shadow: 0 4px 12px rgba(0,0,0,.3); max-width: 240px; } .demo .hasTip { display: inline-block; margin: .5rem 1rem .5rem 0; padding: .5rem 1rem; background: #eef2fb; border: 1px dashed #55637d; border-radius: 8px; cursor: help; }

// JavaScript
const $ = (id) => document.getElementById(id);
let tip = null;
document.querySelectorAll('.hasTip').forEach((el) => {
  el.addEventListener('mouseover', () => {
    tip = document.createElement('div');
    tip.className = 'tip';
    tip.textContent = el.dataset.tip;
    $('demo').appendChild(tip);
  });
  el.addEventListener('mousemove', (e) => {
    if (!tip) return;
    tip.style.left = e.clientX + 14 + 'px';
    tip.style.top = e.clientY + 16 + 'px';
  });
  el.addEventListener('mouseout', () => { if (tip) { tip.remove(); tip = null; } });
});
