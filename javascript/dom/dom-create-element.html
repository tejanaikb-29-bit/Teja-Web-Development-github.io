Create a new HTML element dynamically and add it to the page

createElement() builds a new element in memory and appendChild() / prepend() puts it into the page.

Working demonstration

Existing paragraph

View source code of this program 
<!-- HTML -->
<div class="row"><input id="txt" type="text" value="New paragraph" style="flex:1;min-width:200px">
<button id="add">Add to end</button><button id="pre" class="secondary">Add to start</button></div>
<div id="box" class="card" style="min-height:80px"><p>Existing paragraph</p></div>

// JavaScript
const $ = (id) => document.getElementById(id);
let n = 0;
function makeParagraph() {
  const p = document.createElement('p');                 // 1. create
  p.textContent = `${++n}. ${$('txt').value || '(empty)'}`;   // 2. set content
  p.style.borderLeft = '4px solid #2b59c3';
  p.style.paddingLeft = '8px';
  return p;
}
$('add').addEventListener('click', () => $('box').appendChild(makeParagraph()));   // 3. add to the page
$('pre').addEventListener('click', () => $('box').prepend(makeParagraph()));
