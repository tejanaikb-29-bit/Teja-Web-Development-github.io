Delete a selected row from an HTML table

Click a row to select it (it turns yellow), then press Delete selected. deleteRow() removes it.

Working demonstration

IDProductPrice

View source code of this program 
<!-- HTML -->
<table class="data"><thead><tr><th>ID</th><th>Product</th><th>Price</th></tr></thead><tbody id="tb"></tbody></table>
<div class="row"><button id="del" class="danger">Delete selected row</button><button id="reset" class="secondary">Restore rows</button></div><p class="muted" id="msg"></p>

/* CSS */
.demo tr.sel td { background: #fff2a8; } .demo #tb tr { cursor: pointer; }

// JavaScript
const $ = (id) => document.getElementById(id);
let selected = null;
function fill() {
  $('tb').innerHTML = '';
  [['1', 'Notebook', '$3.50'], ['2', 'Pen pack', '$2.00'], ['3', 'Backpack', '$25.00'], ['4', 'Headphones', '$18.75']].forEach((r) => {
    const tr = $('tb').insertRow();
    r.forEach((v) => { tr.insertCell().textContent = v; });
  });
  selected = null;
}
$('tb').addEventListener('click', (e) => {
  const tr = e.target.closest('tr');
  if (!tr) return;
  if (selected) selected.classList.remove('sel');
  selected = tr;
  tr.classList.add('sel');
});
$('del').addEventListener('click', () => {
  if (!selected) { $('msg').textContent = 'Click a row first.'; return; }
  $('msg').textContent = `Deleted row: ${selected.cells[1].textContent}`;
  $('tb').deleteRow(selected.rowIndex - 1);       // rowIndex counts the header row too
  selected = null;
});
$('reset').addEventListener('click', () => { fill(); $('msg').textContent = ''; });
fill();
