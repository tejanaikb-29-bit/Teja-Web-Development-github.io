JavaScript › DOM Manipulation

Create an image slideshow using DOM manipulation

Previous / next buttons, clickable dots, keyboard arrows and an auto-play option.

Working demonstration

View source code of this program 
<!-- HTML -->
<div class="show"><img id="img" src="" alt=""><div class="cap" id="cap"></div></div>
<div class="dots" id="dots"></div>
<div class="row" style="justify-content:center"><button id="prev" class="secondary">&larr; Prev</button><button id="auto">Start auto-play</button><button id="next" class="secondary">Next &rarr;</button></div>

/* CSS */
.demo .show { position: relative; max-width: 560px; margin: 0 auto; } .demo .show img { width: 100%; height: 300px; object-fit: cover; border-radius: 12px; display: block; } .demo .cap { position: absolute; left: 0; right: 0; bottom: 0; padding: .6rem 1rem; background: linear-gradient(transparent, rgba(0,0,0,.7)); color: #fff; font-weight: 700; border-radius: 0 0 12px 12px; } .demo .dots { display: flex; justify-content: center; gap: .4rem; margin: .8rem 0; } .demo .dots i { width: 12px; height: 12px; border-radius: 50%; background: #b9c3d6; cursor: pointer; } .demo .dots i.on { background: #14213d; }

// JavaScript
const $ = (id) => document.getElementById(id);
const slides = [['img1', 'Misty mountains'], ['img2', 'Sunset hills'], ['img3', 'Blue ocean'], ['img4', 'Green forest'], ['img5', 'Starry night']];
let i = 0, timer = null;
slides.forEach((s, k) => { const d = document.createElement('i'); d.addEventListener('click', () => go(k)); $('dots').appendChild(d); });
function go(n) {
  i = (n + slides.length) % slides.length;
  $('img').src = `../assets/images/${slides[i][0]}.svg`;
  $('img').alt = slides[i][1];
  $('cap').textContent = `${i + 1} / ${slides.length}  -  ${slides[i][1]}`;
  [...$('dots').children].forEach((d, k) => d.classList.toggle('on', k === i));
}
$('prev').addEventListener('click', () => go(i - 1));
$('next').addEventListener('click', () => go(i + 1));
$('auto').addEventListener('click', () => {
  if (timer) { clearInterval(timer); timer = null; $('auto').textContent = 'Start auto-play'; }
  else { timer = setInterval(() => go(i + 1), 2000); $('auto').textContent = 'Stop auto-play'; }
});
document.addEventListener('keydown', (e) => { if (e.key === 'ArrowLeft') go(i - 1); if (e.key === 'ArrowRight') go(i + 1); });
go(0);
