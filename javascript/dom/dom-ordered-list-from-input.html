Create a dynamic ordered list from user input

Type an item and press Add: it is appended to an <ol>, which numbers the items automatically.

Working demonstration

    View source code of this program 
    <!-- HTML -->
<div class="row"><input id="item" type="text" placeholder="Type an item and press Enter" style="flex:1;min-width:200px"><button id="add">Add</button><button id="clear" class="danger">Clear</button></div>
<ol id="list"></ol><p class="muted" id="count"></p>

// JavaScript
const $ = (id) => document.getElementById(id);
function count() { $('count').textContent = `${$('list').children.length} item(s)`; }
function add() {
  const v = $('item').value.trim();
  if (!v) return;
  const li = document.createElement('li');
  li.textContent = v;
  $('list').appendChild(li);
  $('item').value = '';
  $('item').focus();
  count();
}
$('add').addEventListener('click', add);
$('item').addEventListener('keydown', (e) => { if (e.key === 'Enter') add(); });
$('clear').addEventListener('click', () => { $('list').innerHTML = ''; count(); });
count();
