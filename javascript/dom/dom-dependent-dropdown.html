Dependent dropdown: Country, State and City

Choosing a country fills the state list, and choosing a state fills the city list.

Working demonstration

Select a country to begin.
View source code of this program 
<!-- HTML -->
<div class="row"><label class="field">Country<select id="country"></select></label><label class="field">State<select id="state" disabled></select></label><label class="field">City<select id="city" disabled></select></label></div>
<div class="out" id="out">Select a country to begin.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const data = {
  India: { 'Tamil Nadu': ['Chennai', 'Madurai', 'Coimbatore'], Kerala: ['Kochi', 'Thiruvananthapuram'], Karnataka: ['Bengaluru', 'Mysuru'] },
  USA: { California: ['Los Angeles', 'San Francisco'], Texas: ['Houston', 'Austin'] },
  UK: { England: ['London', 'Manchester'], Scotland: ['Edinburgh', 'Glasgow'] },
};
function fill(select, items, label) {
  select.innerHTML = '';
  select.add(new Option(`-- ${label} --`, ''));
  items.forEach((i) => select.add(new Option(i, i)));
  select.disabled = items.length === 0;
}
function summary() {
  const parts = [$('country').value, $('state').value, $('city').value].filter(Boolean);
  $('out').textContent = parts.length ? 'Selected: ' + parts.join(' > ') : 'Select a country to begin.';
}
fill($('country'), Object.keys(data), 'Select country');
fill($('state'), [], 'Select state');
fill($('city'), [], 'Select city');
$('country').addEventListener('change', () => {
  fill($('state'), $('country').value ? Object.keys(data[$('country').value]) : [], 'Select state');
  fill($('city'), [], 'Select city');
  summary();
});
$('state').addEventListener('change', () => {
  fill($('city'), $('state').value ? data[$('country').value][$('state').value] : [], 'Select city');
  summary();
});
$('city').addEventListener('change', summary);
