<!-- HTML -->
<div class="stage" id="stage"><div class="n" style="background:#1769b5">Box 1</div><div class="n" style="background:#178a4a">Box 2</div></div>
<div class="row"><button id="c">createElement + appendChild</button><button id="r" class="danger">remove() last</button><button id="rc" class="secondary">replaceChild() first</button></div>
<pre class="out" id="log" style="font-size:.85rem">Log:</pre>

/* CSS */
.demo .stage { display: flex; gap: .6rem; flex-wrap: wrap; min-height: 70px; padding: .8rem; border: 2px dashed #8a97b3; border-radius: 10px; } .demo .stage .n { padding: .6rem 1rem; border-radius: 8px; color: #fff; font-weight: 700; }

// JavaScript
const $ = (id) => document.getElementById(id);
let n = 2;
const colors = ['#d9481f', '#7b2cbf', '#0f6b5c', '#d98a00', '#14213d'];
const log = (t) => { $('log').textContent += '\n' + t; };
function newBox() {
  const d = document.createElement('div');          // 1. create an element
  d.className = 'n';
  d.textContent = 'Box ' + (++n);
  d.style.background = colors[n % colors.length];
  return d;
}
$('c').addEventListener('click', () => { $('stage').appendChild(newBox()); log(`createElement() + appendChild() -> Box ${n} added`); });
$('r').addEventListener('click', () => {
  const last = $('stage').lastElementChild;
  if (!last) return log('remove(): nothing left to remove');
  log(`remove() -> "${last.textContent}" removed`);
  last.remove();
});
$('rc').addEventListener('click', () => {
  const first = $('stage').firstElementChild;
  if (!first) return log('replaceChild(): stage is empty');
  const fresh = newBox();
  $('stage').replaceChild(fresh, first);
  log(`replaceChild() -> "${first.textContent}" replaced by "${fresh.textContent}"`);
});
