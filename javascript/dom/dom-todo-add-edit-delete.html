Add, edit and delete to-do items

Full task management: add a task, click it to mark done, press Edit to change the text and Delete to remove it.

Working demonstration

    View source code of this program 
    <!-- HTML -->
<div class="row">
  <input id="task" type="text" placeholder="Add a new task..." style="flex:1;min-width:200px">
  <button id="add">Add task</button>
</div>
<ul id="list" class="todo"></ul>
<p class="muted" id="count"></p>
<button id="clear" class="secondary small">Clear completed</button>

/* CSS */
.todo { list-style: none; padding: 0; }
.todo li { display: flex; align-items: center; gap: .5rem; padding: .6rem .8rem; border: 1px solid #d5dbe6; border-radius: 8px; margin: .4rem 0; background: #fff; }
.todo li span { flex: 1; cursor: pointer; word-break: break-word; }
.todo li.done span { text-decoration: line-through; color: #8a97b3; }

// JavaScript
const $ = (id) => document.getElementById(id);
let todos = [];
function save() { /* nothing to save: data lives only in memory */ }

function render() {
  const list = $('list');
  list.innerHTML = '';
  todos.forEach((todo, i) => {
    const li = document.createElement('li');
    if (todo.done) li.classList.add('done');
    const label = document.createElement('span');
    label.textContent = todo.text;
    label.title = 'Click to mark as done';
    label.addEventListener('click', () => { todo.done = !todo.done; save(); render(); });

    const editBtn = document.createElement('button');
    editBtn.textContent = 'Edit';
    editBtn.className = 'secondary small';
    editBtn.addEventListener('click', () => {
      const text = prompt('Edit task', todo.text);
      if (text !== null && text.trim() !== '') { todo.text = text.trim(); save(); render(); }
    });

    const del = document.createElement('button');
    del.textContent = 'Delete';
    del.className = 'danger small';
    del.addEventListener('click', () => { todos.splice(i, 1); save(); render(); });
    li.append(label, editBtn, del);
    list.appendChild(li);
  });
  const left = todos.filter((t) => !t.done).length;
  $('count').textContent = todos.length ? `${left} of ${todos.length} tasks remaining` : 'No tasks yet. Add one above!';
}
function addTask() {
  const text = $('task').value.trim();
  if (!text) return;
  todos.push({ text, done: false });
  $('task').value = '';
  $('task').focus();
  save();
  render();
}
$('add').addEventListener('click', addTask);
$('task').addEventListener('keydown', (e) => { if (e.key === 'Enter') addTask(); });
$('clear').addEventListener('click', () => { todos = todos.filter((t) => !t.done); save(); render(); });
render();
