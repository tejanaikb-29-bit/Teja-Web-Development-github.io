<!-- HTML -->
<textarea id="t" rows="6" style="width:100%" placeholder="Start typing or paste some text here...">JavaScript makes web pages interactive. Type here and watch the numbers change!</textarea>
<div class="stats"><div><b id="w">0</b>Words</div><div><b id="c">0</b>Characters</div><div><b id="cs">0</b>No spaces</div><div><b id="s">0</b>Sentences</div><div><b id="p">0</b>Paragraphs</div><div><b id="r">0</b>Min to read</div></div>

/* CSS */
.demo .stats { display: grid; grid-template-columns: repeat(auto-fit, minmax(110px, 1fr)); gap: .6rem; margin-top: .8rem; } .demo .stats div { background: #f1f4fa; border-radius: 8px; padding: .6rem; text-align: center; } .demo .stats b { display: block; font-size: 1.6rem; color: #1f419a; }

// JavaScript
const $ = (id) => document.getElementById(id);
function update() {
  const text = $('t').value;
  const words = text.trim() === '' ? 0 : text.trim().split(/\s+/).length;
  $('w').textContent = words;
  $('c').textContent = text.length;
  $('cs').textContent = text.replace(/\s/g, '').length;
  $('s').textContent = (text.match(/[^.!?]+[.!?]+/g) || []).length;
  $('p').textContent = text.split(/\n+/).filter((x) => x.trim()).length;
  $('r').textContent = words ? Math.max(1, Math.ceil(words / 200)) : 0;     // about 200 words per minute
}
$('t').addEventListener('input', update);
update();
