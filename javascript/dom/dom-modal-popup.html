Create a modal popup using JavaScript

A modal dialog opens over the page, closes with the X button, Escape key or a click on the dark backdrop, and returns the answer.

Working demonstration

Delete this file?

This action cannot be undone. Are you sure?

View source code of this program 
<!-- HTML -->
<button id="open">Open modal</button> <p id="answer" class="muted"></p>
<div class="overlay" id="overlay"><div class="modal" role="dialog" aria-modal="true"><button class="x" id="x" aria-label="Close">&times;</button>
<h3 style="margin-top:0">Delete this file?</h3><p>This action cannot be undone. Are you sure?</p>
<div class="row"><button id="yes" class="danger">Yes, delete</button><button id="no" class="secondary">Cancel</button></div></div></div>

/* CSS */
.demo .overlay { position: fixed; inset: 0; background: rgba(15, 23, 42, .6); display: none; place-items: center; z-index: 1000; padding: 1rem; }
.demo .overlay.show { display: grid; }
.demo .modal { background: #fff; width: min(420px, 100%); border-radius: 14px; padding: 1.4rem; position: relative; box-shadow: 0 20px 50px rgba(0,0,0,.4); animation: pop .25s; }
.demo .modal .x { position: absolute; top: .5rem; right: .7rem; background: none; color: #55637d; font-size: 1.6rem; padding: 0 .4rem; }
@keyframes pop { from { transform: scale(.85); opacity: 0; } }

// JavaScript
const $ = (id) => document.getElementById(id);
const overlay = $('overlay');
const open = () => { overlay.classList.add('show'); $('no').focus(); };
const close = (answer) => { overlay.classList.remove('show'); if (answer) $('answer').textContent = 'You chose: ' + answer; };
$('open').addEventListener('click', open);
$('x').addEventListener('click', () => close('closed without choosing'));
$('yes').addEventListener('click', () => close('Yes, delete'));
$('no').addEventListener('click', () => close('Cancel'));
overlay.addEventListener('click', (e) => { if (e.target === overlay) close('clicked outside'); });
document.addEventListener('keydown', (e) => { if (e.key === 'Escape' && overlay.classList.contains('show')) close('pressed Escape'); });
