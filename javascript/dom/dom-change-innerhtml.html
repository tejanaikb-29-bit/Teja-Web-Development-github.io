Change the HTML content of an element using innerHTML

innerHTML replaces the HTML inside an element, so tags such as <b> and <i> are rendered.

Working demonstration

Original content





View source code of this program

<!-- HTML -->
<div id="target" class="out" style="min-height:4rem">Original content</div>
<label class="field" style="margin-top:.8rem">HTML to insert<textarea id="html" rows="3">&lt;h3&gt;A heading&lt;/h3&gt;&lt;p&gt;A &lt;b&gt;bold&lt;/b&gt; and &lt;i&gt;italic&lt;/i&gt; sentence.&lt;/p&gt;</textarea></label>
<div class="row"><button id="go">Set innerHTML</button><button id="show" class="secondary">Show current innerHTML</button></div>
<pre class="out" id="src"></pre>

// JavaScript
const $ = (id) => document.getElementById(id);
$('go').addEventListener('click', () => { $('target').innerHTML = $('html').value; });
$('show').addEventListener('click', () => { $('src').textContent = $('target').innerHTML; });
