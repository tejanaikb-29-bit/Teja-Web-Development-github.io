Change the style of an element dynamically

Sliders and pickers change the colour, size, padding, border-radius and rotation of a box through element.style.

Working demonstration

Style me!
View source code of this program 
<!-- HTML -->
<div style="min-height:150px;display:grid;place-items:center"><div id="box" style="background:#1769b5;color:#fff;padding:20px;border-radius:8px;font-weight:700">Style me!</div></div>
<div class="row">
  <label class="field">Background<input id="bg" type="color" value="#1769b5"></label>
  <label class="field">Text colour<input id="fg" type="color" value="#ffffff"></label>
  <label class="field">Font size<input id="fs" type="range" min="12" max="48" value="18"></label>
  <label class="field">Padding<input id="pd" type="range" min="0" max="60" value="20"></label>
  <label class="field">Radius<input id="rd" type="range" min="0" max="80" value="8"></label>
  <label class="field">Rotate<input id="rt" type="range" min="-45" max="45" value="0"></label>
</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const box = $('box');
$('bg').oninput = (e) => { box.style.backgroundColor = e.target.value; };
$('fg').oninput = (e) => { box.style.color = e.target.value; };
$('fs').oninput = (e) => { box.style.fontSize = e.target.value + 'px'; };
$('pd').oninput = (e) => { box.style.padding = e.target.value + 'px'; };
$('rd').oninput = (e) => { box.style.borderRadius = e.target.value + 'px'; };
$('rt').oninput = (e) => { box.style.transform = `rotate(${e.target.value}deg)`; };
