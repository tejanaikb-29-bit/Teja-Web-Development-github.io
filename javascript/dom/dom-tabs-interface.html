Create a tabs interface using DOM manipulation

Clicking a tab button shows its panel and hides the others. Arrow keys also move between tabs.

Working demonstration

HTML

HTML gives a page its structure with elements such as headings, lists and forms.

CSS

CSS controls colours, fonts and layout.

JavaScript

JavaScript adds behaviour and interactivity.

<!-- HTML -->
<div class="tabbar" role="tablist"><button role="tab" data-t="0">HTML</button><button role="tab" data-t="1">CSS</button><button role="tab" data-t="2">JavaScript</button></div>
<div class="tabpanel" id="p0"><h4>HTML</h4><p>HTML gives a page its structure with elements such as headings, lists and forms.</p></div>
<div class="tabpanel" id="p1"><h4>CSS</h4><p>CSS controls colours, fonts and layout.</p></div>
<div class="tabpanel" id="p2"><h4>JavaScript</h4><p>JavaScript adds behaviour and interactivity.</p></div>

/* CSS */
.demo .tabbar { display: flex; gap: .2rem; border-bottom: 3px solid #d5dbe6; flex-wrap: wrap; } .demo .tabbar button { background: #eef2fb; color: #1c2437; border-radius: 8px 8px 0 0; } .demo .tabbar button[aria-selected=true] { background: #1f419a; color: #fff; } .demo .tabpanel { padding: 1rem; background: #f7f9fe; border-radius: 0 0 10px 10px; min-height: 100px; }

// JavaScript
const $ = (id) => document.getElementById(id);
const tabs = [...document.querySelectorAll('[role=tab]')];
function select(n) {
  tabs.forEach((t, i) => {
    t.setAttribute('aria-selected', i === n);
    $('p' + i).hidden = i !== n;
  });
}
tabs.forEach((t, i) => {
  t.addEventListener('click', () => select(i));
  t.addEventListener('keydown', (e) => {
    if (e.key === 'ArrowRight') { select((i + 1) % tabs.length); tabs[(i + 1) % tabs.length].focus(); }
    if (e.key === 'ArrowLeft') { select((i + tabs.length - 1) % tabs.length); tabs[(i + tabs.length - 1) % tabs.length].focus(); }
  });
});
select(0);
