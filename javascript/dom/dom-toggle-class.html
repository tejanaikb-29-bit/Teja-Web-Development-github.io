Toggle a CSS class using JavaScript

classList.toggle() adds the class if it is missing and removes it if present. The card switches between two looks.

Working demonstration

Toggle card

Click the button or the card itself.

State: off
View source code of this program 
<!-- HTML -->
<div id="c" class="card2"><h3 style="margin-top:0">Toggle card</h3><p>Click the button or the card itself.</p></div>
<div class="row"><button id="t">Toggle class "active"</button><span>State: <strong id="state">off</strong></span></div>

/* CSS */
.demo .card2 { padding: 1.4rem; border-radius: 10px; background: #fff; border: 2px solid #d5dbe6; transition: .3s; } .demo .card2.active { background: #14213d; color: #f2c200; border-color: #f2c200; transform: scale(1.03); }

// JavaScript
const $ = (id) => document.getElementById(id);
function toggle() {
  const on = $('c').classList.toggle('active');    // returns true when the class was added
  $('state').textContent = on ? 'on' : 'off';
}
$('t').addEventListener('click', toggle);
$('c').addEventListener('click', toggle);
