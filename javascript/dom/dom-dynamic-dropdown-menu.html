Create a dynamic dropdown menu using JavaScript

The <select> is built from an array with createElement(). Choosing an option shows its value.

Working demonstration

View source code of this program 
<!-- HTML -->
<div class="row"><label class="field">Choose a course<select id="sel"></select></label></div><div class="out" id="out"></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const courses = [
  { value: 'html', label: 'HTML Fundamentals' }, { value: 'css', label: 'CSS Layouts' },
  { value: 'js', label: 'JavaScript Basics' }, { value: 'react', label: 'React Introduction' },
];
const sel = $('sel');
const placeholder = new Option('-- Select a course --', '');
sel.add(placeholder);
courses.forEach((c) => {
  const opt = document.createElement('option');
  opt.value = c.value;
  opt.textContent = c.label;
  sel.appendChild(opt);
});
sel.addEventListener('change', () => {
  $('out').textContent = sel.value ? `Selected: ${sel.options[sel.selectedIndex].text}  (value = "${sel.value}")` : 'Nothing selected.';
});
$('out').textContent = 'Nothing selected.';
