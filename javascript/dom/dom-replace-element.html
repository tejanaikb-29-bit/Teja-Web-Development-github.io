Replace an existing HTML element with a new element

Replaces a heading with a paragraph, then a button, using replaceWith() and replaceChild().

Working demonstration

I am an old heading

View source code of this program 
<!-- HTML -->
<div id="parent" class="card"><h3 id="old">I am an old heading</h3></div>
<div class="row"><button id="p">Replace with paragraph</button><button id="b" class="secondary">Replace with button</button><button id="h" class="success">Restore heading</button></div>

// JavaScript
const $ = (id) => document.getElementById(id);
function swapTo(tag, text) {
  const current = $('parent').firstElementChild;
  const fresh = document.createElement(tag);
  fresh.textContent = text;
  $('parent').replaceChild(fresh, current);        // parent.replaceChild(newNode, oldNode)
}
$('p').addEventListener('click', () => swapTo('p', 'I am a new paragraph that replaced the heading.'));
$('b').addEventListener('click', () => {
  const btn = document.createElement('button');
  btn.textContent = 'I am a button now';
  $('parent').firstElementChild.replaceWith(btn);  // element.replaceWith(newNode)
});
$('h').addEventListener('click', () => swapTo('h3', 'I am an old heading'));
