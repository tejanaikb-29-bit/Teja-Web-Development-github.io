Create a dynamic student table using JavaScript

Builds a complete table (header + rows) from an array of student objects, including a computed grade column.

Working demonstration

View source code of this program 
<!-- HTML -->
<div style="overflow-x:auto"><table class="data" id="table"></table></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const students = [
  { roll: 101, name: 'Asha', dept: 'CS', marks: 88 }, { roll: 102, name: 'Ravi', dept: 'Maths', marks: 72 },
  { roll: 103, name: 'Meena', dept: 'Physics', marks: 91 }, { roll: 104, name: 'Kiran', dept: 'CS', marks: 54 },
];
const table = $('table');
const head = table.createTHead().insertRow();
['Roll No', 'Name', 'Department', 'Marks', 'Grade'].forEach((t) => { const th = document.createElement('th'); th.textContent = t; head.appendChild(th); });
const body = table.createTBody();
students.forEach((s) => {
  const row = body.insertRow();
  const grade = s.marks >= 85 ? 'A' : s.marks >= 70 ? 'B' : s.marks >= 55 ? 'C' : 'D';
  [s.roll, s.name, s.dept, s.marks, grade].forEach((v) => { row.insertCell().textContent = v; });
});
