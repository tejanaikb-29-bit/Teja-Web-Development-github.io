Create a dynamic unordered list from an array

Loops over an array, creates an <li> for each item and appends them to a <ul>.

    View source code of this program 
    <!-- HTML -->
<div class="row"><button id="build">Build list</button><button id="shuffle" class="secondary">Shuffle and rebuild</button></div>
<ul id="list"></ul><p class="muted">Array: <code id="arr"></code></p>

// JavaScript
const $ = (id) => document.getElementById(id);
let fruits = ['Apple', 'Banana', 'Mango', 'Orange', 'Grapes', 'Papaya'];
function build() {
  const ul = $('list');
  ul.innerHTML = '';
  fruits.forEach((f) => {
    const li = document.createElement('li');
    li.textContent = f;
    ul.appendChild(li);
  });
  $('arr').textContent = JSON.stringify(fruits);
}
$('build').addEventListener('click', build);
$('shuffle').addEventListener('click', () => { fruits = [...fruits].sort(() => Math.random() - 0.5); build(); });
build();
