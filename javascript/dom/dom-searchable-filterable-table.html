Searchable / filterable table

Type in the search box or pick a department to hide the rows that do not match.

Working demonstration

<!-- HTML -->
<div class="row"><input id="q" type="search" placeholder="Search name or city..."><select id="dept"><option value="">All departments</option><option>CS</option><option>Maths</option><option>Physics</option></select><span class="muted" id="cnt"></span></div>
<table class="data"><thead><tr><th>Name</th><th>Department</th><th>City</th></tr></thead><tbody id="tb"></tbody></table>

// JavaScript
const $ = (id) => document.getElementById(id);
const data = [['Asha', 'CS', 'Chennai'], ['Ravi', 'Maths', 'Madurai'], ['Meena', 'Physics', 'Chennai'], ['Kiran', 'CS', 'Coimbatore'], ['Divya', 'Maths', 'Salem'], ['Arun', 'Physics', 'Trichy'], ['Sara', 'CS', 'Madurai']];
function render() {
  const q = $('q').value.trim().toLowerCase(), d = $('dept').value;
  $('tb').innerHTML = '';
  let shown = 0;
  data.forEach((r) => {
    if ((d && r[1] !== d) || (q && !(r[0] + ' ' + r[2]).toLowerCase().includes(q))) return;
    const tr = $('tb').insertRow();
    r.forEach((v) => { tr.insertCell().textContent = v; });
    shown++;
  });
  if (!shown) { const c = $('tb').insertRow().insertCell(); c.colSpan = 3; c.textContent = 'No matching records.'; }
  $('cnt').textContent = `${shown} of ${data.length} shown`;
}
$('q').addEventListener('input', render);
$('dept').addEventListener('change', render);
render();
