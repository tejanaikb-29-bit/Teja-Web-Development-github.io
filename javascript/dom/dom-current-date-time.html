Dynamically display the current date and time

Shows the date and time in several formats, refreshed every second, with a choice of time zone.

Working demonstration

View source code of this program 
<!-- HTML -->
<label class="field" style="max-width:260px">Time zone<select id="tz"><option value="">My local time</option><option>UTC</option><option>Asia/Kolkata</option><option>America/New_York</option><option>Europe/London</option><option>Asia/Tokyo</option><option>Australia/Sydney</option></select></label>
<div class="out"><div class="big" id="time"></div><div class="fmt"><span id="f1"></span><span id="f2"></span><span id="f3"></span><span id="f4"></span></div></div>

/* CSS */
.demo .big { font: 700 2.4rem 'Courier New', monospace; } .demo .fmt { display: grid; gap: .3rem; margin-top: .6rem; }

// JavaScript
const $ = (id) => document.getElementById(id);
function update() {
  const now = new Date();
  const tz = $('tz').value;
  const opt = tz ? { timeZone: tz } : {};
  $('time').textContent = now.toLocaleTimeString('en-GB', opt);
  $('f1').textContent = 'Long date : ' + now.toLocaleDateString('en-US', { ...opt, weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });
  $('f2').textContent = 'Short date: ' + now.toLocaleDateString('en-GB', opt);
  $('f3').textContent = '12-hour   : ' + now.toLocaleTimeString('en-US', opt);
  $('f4').textContent = 'ISO (UTC) : ' + now.toISOString();
}
$('tz').addEventListener('change', update);
update();
setInterval(update, 1000);
