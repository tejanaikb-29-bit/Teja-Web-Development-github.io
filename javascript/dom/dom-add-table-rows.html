Add new rows to an HTML table dynamically

A small form adds a new row to the table with insertRow() and insertCell().

<!-- HTML -->
<form id="f" class="row"><label class="field">Name<input id="n" type="text"></label><label class="field">Course<input id="c" type="text"></label><label class="field">Fees<input id="fee" type="number" min="0"></label><button>Add row</button></form>
<p class="err" id="err"></p>
<table class="data"><thead><tr><th>#</th><th>Name</th><th>Course</th><th>Fees</th></tr></thead><tbody id="tb"></tbody><tfoot><tr><th colspan="3">Total fees</th><th id="total">0</th></tr></tfoot></table>

// JavaScript
const $ = (id) => document.getElementById(id);
let total = 0;
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  const n = $('n').value.trim(), c = $('c').value.trim(), fee = parseFloat($('fee').value);
  if (!n || !c || Number.isNaN(fee)) { $('err').textContent = 'Fill in all three fields.'; return; }
  $('err').textContent = '';
  const row = $('tb').insertRow();
  [$('tb').rows.length, n, c, fee].forEach((v) => { row.insertCell().textContent = v; });
  total += fee;
  $('total').textContent = total;
  $('f').reset();
  $('n').focus();
});
