Sort table records using JavaScript

Click a column heading to sort ascending, click again for descending. Numbers are sorted numerically.

Working demonstration

NameDepartmentMarksAge

View source code of this program 
<!-- HTML -->
<table class="data"><thead><tr><th class="sortable" data-k="name">Name</th><th class="sortable" data-k="dept">Department</th><th class="sortable" data-k="marks">Marks</th><th class="sortable" data-k="age">Age</th></tr></thead><tbody id="tb"></tbody></table><p class="muted" id="msg"></p>

/* CSS */
.demo th.sortable { cursor: pointer; user-select: none; } .demo th.sortable:hover { background: #d3dcf3; }

// JavaScript
const $ = (id) => document.getElementById(id);
let rows = [
  { name: 'Asha', dept: 'CS', marks: 88, age: 20 }, { name: 'Ravi', dept: 'Maths', marks: 72, age: 21 }, { name: 'Meena', dept: 'Physics', marks: 91, age: 19 },
  { name: 'Kiran', dept: 'CS', marks: 54, age: 22 }, { name: 'Divya', dept: 'Maths', marks: 79, age: 20 },
];
let sortKey = null, asc = true;
function render() {
  $('tb').innerHTML = '';
  rows.forEach((r) => { const tr = $('tb').insertRow(); ['name', 'dept', 'marks', 'age'].forEach((k) => { tr.insertCell().textContent = r[k]; }); });
}
document.querySelectorAll('th.sortable').forEach((th) => {
  th.addEventListener('click', () => {
    const k = th.dataset.k;
    asc = sortKey === k ? !asc : true;
    sortKey = k;
    rows.sort((a, b) => {
      const x = a[k], y = b[k];
      const cmp = typeof x === 'number' ? x - y : String(x).localeCompare(String(y));
      return asc ? cmp : -cmp;
    });
    document.querySelectorAll('th.sortable').forEach((h) => { h.textContent = h.textContent.replace(/ [\u25b2\u25bc]$/, ''); });
    th.textContent += asc ? ' \u25b2' : ' \u25bc';
    $('msg').textContent = `Sorted by ${k} (${asc ? 'ascending' : 'descending'})`;
    render();
  });
});
render();
