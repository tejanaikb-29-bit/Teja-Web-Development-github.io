Change the background colour of an element when a button is clicked

Each button sets the background colour of one box. The box name shows the colour that was applied.

Working demonstration

Current colour: #eee
View source code of this program 
<!-- HTML -->
<div id="box" style="padding:2rem;border-radius:10px;background:#eee;text-align:center;font-weight:700;transition:background .3s">Current colour: #eee</div>
<div class="row"><button data-c="#ffb3b3" class="danger">Red</button><button data-c="#b8e6c4" class="success">Green</button><button data-c="#b3d4ff">Blue</button><button data-c="#fff2a8" class="secondary">Yellow</button><button data-c="#eee" class="secondary">Reset</button></div>

// JavaScript
const $ = (id) => document.getElementById(id);
document.querySelectorAll('button[data-c]').forEach((btn) => {
  btn.addEventListener('click', () => {
    $('box').style.backgroundColor = btn.dataset.c;
    $('box').textContent = 'Current colour: ' + btn.dataset.c;
  });
});
