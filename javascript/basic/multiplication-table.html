Generate the multiplication table of a number

A for loop prints the table from 1 up to the chosen limit.

<!-- HTML -->
<div class="row">
  <label class="field">Number
    <input id="n" type="number" value="7" step="any">
  </label>
  <label class="field">Up to
    <input id="limit" type="number" value="10" step="any">
  </label>
</div>
<button class="btn" id="run">Show table</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('n')) || Number.isNaN(num('limit'))) { show('Please enter valid input.'); return; }
  const n = num('n'), limit = num('limit');
  if (!Number.isInteger(limit) || limit < 1 || limit > 100) return show('Limit must be a whole number from 1 to 100.');
  let table = '';
  for (let i = 1; i <= limit; i++) table += `${n} x ${String(i).padStart(2, ' ')} = ${n * i}\n`;
  show(table);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
