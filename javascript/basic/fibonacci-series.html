Generate the Fibonacci series

Each term is the sum of the previous two terms: 0, 1, 1, 2, 3, 5, 8, ...

<!-- HTML -->
<div class="row">
  <label class="field">How many terms? (1 - 80)
    <input id="n" type="number" value="12" step="any">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('n'))) { show('Please enter valid input.'); return; }
  const n = num('n');
  if (!Number.isInteger(n) || n < 1 || n > 80) return show('Enter a whole number from 1 to 80.');
  let a = 0, b = 1;
  const series = [];
  for (let i = 0; i < n; i++) {
    series.push(a);
    [a, b] = [b, a + b];          // move forward one step
  }
  show(`First ${n} terms:\n${series.join(', ')}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
