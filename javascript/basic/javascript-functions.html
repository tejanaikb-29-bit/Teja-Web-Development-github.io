<!-- HTML -->
<div class="row">
  <label class="field">Number x
    <input id="x" type="number" value="4" step="any">
  </label>
  <label class="field">Number y
    <input id="y" type="number" value="3" step="any">
  </label>
  <label class="field">A name
    <input id="name" type="text" value="Asha">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };
function add(a, b) { return a + b; }                 // declaration with parameters and return value
function greet(who = 'Guest') { return `Hello, ${who}!`; }   // default parameter
const square = function (n) { return n * n; };       // function expression
function factorial(n) { return n <= 1 ? 1 : n * factorial(n - 1); }   // recursion
function applyTwice(fn, value) { return fn(fn(value)); }              // function as an argument

function run() {
  if (Number.isNaN(num('x')) || Number.isNaN(num('y'))) { show('Please enter valid input.'); return; }
  const x = num('x'), y = num('y'), name = txt('name');
  show([
    `add(${x}, ${y}) = ${add(x, y)}`,
    `greet() = ${greet()}`,
    `greet('${name}') = ${greet(name)}`,
    `square(${x}) = ${square(x)}`,
    `factorial(5) = ${factorial(5)}`,
    `applyTwice(square, ${y}) = ${applyTwice(square, y)}`,
  ].join('\n'));
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
