// JavaScript
const $ = (id) => document.getElementById(id);
const site = $('site');
// ---- theme + saved name (localStorage) ----
if (localStorage.getItem('planner-theme') === 'dark') site.classList.add('dark');
$('theme').textContent = site.classList.contains('dark') ? 'Light mode' : 'Dark mode';
$('theme').addEventListener('click', () => {
  site.classList.toggle('dark');
  const dark = site.classList.contains('dark');
  localStorage.setItem('planner-theme', dark ? 'dark' : 'light');
  $('theme').textContent = dark ? 'Light mode' : 'Dark mode';
});
$('today').textContent = new Date().toDateString();
function greet(name) {
  const h = new Date().getHours();
  const part = h < 12 ? 'Good morning' : h < 18 ? 'Good afternoon' : 'Good evening';
  $('greet').textContent = `${part}${name ? ', ' + name : ''}!`;
}
$('who').value = localStorage.getItem('planner-name') || '';
greet($('who').value);
$('hi').addEventListener('click', () => { localStorage.setItem('planner-name', $('who').value.trim()); greet($('who').value.trim()); });

// ---- tabs ----
const tabs = {
  plan: () => { $('panel').innerHTML = '<strong>Tasks done: <span id="n">0</span></strong> <button id="plus">+1</button> <button id="minus" class="secondary">-1</button>'; let n = 0;
    $('plus').onclick = () => { $('n').textContent = ++n; }; $('minus').onclick = () => { if (n > 0) $('n').textContent = --n; }; },
  tips: () => { $('panel').innerHTML = '<ul><li>Study in 25 minute blocks.</li><li>Plan tomorrow tonight.</li><li>Take short breaks.</li></ul>'; },
  about: () => { $('panel').textContent = 'A tiny page that shows HTML, CSS and JavaScript working together.'; },
};
$('tabs').addEventListener('click', (e) => {
  const b = e.target.closest('button');
  if (!b) return;
  document.querySelectorAll('#tabs button').forEach((x) => x.classList.toggle('on', x === b));
  tabs[b.dataset.tab]();
});
tabs.plan();

// ---- contact form validation ----
$('contact').addEventListener('submit', (e) => {
  e.preventDefault();
  const mailOk = /^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test($('cmail').value.trim());
  const msgOk = $('cmsg').value.trim().length >= 10;
  $('cmailMsg').textContent = mailOk ? '' : 'Enter a valid email address.';
  $('cmsgMsg').textContent = msgOk ? '' : 'Message must have at least 10 characters.';
  $('sent').textContent = mailOk && msgOk ? 'Thanks! Your message was validated (demo only).' : '';
});
