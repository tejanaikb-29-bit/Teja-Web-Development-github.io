Greatest common divisor (GCD) of two numbers

Uses Euclid's algorithm: replace (a, b) with (b, a % b) until b becomes 0.

<!-- HTML -->
<div class="row">
  <label class="field">First number
    <input id="a" type="number" value="48" step="any">
  </label>
  <label class="field">Second number
    <input id="b" type="number" value="36" step="any">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('a')) || Number.isNaN(num('b'))) { show('Please enter valid input.'); return; }
  let a = num('a'), b = num('b');
  if (!Number.isInteger(a) || !Number.isInteger(b) || a <= 0 || b <= 0) return show('Enter two positive whole numbers.');
  const x = a, y = b;
  while (b !== 0) {
    [a, b] = [b, a % b];
  }
  show(`GCD of ${x} and ${y} = ${a}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
