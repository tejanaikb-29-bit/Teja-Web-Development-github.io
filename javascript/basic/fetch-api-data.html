Fetch data from an API using the Fetch API

Requests one record from the free JSONPlaceholder API and shows the raw JSON. Needs an internet connection.

Working demonstration

Press "Fetch data".
View source code of this program 
<!-- HTML -->
<div class="row"><label class="field">Todo id (1 - 200)<input id="id" type="number" min="1" max="200" value="1" style="width:120px"></label><button id="go">Fetch data</button></div>
<div class="out" id="out" style="font-family:Consolas,monospace">Press "Fetch data".</div>

// JavaScript
const $ = (id) => document.getElementById(id);
async function getTodo() {
  const id = parseInt($('id').value) || 1;
  $('out').textContent = 'Loading...';
  try {
    const response = await fetch(`https://jsonplaceholder.typicode.com/todos/${id}`);
    if (!response.ok) throw new Error(`Server replied with status ${response.status}`);
    const data = await response.json();                 // convert the JSON text into an object
    $('out').textContent = `Status: ${response.status}\n` + JSON.stringify(data, null, 2);
  } catch (error) {
    $('out').textContent = 'Could not load data: ' + error.message + '\nCheck your internet connection.';
  }
}
$('go').addEventListener('click', getTodo);
