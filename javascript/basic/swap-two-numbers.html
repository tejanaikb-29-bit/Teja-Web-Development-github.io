Swap two numbers

Shows three ways to swap: a temporary variable, arithmetic and array destructuring.

<!-- HTML -->
<div class="row">
  <label class="field">First number (a)
    <input id="a" type="number" value="15" step="any">
  </label>
  <label class="field">Second number (b)
    <input id="b" type="number" value="40" step="any">
  </label>
</div>
<button class="btn" id="run">Swap</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('a')) || Number.isNaN(num('b'))) { show('Please enter valid input.'); return; }
  let a = num('a'), b = num('b');
  const lines = [`Before swap : a = ${a}, b = ${b}`];
  let x = a, y = b;
  let temp = x; x = y; y = temp;                    // 1. temporary variable
  lines.push(`Temp variable: a = ${x}, b = ${y}`);
  x = a; y = b;
  x = x + y; y = x - y; x = x - y;                  // 2. arithmetic, no extra variable
  lines.push(`Arithmetic   : a = ${x}, b = ${y}`);
  x = a; y = b;
  [x, y] = [y, x];                                  // 3. destructuring (ES6)
  lines.push(`Destructuring: a = ${x}, b = ${y}`);
  show(lines.join('\n'));
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
