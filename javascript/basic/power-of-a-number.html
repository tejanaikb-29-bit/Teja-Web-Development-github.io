Calculate the power of a number

Computes base ^ exponent with a loop, Math.pow() and the ** operator.

<!-- HTML -->
<div class="row">
  <label class="field">Base
    <input id="base" type="number" value="2" step="any">
  </label>
  <label class="field">Exponent (whole number, 0 or more)
    <input id="exp" type="number" value="10" step="any">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('base')) || Number.isNaN(num('exp'))) { show('Please enter valid input.'); return; }
  const base = num('base'), exp = num('exp');
  if (!Number.isInteger(exp) || exp < 0) return show('Exponent must be a whole number, 0 or more.');
  let result = 1;
  for (let i = 0; i < exp; i++) result *= base;       // repeated multiplication
  show(`Using a loop      : ${result}\nUsing Math.pow() : ${Math.pow(base, exp)}\nUsing ** operator: ${base ** exp}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
