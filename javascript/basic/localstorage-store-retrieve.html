Store and retrieve data using localStorage

Saves key / value pairs in the browser with setItem(), reads them with getItem() and lists everything stored.

<!-- HTML -->
<div class="row">
  <label class="field">Key<input id="key" type="text" value="username"></label>
  <label class="field">Value<input id="value" type="text" value="Asha"></label>
</div>
<div class="row"><button id="save">Save</button><button id="load" class="secondary">Read value</button><button id="remove" class="danger">Remove key</button></div>
<div class="out" id="out">Nothing yet.</div>
<h4>Everything in localStorage for this page</h4><pre class="out" id="all"></pre>

// JavaScript
const $ = (id) => document.getElementById(id);
function showAll() {
  const items = [];
  for (let i = 0; i < localStorage.length; i++) {
    const k = localStorage.key(i);
    items.push(`${k} = ${localStorage.getItem(k)}`);
  }
  $('all').textContent = items.length ? items.join('\n') : '(empty)';
}
$('save').addEventListener('click', () => {
  const k = $('key').value.trim();
  if (!k) return ($('out').textContent = 'Enter a key.');
  localStorage.setItem(k, $('value').value);
  $('out').textContent = `Saved "${k}". Refresh the page - it is still there.`;
  showAll();
});
$('load').addEventListener('click', () => {
  const v = localStorage.getItem($('key').value.trim());
  $('out').textContent = v === null ? 'No value stored for that key.' : `Value: ${v}`;
});
$('remove').addEventListener('click', () => {
  localStorage.removeItem($('key').value.trim());
  $('out').textContent = 'Key removed.';
  showAll();
});
showAll();
