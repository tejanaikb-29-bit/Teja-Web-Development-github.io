Check whether a string is a palindrome

Ignores case, spaces and punctuation, then compares the text with its reverse.

<!-- HTML -->
<div class="row">
  <label class="field">Enter a word or sentence
    <input id="s" type="text" value="A man, a plan, a canal: Panama">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  const s = txt('s');
  const clean = s.toLowerCase().replace(/[^a-z0-9]/g, '');
  const reversed = clean.split('').reverse().join('');
  show(`Cleaned : ${clean}\nReversed: ${reversed}\n"${s}" ${clean && clean === reversed ? 'IS' : 'is NOT'} a palindrome`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
