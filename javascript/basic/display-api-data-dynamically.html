Display API data dynamically on a webpage

Loads a list of users from an API and builds a card for each one with DOM methods. Includes a live search box.

Working demonstration

View source code of this program 
<!-- HTML -->
<div class="row"><button id="load">Load users</button><input id="search" type="search" placeholder="Filter by name..."></div>
<p id="status" class="muted"></p><div class="users" id="users"></div>

/* CSS */
.demo .users { display: grid; grid-template-columns: repeat(auto-fill, minmax(220px, 1fr)); gap: 1rem; margin-top: 1rem; }
.demo .ucard { border: 1px solid #d5dbe6; border-radius: 10px; padding: 1rem; background: #fff; }
.demo .ucard h4 { margin: 0 0 .3rem; } .demo .ucard p { margin: .15rem 0; font-size: .9rem; color: #55637d; }

// JavaScript
const $ = (id) => document.getElementById(id);
let users = [];
function render() {
  const q = $('search').value.toLowerCase();
  const box = $('users');
  box.innerHTML = '';
  users.filter((u) => u.name.toLowerCase().includes(q)).forEach((u) => {
    const card = document.createElement('div');
    card.className = 'ucard';
    card.innerHTML = `<h4></h4><p></p><p></p><p></p>`;
    const [h, p1, p2, p3] = card.children;
    h.textContent = u.name;
    p1.textContent = 'Email: ' + u.email;
    p2.textContent = 'City: ' + u.address.city;
    p3.textContent = 'Company: ' + u.company.name;
    box.appendChild(card);
  });
}
$('load').addEventListener('click', async () => {
  $('status').textContent = 'Loading...';
  try {
    const res = await fetch('https://jsonplaceholder.typicode.com/users');
    if (!res.ok) throw new Error('HTTP ' + res.status);
    users = await res.json();
    $('status').textContent = `${users.length} users loaded.`;
    render();
  } catch (e) {
    $('status').textContent = 'Could not load users (' + e.message + '). Check your internet connection.';
  }
});
$('search').addEventListener('input', render);
