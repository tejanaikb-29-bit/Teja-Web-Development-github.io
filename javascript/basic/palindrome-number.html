Check whether a number is a palindrome

A palindrome number reads the same forwards and backwards, like 12321.

Working demonstration

The result will appear here.
View source code of this program 
<!-- HTML -->
<div class="row">
  <label class="field">Enter a non-negative integer
    <input id="n" type="number" value="12321" step="any">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('n'))) { show('Please enter valid input.'); return; }
  const n = num('n');
  if (!Number.isInteger(n) || n < 0) return show('Enter a non-negative whole number.');
  let temp = n, reversed = 0;
  while (temp > 0) {
    reversed = reversed * 10 + (temp % 10);
    temp = Math.floor(temp / 10);
  }
  show(reversed === n ? `${n} is a PALINDROME` : `${n} is NOT a palindrome (reverse = ${reversed})`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
