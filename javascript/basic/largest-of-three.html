Find the largest of three numbers

Uses if / else if to compare three numbers and print the largest.

The result will appear here.
View source code of this program 
<!-- HTML -->
<div class="row">
  <label class="field">First
    <input id="a" type="number" value="12" step="any">
  </label>
  <label class="field">Second
    <input id="b" type="number" value="47" step="any">
  </label>
  <label class="field">Third
    <input id="c" type="number" value="30" step="any">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('a')) || Number.isNaN(num('b')) || Number.isNaN(num('c'))) { show('Please enter valid input.'); return; }
  const a = num('a'), b = num('b'), c = num('c');
  let largest;
  if (a >= b && a >= c) largest = a;
  else if (b >= a && b >= c) largest = b;
  else largest = c;
  show(`The largest of ${a}, ${b} and ${c} is ${largest}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
