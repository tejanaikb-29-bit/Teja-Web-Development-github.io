Counter application with increment and decrement

Two buttons change a number. The number turns green for positive and red for negative values.

Working demonstration

0
View source code of this program 
<!-- HTML -->
<div class="num" id="value">0</div>
<div class="row" style="justify-content:center"><button id="dec" class="danger">- Decrement</button><button id="reset" class="secondary">Reset</button><button id="inc" class="success">+ Increment</button></div>

/* CSS */
.demo .num { font: 800 4rem/1 'Courier New', monospace; text-align: center; margin: 1rem 0; transition: color .2s; }

// JavaScript
const $ = (id) => document.getElementById(id);
let count = 0;
function update() {
  $('value').textContent = count;
  $('value').style.color = count > 0 ? '#178a4a' : count < 0 ? '#c62828' : '#1c2437';
}
$('inc').addEventListener('click', () => { count++; update(); });
$('dec').addEventListener('click', () => { count--; update(); });
$('reset').addEventListener('click', () => { count = 0; update(); });
update();
