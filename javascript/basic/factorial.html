Find the factorial of a number

Multiplies 1 x 2 x ... x n using a loop. BigInt is used so large results stay exact.

<!-- HTML -->
<div class="row">
  <label class="field">Enter n (0 - 500)
    <input id="n" type="number" value="10" step="any">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (Number.isNaN(num('n'))) { show('Please enter valid input.'); return; }
  const n = num('n');
  if (!Number.isInteger(n) || n < 0 || n > 500) return show('Enter a whole number between 0 and 500.');
  let fact = 1n;
  for (let i = 2n; i <= BigInt(n); i++) fact *= i;
  show(`${n}! = ${fact}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
