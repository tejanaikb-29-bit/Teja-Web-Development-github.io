The Window object

window properties (size, position, scroll), the window.open() / close() methods and live updates when you resize or scroll.

Working demonstration

Resize or scroll the window and watch the values change.

View source code of this program 
<!-- HTML -->
<table class="data"><tbody id="tb"></tbody></table><div class="row"><button id="open">window.open() a small window</button><button id="close" class="secondary">Close it</button><button id="scroll" class="secondary">window.scrollTo(0, 0)</button></div><p class="muted" id="msg">Resize or scroll the window and watch the values change.</p>

/* CSS */
.demo table.data th { width: 38%; }

// JavaScript
const $ = (id) => document.getElementById(id);
function update() {
  const rows = [
    ['window.innerWidth x innerHeight', `${window.innerWidth} x ${window.innerHeight} px (viewport)`],
    ['window.outerWidth x outerHeight', `${window.outerWidth} x ${window.outerHeight} px (whole browser window)`],
    ['window.screenX, screenY', `${window.screenX}, ${window.screenY}`],
    ['window.scrollX, scrollY', `${Math.round(window.scrollX)}, ${Math.round(window.scrollY)}`],
    ['window.devicePixelRatio', window.devicePixelRatio],
    ['window.name', window.name || '(empty)'],
    ['window.history.length', window.history.length],
    ['window.self === window', window.self === window],
    ['window.frames.length', window.frames.length],
  ];
  $('tb').innerHTML = '';
  rows.forEach(([k, v]) => { const tr = $('tb').insertRow(); tr.insertCell().textContent = k; tr.insertCell().textContent = v; });
}
let popup = null;
$('open').addEventListener('click', () => {
  popup = window.open('', 'demoPopup', 'width=380,height=220');
  if (!popup) { $('msg').textContent = 'The browser blocked the popup. Allow popups for this page.'; return; }
  popup.document.write('<h3 style="font-family:sans-serif">Hello from a new window</h3><p style="font-family:sans-serif">Created with window.open().</p>');
  $('msg').textContent = 'A new window was opened.';
});
$('close').addEventListener('click', () => { if (popup && !popup.closed) { popup.close(); $('msg').textContent = 'The window was closed with popup.close().'; } else $('msg').textContent = 'No open popup.'; });
$('scroll').addEventListener('click', () => window.scrollTo({ top: 0, behavior: 'smooth' }));
window.addEventListener('resize', update);
window.addEventListener('scroll', update);
update();
