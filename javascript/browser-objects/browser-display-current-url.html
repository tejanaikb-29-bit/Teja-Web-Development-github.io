Display the current webpage URL

Reads location.href, the page title and document.referrer and offers a copy button.

Working demonstration

File name
Referrer
View source code of this program 
<!-- HTML -->
<div class="url" id="url"></div><table class="data" style="margin-top:1rem"><tr><th>Page title</th><td id="t"></td></tr><tr><th>File name</th><td id="f"></td></tr><tr><th>Referrer</th><td id="r"></td></tr></table><div class="row"><button id="copy">Copy URL</button></div>

/* CSS */
.demo .url { font: 600 1rem Consolas, monospace; word-break: break-all; padding: 1rem; background: #101828; color: #5eead4; border-radius: 10px; }

// JavaScript
const $ = (id) => document.getElementById(id);
$('url').textContent = window.location.href;
$('t').textContent = document.title;
$('f').textContent = location.pathname.split('/').pop() || '(none)';
$('r').textContent = document.referrer || '(none - opened directly)';
$('copy').addEventListener('click', async () => {
  try { await navigator.clipboard.writeText(location.href); $('copy').textContent = 'Copied!'; }
  catch (e) { $('copy').textContent = 'Copy failed - select the text manually'; }
  setTimeout(() => { $('copy').textContent = 'Copy URL'; }, 1600);
});
