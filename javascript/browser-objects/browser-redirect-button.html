Button to redirect the user to another webpage

location.href, location.assign() and location.replace() send the visitor to another page. An automatic redirect with a cancel option is included.

Working demonstration

Automatic redirect
Press the button to start a 5 second countdown to the home page.
View source code of this program 
<!-- HTML -->
<div class="row"><button id="home">Go to Home (location.href)</button><button id="list" class="secondary">Go to JavaScript list (location.assign)</button><button id="ext" class="success">Open MDN (new tab)</button></div>
<div class="out"><strong>Automatic redirect</strong><br><span id="msg">Press the button to start a 5 second countdown to the home page.</span></div><div class="row"><button id="auto">Redirect in 5 seconds</button><button id="cancel" class="danger">Cancel</button></div>

// JavaScript
const $ = (id) => document.getElementById(id);
$('home').addEventListener('click', () => { location.href = '../index.html'; });
$('list').addEventListener('click', () => { location.assign('index.html'); });     // keeps this page in history (Back works)
$('ext').addEventListener('click', () => { window.open('https://developer.mozilla.org/', '_blank', 'noopener'); });
let timer = null;
$('auto').addEventListener('click', () => {
  clearInterval(timer);
  let left = 5;
  $('msg').textContent = `Redirecting in ${left}...`;
  timer = setInterval(() => {
    left--;
    if (left <= 0) { clearInterval(timer); location.replace('../index.html'); }      // replace() removes this page from history
    else $('msg').textContent = `Redirecting in ${left}...`;
  }, 1000);
});
$('cancel').addEventListener('click', () => { clearInterval(timer); $('msg').textContent = 'Redirect cancelled.'; });
