Display the user's current coordinates using the Geolocation API

Shows latitude, longitude, accuracy, altitude and speed, a link to the map, and can keep tracking with watchPosition().

<!-- HTML -->
<div class="row"><button id="get">Get my coordinates</button><button id="watch" class="secondary">Start tracking</button></div><table class="data"><tbody id="tb"><tr><td colspan="2" class="muted">No data yet.</td></tr></tbody></table><p id="map"></p><p class="err" id="err"></p>

/* CSS */
.demo table.data th { width: 38%; }

// JavaScript
const $ = (id) => document.getElementById(id);
let watchId = null;
function show(pos) {
  const c = pos.coords;
  const rows = [['Latitude', c.latitude.toFixed(6)], ['Longitude', c.longitude.toFixed(6)], ['Accuracy', Math.round(c.accuracy) + ' metres'], ['Altitude', c.altitude === null ? 'not available' : Math.round(c.altitude) + ' m'], ['Speed', c.speed === null ? 'not available' : c.speed.toFixed(1) + ' m/s'], ['Time', new Date(pos.timestamp).toLocaleTimeString()]];
  $('tb').innerHTML = '';
  rows.forEach(([k, v]) => { const tr = $('tb').insertRow(); tr.insertCell().textContent = k; tr.insertCell().textContent = v; });
  $('map').innerHTML = `<a target="_blank" rel="noopener" href="https://www.openstreetmap.org/?mlat=${c.latitude}&mlon=${c.longitude}#map=16/${c.latitude}/${c.longitude}">Open this position on OpenStreetMap</a>`;
  $('err').textContent = '';
}
function fail(err) { $('err').textContent = err.code === 1 ? 'Permission denied.' : 'Could not get position: ' + err.message; }
$('get').addEventListener('click', () => {
  if (!navigator.geolocation) return ($('err').textContent = 'Geolocation not supported.');
  navigator.geolocation.getCurrentPosition(show, fail, { enableHighAccuracy: true, timeout: 10000 });
});
$('watch').addEventListener('click', () => {
  if (!navigator.geolocation) return;
  if (watchId !== null) { navigator.geolocation.clearWatch(watchId); watchId = null; $('watch').textContent = 'Start tracking'; return; }
  watchId = navigator.geolocation.watchPosition(show, fail, { enableHighAccuracy: true });
  $('watch').textContent = 'Stop tracking';
});
