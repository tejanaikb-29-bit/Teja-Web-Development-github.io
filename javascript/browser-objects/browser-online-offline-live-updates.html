Webpage that updates information when the browser goes online / offline

A banner, an icon and an event history change instantly using the online and offline events. Unsaved messages are queued while offline.

      <!-- HTML -->
<div class="banner-net" id="banner"></div><div class="row"><input id="msg" type="text" placeholder="Type a message" style="flex:1;min-width:180px"><button id="send">Send</button></div><p><strong>Queued (waiting for internet):</strong></p><ul class="queue" id="queue"></ul><p><strong>Sent:</strong></p><ul id="sent"></ul><div class="out" id="log" style="font-size:.85rem">Event history</div>

/* CSS */
.demo .banner-net { padding: .8rem 1rem; border-radius: 10px; font-weight: 700; color: #fff; margin-bottom: .8rem; transition: background .3s; } .demo .queue li { margin: .2rem 0; }

// JavaScript
const $ = (id) => document.getElementById(id);
const queue = [];
const stamp = () => new Date().toLocaleTimeString();
function log(t) { $('log').textContent = `${stamp()}  ${t}\n` + $('log').textContent.replace('Event history', '').trim(); }
function render() {
  const on = navigator.onLine;
  $('banner').textContent = on ? 'You are back online.' : 'You are offline. Messages will be queued.';
  $('banner').style.background = on ? '#178a4a' : '#c62828';
  $('queue').innerHTML = queue.length ? '' : '<li class="muted">(none)</li>';
  queue.forEach((m) => { const li = document.createElement('li'); li.textContent = m; $('queue').appendChild(li); });
}
function deliver(m) { const li = document.createElement('li'); li.textContent = `${m}  (sent ${stamp()})`; $('sent').appendChild(li); }
$('send').addEventListener('click', () => {
  const m = $('msg').value.trim();
  if (!m) return;
  $('msg').value = '';
  if (navigator.onLine) { deliver(m); log('Message sent immediately'); } else { queue.push(m); log('Offline - message queued'); }
  render();
});
window.addEventListener('offline', () => { log('offline event'); render(); });
window.addEventListener('online', () => { log('online event - sending queued messages'); while (queue.length) deliver(queue.shift()); render(); });
render();
