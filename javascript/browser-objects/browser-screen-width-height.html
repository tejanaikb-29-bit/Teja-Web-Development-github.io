Display the user's screen width and height

Compares the screen size, the available screen area and the browser viewport. The values update when you resize the window.

Working demonstration

Screen sizescreen.width x height
Available areascreen.availWidth x availHeight
Browser viewportwindow.innerWidth x innerHeight
Pixel ratio
Orientation

<!-- HTML -->
<div class="cards"><div>Screen size<b id="scr"></b>screen.width x height</div><div>Available area<b id="avail"></b>screen.availWidth x availHeight</div><div>Browser viewport<b id="vp"></b>window.innerWidth x innerHeight</div></div><table class="data" style="margin-top:1rem"><tr><th>Pixel ratio</th><td id="dpr"></td></tr><tr><th>Orientation</th><td id="ori"></td></tr><tr><th>Colour depth</th><td id="cd"></td></tr></table>

/* CSS */
.demo .cards { display: grid; grid-template-columns: repeat(auto-fit, minmax(180px, 1fr)); gap: .8rem; } .demo .cards div { background: #f1f4fa; border-radius: 10px; padding: 1rem; text-align: center; } .demo .cards b { display: block; font-size: 1.5rem; color: #1f419a; }

// JavaScript
const $ = (id) => document.getElementById(id);
function update() {
  $('scr').textContent = `${screen.width} x ${screen.height}`;
  $('avail').textContent = `${screen.availWidth} x ${screen.availHeight}`;
  $('vp').textContent = `${window.innerWidth} x ${window.innerHeight}`;
  $('dpr').textContent = window.devicePixelRatio;
  $('ori').textContent = (screen.orientation && screen.orientation.type) || (innerWidth > innerHeight ? 'landscape' : 'portrait');
  $('cd').textContent = screen.colorDepth + ' bit';
}
window.addEventListener('resize', update);
update();
