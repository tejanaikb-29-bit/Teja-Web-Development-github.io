Program using setTimeout()

setTimeout() runs a function once after a delay. You can cancel it with clearTimeout() before it fires.

Working demonstration

Nothing scheduled.
View source code of this program 
<!-- HTML -->
<div class="row"><label class="field">Delay (seconds)<input id="sec" type="number" min="1" max="10" value="3" style="width:110px"></label><button id="start">Start timeout</button><button id="cancel" class="danger">clearTimeout()</button></div><div class="out" id="out">Nothing scheduled.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
let id = null, ticker = null;
$('start').addEventListener('click', () => {
  clearTimeout(id); clearInterval(ticker);
  const sec = Math.min(10, Math.max(1, parseInt($('sec').value) || 3));
  let left = sec;
  $('out').textContent = `Timeout scheduled: message in ${left} second(s)...`;
  ticker = setInterval(() => { left--; if (left > 0) $('out').textContent = `Timeout scheduled: message in ${left} second(s)...`; }, 1000);
  id = setTimeout(() => { clearInterval(ticker); $('out').textContent = `The timeout finished after ${sec} second(s). This ran only once!`; id = null; }, sec * 1000);
});
$('cancel').addEventListener('click', () => { clearTimeout(id); clearInterval(ticker); id = null; $('out').textContent = 'Timeout cancelled with clearTimeout().'; });
