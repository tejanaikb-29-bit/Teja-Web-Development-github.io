alert(), confirm() and prompt()

The three built-in dialog boxes: alert shows a message, confirm returns true / false and prompt returns the typed text.

Working demonstration

Click a button.
View source code of this program 
<!-- HTML -->
<div class="row"><button id="a">alert()</button><button id="c" class="secondary">confirm()</button><button id="p" class="success">prompt()</button></div><div class="out" id="out">Click a button.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
$('a').addEventListener('click', () => {
  alert('This is an alert box. It only has an OK button.');
  $('out').textContent = 'alert() shown - it returns nothing (undefined).';
});
$('c').addEventListener('click', () => {
  const answer = confirm('Do you want to continue?');       // true for OK, false for Cancel
  $('out').textContent = `confirm() returned: ${answer}\n${answer ? 'You pressed OK.' : 'You pressed Cancel.'}`;
});
$('p').addEventListener('click', () => {
  const name = prompt('What is your name?', 'Guest');       // string, or null if Cancel is pressed
  $('out').textContent = name === null ? 'prompt() returned null (Cancel pressed).' : name.trim() === '' ? 'prompt() returned an empty string.' : `prompt() returned: "${name}"\nHello, ${name}!`;
});
