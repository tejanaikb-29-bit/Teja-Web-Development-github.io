Display the current browser language

Shows navigator.language, the full language list and formats a date and number in that language.

<div class="out"><div style="font-size:1.6rem;font-weight:800" id="main"></div><div id="list"></div></div><table class="data" style="margin-top:1rem"><tr><th>Date in your language</th><td id="d"></td></tr><tr><th>Number in your language</th><td id="n"></td></tr><tr><th>Currency</th><td id="c"></td></tr></table>

// JavaScript
const $ = (id) => document.getElementById(id);
// Some systems report unusual language tags, so fall back to en-US if a tag is not valid
const safeLocale = (tag) => { try { new Intl.DateTimeFormat(tag); return tag; } catch (e) { return 'en-US'; } };
function update() {
  const loc = safeLocale($('loc').value || navigator.language);
  const short = loc.split('-')[0];
  let full = short;
  try { full = new Intl.DisplayNames([loc], { type: 'language' }).of(short); } catch (e) { /* keep the code */ }
  $('main').textContent = `${loc}  (${full})`;
  $('list').textContent = 'navigator.languages: ' + navigator.languages.join(', ');
  const now = new Date();
  $('d').textContent = now.toLocaleDateString(loc, { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' });
  $('n').textContent = (1234567.89).toLocaleString(loc);
  $('c').textContent = (2499.5).toLocaleString(loc, { style: 'currency', currency: 'INR' });
}
$('loc').addEventListener('change', update);
update();
