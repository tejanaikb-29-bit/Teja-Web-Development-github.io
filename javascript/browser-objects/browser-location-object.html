The Location object

Shows every part of the current address (protocol, host, path, search, hash) and lets you change the hash and query string.

Working demonstration

Pages opened directly from a folder use the file: protocol, so host and port may be empty.

View source code of this program 
<!-- HTML -->
<table class="data"><tbody id="tb"></tbody></table><div class="row"><button id="hash">Set location.hash = "#section2"</button><button id="clearHash" class="secondary">Clear hash</button><button id="reload" class="secondary">location.reload()</button></div><p class="muted">Pages opened directly from a folder use the file: protocol, so host and port may be empty.</p>

/* CSS */
.demo table.data th { width: 38%; } .demo td { word-break: break-all; }

// JavaScript
const $ = (id) => document.getElementById(id);
function update() {
  const keys = ['href', 'protocol', 'host', 'hostname', 'port', 'pathname', 'search', 'hash', 'origin'];
  $('tb').innerHTML = '';
  keys.forEach((k) => { const tr = $('tb').insertRow(); tr.insertCell().textContent = 'location.' + k; tr.insertCell().textContent = location[k] || '(empty)'; });
}
$('hash').addEventListener('click', () => { location.hash = '#section2'; update(); });
$('clearHash').addEventListener('click', () => { history.replaceState(null, '', location.pathname + location.search); update(); });
$('reload').addEventListener('click', () => location.reload());
window.addEventListener('hashchange', update);
update();
