Detect whether the browser is online or offline

Reads navigator.onLine and lets you test a real network request. Turn off your Wi-Fi to see the status change.

Working demonstration

navigator.onLine only says whether a network is connected, not whether the internet works.
View source code of this program 
<!-- HTML -->
<div class="status" id="status"></div><div class="row"><button id="test">Test connection with a real request</button></div><div class="out" id="out">navigator.onLine only says whether a network is connected, not whether the internet works.</div>

/* CSS */
.demo .status { text-align: center; padding: 1.6rem; border-radius: 14px; font: 800 1.8rem 'Segoe UI', sans-serif; color: #fff; transition: background .3s; }

// JavaScript
const $ = (id) => document.getElementById(id);
function update() {
  const online = navigator.onLine;
  $('status').textContent = online ? 'ONLINE' : 'OFFLINE';
  $('status').style.background = online ? '#178a4a' : '#c62828';
}
window.addEventListener('online', update);
window.addEventListener('offline', update);
$('test').addEventListener('click', async () => {
  $('out').textContent = 'Testing...';
  try {
    await fetch('https://jsonplaceholder.typicode.com/todos/1', { cache: 'no-store' });
    $('out').textContent = 'Request succeeded - the internet is reachable.';
  } catch (e) { $('out').textContent = 'Request failed - no internet access.'; }
});
update();
