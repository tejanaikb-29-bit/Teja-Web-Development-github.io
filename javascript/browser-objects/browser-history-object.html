The History object

history.length, history.pushState() and the popstate event: add fake history entries and move through them with Back / Forward.

<!-- HTML -->
<div class="row"><button id="push">history.pushState()</button><button id="back" class="secondary">history.back()</button><button id="fwd" class="secondary">history.forward()</button><button id="go" class="secondary">history.go(-2)</button></div><p>history.length: <strong id="len"></strong> &middot; current state: <strong id="state">none</strong></p><div class="out" id="log" style="font-family:Consolas,monospace;font-size:.85rem">Log</div>

// JavaScript
const $ = (id) => document.getElementById(id);
let n = 0;
const log = (t) => { $('log').textContent = t + '\n' + $('log').textContent.split('\n').slice(0, 6).join('\n'); };
function update() { $('len').textContent = history.length; $('state').textContent = history.state ? JSON.stringify(history.state) : 'none'; }
$('push').addEventListener('click', () => { n++; history.pushState({ step: n }, '', '#step' + n); log(`pushState({step: ${n}}) - URL now ends with #step${n}`); update(); });
$('back').addEventListener('click', () => history.back());
$('fwd').addEventListener('click', () => history.forward());
$('go').addEventListener('click', () => history.go(-2));
window.addEventListener('popstate', (e) => { log('popstate event, state = ' + JSON.stringify(e.state)); update(); });
update();
