The Navigator object

Reads properties of the navigator object: userAgent, platform, language, online status, cookies, hardware and more.

Working demonstration

View source code of this program 
<!-- HTML -->
<table class="data"><tbody id="tb"></tbody></table>

/* CSS */
.demo table.data th { width: 38%; } .demo td { word-break: break-word; }

// JavaScript
const $ = (id) => document.getElementById(id);
const n = navigator;
const rows = [
  ['navigator.userAgent', n.userAgent], ['navigator.platform', n.platform || '(not provided)'], ['navigator.vendor', n.vendor || '(none)'],
  ['navigator.language', n.language], ['navigator.languages', (n.languages || []).join(', ')], ['navigator.onLine', n.onLine],
  ['navigator.cookieEnabled', n.cookieEnabled], ['navigator.hardwareConcurrency', (n.hardwareConcurrency || '?') + ' logical CPU cores'],
  ['navigator.deviceMemory', n.deviceMemory ? n.deviceMemory + ' GB (approx.)' : '(not available)'], ['navigator.maxTouchPoints', n.maxTouchPoints],
  ['"geolocation" in navigator', 'geolocation' in n], ['"clipboard" in navigator', 'clipboard' in n], ['"serviceWorker" in navigator', 'serviceWorker' in n],
];
rows.forEach(([k, v]) => { const tr = $('tb').insertRow(); tr.insertCell().textContent = k; tr.insertCell().textContent = v; });
