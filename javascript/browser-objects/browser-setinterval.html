Program using setInterval()

setInterval() repeats a function at a fixed interval until clearInterval() stops it. Change the speed while it runs.

<!-- HTML -->
<div class="big" id="n">0</div><div class="row" style="justify-content:center"><label class="field">Interval (ms)<input id="ms" type="number" value="1000" min="100" step="100" style="width:120px"></label><button id="start" class="success">Start</button><button id="stop" class="danger">Stop</button><button id="reset" class="secondary">Reset</button></div><p class="center muted" id="msg"></p>

/* CSS */
.demo .big { font: 800 4rem 'Courier New', monospace; text-align: center; }

// JavaScript
const $ = (id) => document.getElementById(id);
let n = 0, id = null;
function start() {
  if (id) clearInterval(id);
  const ms = Math.max(100, parseInt($('ms').value) || 1000);
  id = setInterval(() => { $('n').textContent = ++n; }, ms);
  $('msg').textContent = `Running: the number increases every ${ms} ms.`;
}
$('start').addEventListener('click', start);
$('stop').addEventListener('click', () => { clearInterval(id); id = null; $('msg').textContent = 'Stopped with clearInterval().'; });
$('reset').addEventListener('click', () => { n = 0; $('n').textContent = 0; });
