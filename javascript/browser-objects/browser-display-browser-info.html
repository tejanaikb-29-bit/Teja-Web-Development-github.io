Display browser information using JavaScript

Detects the browser name, version and operating system from the user-agent string and shows extra device details.

Working demonstration

View source code of this program 
<!-- HTML -->
<div id="badges"></div><table class="data" style="margin-top:.8rem"><tbody id="tb"></tbody></table>

/* CSS */
.demo .badge { display: inline-block; padding: .3rem .9rem; border-radius: 999px; background: #14213d; color: #f2c200; font-weight: 700; margin: .2rem .3rem .2rem 0; }.demo table.data th { width: 38%; }

// JavaScript
const $ = (id) => document.getElementById(id);
const ua = navigator.userAgent;
function detectBrowser() {
  const tests = [['Edge', /Edg\/([\d.]+)/], ['Opera', /OPR\/([\d.]+)/], ['Chrome', /Chrome\/([\d.]+)/], ['Firefox', /Firefox\/([\d.]+)/], ['Safari', /Version\/([\d.]+).*Safari/]];
  for (const [name, re] of tests) { const m = ua.match(re); if (m) return [name, m[1]]; }
  return ['Unknown', ''];
}
function detectOS() {
  if (/Windows/.test(ua)) return 'Windows'; if (/Android/.test(ua)) return 'Android'; if (/iPhone|iPad|iPod/.test(ua)) return 'iOS';
  if (/Mac OS X/.test(ua)) return 'macOS'; if (/CrOS/.test(ua)) return 'Chrome OS'; if (/Linux/.test(ua)) return 'Linux'; return 'Unknown';
}
const [browser, version] = detectBrowser();
const mobile = /Mobi|Android|iPhone|iPad/.test(ua);
[`Browser: ${browser} ${version.split('.')[0]}`, `OS: ${detectOS()}`, mobile ? 'Mobile device' : 'Desktop / laptop'].forEach((t) => { const s = document.createElement('span'); s.className = 'badge'; s.textContent = t; $('badges').appendChild(s); });
[['Browser', `${browser} ${version}`], ['Operating system', detectOS()], ['Device type', mobile ? 'Mobile' : 'Desktop'], ['Language', navigator.language], ['Time zone', Intl.DateTimeFormat().resolvedOptions().timeZone],
 ['Screen', `${screen.width} x ${screen.height}`], ['Colour depth', screen.colorDepth + ' bit'], ['Touch screen', navigator.maxTouchPoints > 0 ? 'Yes' : 'No'], ['Cookies enabled', navigator.cookieEnabled], ['Online', navigator.onLine],
 ['Dark mode preferred', matchMedia('(prefers-color-scheme: dark)').matches ? 'Yes' : 'No'], ['Full user-agent', ua]].forEach(([k, v]) => { const tr = $('tb').insertRow(); tr.insertCell().textContent = k; tr.insertCell().textContent = v; });
