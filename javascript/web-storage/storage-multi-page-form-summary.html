<!-- HTML -->
<p class="muted"><strong>Step 3 of 3</strong> &ndash; Summary</p><table class="data" id="tb"></table><p id="empty" class="err" hidden>No data found. Please start from step 1.</p><div id="done" class="banner good" hidden>Form submitted! The temporary data has been cleared.</div><div class="row" id="btns"><a class="btn secondary" href="storage-multi-page-form-step2.html">&larr; Back</a><button id="submit" class="success">Submit</button></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const KEY = 'multi-page-form';
const data = JSON.parse(sessionStorage.getItem(KEY) || '{}');
const labels = { name: 'Full name', email: 'Email', phone: 'Phone', city: 'City', course: 'Course' };
const filled = Object.keys(labels).filter((k) => data[k]);
filled.forEach((k) => { const tr = $('tb').insertRow(); tr.insertCell().textContent = labels[k]; tr.insertCell().textContent = data[k]; });
if (!filled.length) { $('empty').hidden = false; $('submit').disabled = true; }
$('submit').addEventListener('click', () => {
  sessionStorage.removeItem(KEY);
  $('done').hidden = false; $('btns').hidden = true; $('tb').hidden = true;
});
