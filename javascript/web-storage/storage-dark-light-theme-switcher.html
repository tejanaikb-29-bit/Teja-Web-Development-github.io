Dark / light theme switcher using localStorage

Switch the theme of this whole page. The choice is stored and applied again the next time you open it.

Working demonstration

Current theme:

Open another page of this project after choosing dark: the choice is stored under the key theme, so pages that read it can reuse it.

View source code of this program 
<!-- HTML -->
<div class="sw"><button id="toggle">Switch theme</button><span>Current theme: <strong id="cur"></strong></span></div><p>Open another page of this project after choosing dark: the choice is stored under the key <code>theme</code>, so pages that read it can reuse it.</p><label>Or choose: <select id="sel"><option value="light">Light</option><option value="dark">Dark</option></select></label>

/* CSS */
body.dark-theme { background: #0b1220; color: #e6ecf7; }
body.dark-theme .pp-demo { background: #1a2338; border-color: #2d3a58; color: #e6ecf7; }
body.dark-theme .out { background: #0f172a; }
.demo .sw { display: inline-flex; align-items: center; gap: .6rem; }

// JavaScript
const $ = (id) => document.getElementById(id);
function apply(theme) {
  document.body.classList.toggle('dark-theme', theme === 'dark');
  $('cur').textContent = theme;
  $('sel').value = theme;
  localStorage.setItem('theme', theme);
}
apply(localStorage.getItem('theme') || 'light');       // restore saved choice on load
$('toggle').addEventListener('click', () => apply(document.body.classList.contains('dark-theme') ? 'light' : 'dark'));
$('sel').addEventListener('change', () => apply($('sel').value));
