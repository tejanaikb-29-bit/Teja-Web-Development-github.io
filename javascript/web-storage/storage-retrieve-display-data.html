Retrieve and display data from localStorage

localStorage.getItem(key) reads the saved value on page load. If nothing is stored it returns null.

Working demonstration

Tip: press F12, open Application → Local Storage to see the stored data.

View source code of this program 
<!-- HTML -->
<div class="row"><label class="field">Username<input id="user" type="text"></label><button id="save">Save</button></div><h2 id="hello"></h2><div class="out" id="out"></div><p class="muted">Tip: press F12, open <em>Application &rarr; Local Storage</em> to see the stored data.</p>

// JavaScript
const $ = (id) => document.getElementById(id);
function show() {
  const name = localStorage.getItem('username');     // returns null when the key does not exist
  $('hello').textContent = name ? `Welcome back, ${name}!` : 'Hello, stranger!';
  $('out').textContent = `localStorage.getItem('username') returned: ${name === null ? 'null (nothing stored)' : '"' + name + '"'}`;
}
$('save').addEventListener('click', () => { if ($('user').value.trim()) { localStorage.setItem('username', $('user').value.trim()); show(); } });
show();                                              // runs every time the page loads
