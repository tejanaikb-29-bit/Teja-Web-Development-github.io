Store a JavaScript object in localStorage

An object is saved with JSON.stringify(), read back with JSON.parse(), and its properties are used again after a refresh.

Working demonstration






View source code of this program

<!-- HTML -->
<form id="f" class="row"><label class="field">Name<input id="name" type="text" value="Asha"></label><label class="field">Age<input id="age" type="number" value="20" style="width:90px"></label><label class="field">Course<input id="course" type="text" value="BCA"></label><button>Save object</button><button type="button" id="load" class="secondary">Load object</button></form><pre class="out" id="out"></pre>

// JavaScript
const $ = (id) => document.getElementById(id);
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  const student = { name: $('name').value, age: Number($('age').value), course: $('course').value, subjects: ['HTML', 'CSS', 'JS'] };
  localStorage.setItem('student', JSON.stringify(student));
  $('out').textContent = 'Stored string:\n' + localStorage.getItem('student');
});
function load() {
  const raw = localStorage.getItem('student');
  if (!raw) { $('out').textContent = 'No student object stored yet.'; return; }
  const s = JSON.parse(raw);                          // back to a real object
  $('out').textContent = `Restored object:\nname: ${s.name}\nage: ${s.age} (type ${typeof s.age})\ncourse: ${s.course}\nfirst subject: ${s.subjects[0]}`;
}
$('load').addEventListener('click', load);
load();
