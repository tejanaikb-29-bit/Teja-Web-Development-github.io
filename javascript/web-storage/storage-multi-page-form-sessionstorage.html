Multi-page form using sessionStorage (step 1 of 3)

Step 1 of 3: personal details are saved in sessionStorage, so the next pages can read them. Continue to step 2 and the summary page.

Working demonstration

Step 1 of 3 – Personal details

View source code of this program 
<!-- HTML -->
<p class="muted"><strong>Step 1 of 3</strong> &ndash; Personal details</p><div class="fg"><label>Full name</label><input id="name" type="text"></div><div class="fg"><label>Email</label><input id="email" type="email"></div><div class="fg"><label>Phone</label><input id="phone" type="tel"></div><p class="err" id="err"></p><button id="next">Next: address &rarr;</button>

// JavaScript
const $ = (id) => document.getElementById(id);
const KEY = 'multi-page-form';
const data = JSON.parse(sessionStorage.getItem(KEY) || '{}');
const fields = ['name', 'email', 'phone'];
fields.forEach((id) => { if (data[id]) $(id).value = data[id]; });
$('next').addEventListener('click', () => {
  let ok = true;
  fields.forEach((id) => {
    const v = $(id).value.trim();
    const bad = !v || (id === 'email' && !/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) || (id === 'phone' && !/^[6-9]\d{9}$/.test(v));
    $(id).classList.toggle('invalid', bad);
    if (bad) ok = false; else data[id] = v;
  });
  $('err').textContent = ok ? '' : 'Please complete every field correctly.';
  if (!ok) return;
  sessionStorage.setItem(KEY, JSON.stringify(data));
  location.href = 'storage-multi-page-form-step2.html';
});
