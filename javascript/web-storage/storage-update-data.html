Update data stored in localStorage

Calling setItem() with an existing key replaces its value. The old and new values are shown side by side.

View source code of this program 
<!-- HTML -->
<p>Stored city: <strong id="city"></strong></p><div class="row"><label class="field">New city<input id="in" type="text" placeholder="Type a new value"></label><button id="up">Update</button></div><div class="out" id="out"></div>

// JavaScript
const $ = (id) => document.getElementById(id);
if (localStorage.getItem('city') === null) localStorage.setItem('city', 'Chennai');
function show() { $('city').textContent = localStorage.getItem('city'); }
$('up').addEventListener('click', () => {
  const v = $('in').value.trim();
  if (!v) { $('out').textContent = 'Enter a new value.'; return; }
  const old = localStorage.getItem('city');
  localStorage.setItem('city', v);                   // same key = overwrite
  $('out').textContent = `Updated key "city":\nold value: ${old}\nnew value: ${v}`;
  $('in').value = '';
  show();
});
show();
