JavaScript › Web Storage

Save form data automatically using localStorage

A draft is saved 0.5 seconds after you stop typing. A status line shows when it was last saved.

Working demonstration

Start typing - your draft saves itself.

View source code of this program 
<!-- HTML -->
<form id="f" style="max-width:520px"><label class="field">Title<input id="title" type="text"></label><label class="field" style="margin-top:.6rem">Draft<textarea id="body" rows="6"></textarea></label><p class="muted" id="status">Start typing - your draft saves itself.</p><button type="button" id="clear" class="danger small">Discard draft</button></form>

// JavaScript
const $ = (id) => document.getElementById(id);
let timer = null;
const draft = JSON.parse(localStorage.getItem('autosave-draft') || 'null');
if (draft) { $('title').value = draft.title; $('body').value = draft.body; $('status').textContent = 'Draft restored (saved ' + draft.time + ').'; }
function save() {
  const time = new Date().toLocaleTimeString();
  localStorage.setItem('autosave-draft', JSON.stringify({ title: $('title').value, body: $('body').value, time }));
  $('status').textContent = 'Draft saved at ' + time;
}
['title', 'body'].forEach((id) => $(id).addEventListener('input', () => { $('status').textContent = 'Typing...'; clearTimeout(timer); timer = setTimeout(save, 500); }));   // debounce
$('clear').addEventListener('click', () => { localStorage.removeItem('autosave-draft'); $('title').value = ''; $('body').value = ''; $('status').textContent = 'Draft discarded.'; });
