Student marks database using localStorage

Store marks for each student and see the total, percentage and result calculated automatically. Search, edit and delete records.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="form" novalidate>
  <div class="row">
  <label class="field">Roll no<input id="f_roll" type="text" value=""></label>
  <label class="field">Name<input id="f_name" type="text" value=""></label>
  <label class="field">Maths<input id="f_m1" type="number" value="" step="any"></label>
  <label class="field">Science<input id="f_m2" type="number" value="" step="any"></label>
  <label class="field">English<input id="f_m3" type="number" value="" step="any"></label>
  </div>
  <div class="row"><button type="submit" id="save">Add student mark record</button><button type="button" id="cancel" class="secondary" style="display:none">Cancel edit</button>
  <input id="search" type="search" placeholder="Search..." style="margin-left:auto"></div>
  <p class="msg" id="msg"></p>
</form>
<div style="overflow-x:auto"><table class="data" id="table"></table></div>
<p class="muted" id="info"></p>

// JavaScript
const $ = (id) => document.getElementById(id);
const KEY = 'student-marks-db';
const fields = [["roll", "Roll no", "text"], ["name", "Name", "text"], ["m1", "Maths", "number"], ["m2", "Science", "number"], ["m3", "English", "number"]];                  // [name, label, type]
const computed = [
  { label: "Total", fn: (r) => Number(r.m1) + Number(r.m2) + Number(r.m3) },
  { label: "Percentage", fn: (r) => ((Number(r.m1) + Number(r.m2) + Number(r.m3)) / 3).toFixed(1) + "%" },
  { label: "Result", fn: (r) => Math.min(r.m1, r.m2, r.m3) >= 35 ? "Pass" : "Fail" }
];
function load() {
  try { const d = JSON.parse(localStorage.getItem(KEY)); if (d) return d; } catch (e) { /* ignore */ }
  return [{"id": 1, "roll": "101", "name": "Asha", "m1": 88, "m2": 92, "m3": 79}, {"id": 2, "roll": "102", "name": "Ravi", "m1": 45, "m2": 30, "m3": 62}];
}
function save() { localStorage.setItem(KEY, JSON.stringify(records)); }
let records = load();
let editId = null;

function render() {
  const q = $('search').value.toLowerCase();
  const table = $('table');
  table.innerHTML = '';
  const head = table.insertRow();
  fields.forEach(([, label]) => head.appendChild(Object.assign(document.createElement('th'), { textContent: label })));
  computed.forEach((c) => head.appendChild(Object.assign(document.createElement('th'), { textContent: c.label })));
  head.appendChild(Object.assign(document.createElement('th'), { textContent: 'Actions' }));
  const shown = records.filter((r) => fields.some(([n]) => String(r[n]).toLowerCase().includes(q)));
  shown.forEach((r) => {
    const row = table.insertRow();
    fields.forEach(([n]) => { row.insertCell().textContent = r[n]; });
    computed.forEach((c) => { row.insertCell().textContent = c.fn(r); });
    const cell = row.insertCell();
    const edit = document.createElement('button');
    edit.textContent = 'Edit'; edit.className = 'secondary small';
    edit.addEventListener('click', () => startEdit(r.id));
    const del = document.createElement('button');
    del.textContent = 'Delete'; del.className = 'danger small'; del.style.marginLeft = '.3rem';
    del.addEventListener('click', () => { if (confirm('Delete this student mark record?')) { if (r.id === editId) stopEdit(); records = records.filter((x) => x.id !== r.id); save(); render(); } });
    cell.append(edit, del);
  });
  if (!shown.length) { const cell = table.insertRow().insertCell(); cell.colSpan = fields.length + computed.length + 1; cell.textContent = 'No records to show.'; }
  $('info').textContent = `${records.length} student mark record(s) stored in localStorage under the key "${KEY}".`;
}
function startEdit(id) {
  const r = records.find((x) => x.id === id);
  fields.forEach(([n]) => { $('f_' + n).value = r[n]; });
  editId = id;
  $('save').textContent = 'Update student mark record';
  $('cancel').style.display = 'inline-block';
  $('msg').textContent = 'Editing: change the values and click Update.';
}
function stopEdit() {
  editId = null;
  $('form').reset();
  $('save').textContent = 'Add student mark record';
  $('cancel').style.display = 'none';
}
$('form').addEventListener('submit', (e) => {
  e.preventDefault();
  const data = {};
  for (const [n, label, type] of fields) {
    let v = $('f_' + n).value.trim();
    if (v === '') { $('msg').textContent = `${label} is required.`; return; }
    if (type === 'number') { v = Number(v); if (Number.isNaN(v)) { $('msg').textContent = `${label} must be a number.`; return; } }
    data[n] = v;
  }
  const existing = editId ? records.find((r) => r.id === editId) : null;
  if (existing) Object.assign(existing, data);          // update the record being edited
  else records.push({ id: Date.now(), ...data });       // otherwise create a new one
  save();
  $('msg').textContent = editId ? 'Record updated.' : 'Record added.';
  stopEdit();
  render();
});
$('cancel').addEventListener('click', () => { stopEdit(); $('msg').textContent = ''; });
$('search').addEventListener('input', render);
render();
