Delete an item from localStorage

localStorage.removeItem(key) deletes one entry and leaves the others untouched.

Working demonstration

    View source code of this program 
    <!-- HTML -->
<ul id="list"></ul><div class="row"><button id="add">Add sample items</button></div><div class="out" id="out"></div>

// JavaScript
const $ = (id) => document.getElementById(id);
const sample = { fruit: 'Mango', colour: 'Blue', city: 'Madurai' };
function show() {
  $('list').innerHTML = '';
  Object.keys(sample).forEach((k) => {
    const v = localStorage.getItem('demo-' + k);
    if (v === null) return;
    const li = document.createElement('li');
    li.textContent = `demo-${k} = ${v}  `;
    const b = document.createElement('button');
    b.textContent = 'removeItem'; b.className = 'danger small';
    b.addEventListener('click', () => { localStorage.removeItem('demo-' + k); $('out').textContent = `Removed "demo-${k}". The other items are still stored.`; show(); });
    li.appendChild(b);
    $('list').appendChild(li);
  });
  if (!$('list').children.length) $('list').innerHTML = '<li class="muted">No items stored. Click "Add sample items".</li>';
}
$('add').addEventListener('click', () => { Object.entries(sample).forEach(([k, v]) => localStorage.setItem('demo-' + k, v)); show(); });
show();
