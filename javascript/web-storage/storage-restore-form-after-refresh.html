Restore previously entered form data after refreshing the page

A form with text, select, radio and checkbox fields is stored on every change and refilled when the page loads again. Try the Reload button.

<!-- HTML -->
<form id="f" style="max-width:520px"><div class="row"><label class="field">Name<input name="name" type="text"></label><label class="field">Course<select name="course"><option value="">-- Select --</option><option>BCA</option><option>B.Sc CS</option><option>B.Com</option></select></label></div>
<div class="row"><span>Mode:</span><label><input type="radio" name="mode" value="Full time"> Full time</label><label><input type="radio" name="mode" value="Part time"> Part time</label></div><label><input type="checkbox" name="hostel"> I need hostel accommodation</label><div class="row"><button type="button" id="reload">Reload page</button><button type="button" id="clear" class="danger">Clear stored form</button></div></form><p class="muted" id="status"></p>

// JavaScript
const $ = (id) => document.getElementById(id);
const form = $('f');
function collect() {
  const d = {};
  new FormData(form).forEach((v, k) => { d[k] = v; });
  d.hostel = form.elements.hostel.checked;                 // unchecked boxes are missing from FormData
  return d;
}
function restore() {
  const d = JSON.parse(localStorage.getItem('form-state') || 'null');
  if (!d) return;
  Object.entries(d).forEach(([k, v]) => {
    const el = form.elements[k];
    if (!el) return;
    if (k === 'hostel') el.checked = v;
    else if (el instanceof RadioNodeList) el.value = v;
    else el.value = v;
  });
  $('status').textContent = 'Form restored from localStorage.';
}
form.addEventListener('input', () => { localStorage.setItem('form-state', JSON.stringify(collect())); $('status').textContent = 'Saved.'; });
form.addEventListener('change', () => { localStorage.setItem('form-state', JSON.stringify(collect())); });
$('reload').addEventListener('click', () => location.reload());
$('clear').addEventListener('click', () => { localStorage.removeItem('form-state'); form.reset(); $('status').textContent = 'Stored form cleared.'; });
restore();
