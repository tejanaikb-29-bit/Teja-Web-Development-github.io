Visitor counter using browser storage

Counts how many times this browser has opened the page (localStorage) and how many times in the current session (sessionStorage).

Working demonstration

Total visits (localStorage)
This tab session (sessionStorage)

Refresh the page to increase the counters.

View source code of this program 
<!-- HTML -->
<div class="counts"><div>Total visits (localStorage)<b id="total"></b></div><div>This tab session (sessionStorage)<b id="session"></b></div></div><p id="first" class="muted"></p><div class="row"><button id="reset" class="danger">Reset counters</button></div><p class="muted">Refresh the page to increase the counters.</p>

/* CSS */
.demo .counts { display: grid; grid-template-columns: repeat(auto-fit, minmax(200px, 1fr)); gap: 1rem; } .demo .counts div { text-align: center; padding: 1.2rem; background: #f1f4fa; border-radius: 12px; } .demo .counts b { display: block; font-size: 3rem; color: #1f419a; }

// JavaScript
const $ = (id) => document.getElementById(id);
let visits = Number(localStorage.getItem('visits') || 0) + 1;          // add this visit
localStorage.setItem('visits', visits);
let sess = Number(sessionStorage.getItem('sessionVisits') || 0) + 1;
sessionStorage.setItem('sessionVisits', sess);
if (!localStorage.getItem('firstVisit')) localStorage.setItem('firstVisit', new Date().toLocaleString());
$('total').textContent = visits;
$('session').textContent = sess;
$('first').textContent = 'First visit: ' + localStorage.getItem('firstVisit');
$('reset').addEventListener('click', () => { localStorage.removeItem('visits'); localStorage.removeItem('firstVisit'); sessionStorage.removeItem('sessionVisits'); location.reload(); });
