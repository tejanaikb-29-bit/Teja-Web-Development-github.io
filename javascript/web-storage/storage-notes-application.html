Notes application using localStorage

Write, edit, search and delete notes that persist between visits.

Working demonstration

    View source code of this program 
    <!-- HTML -->
<div class="notes">
  <div>
    <input id="title" type="text" placeholder="Note title" style="width:100%;margin-bottom:.5rem">
    <textarea id="text" rows="8" placeholder="Write your note here..." style="width:100%"></textarea>
    <div class="row"><button id="save">Save note</button><button id="new" class="secondary">New note</button><button id="del" class="danger">Delete</button></div>
    <p class="muted" id="status"></p>
  </div>
  <div><input id="search" type="search" placeholder="Search notes..." style="width:100%"><ul id="list" class="nl"></ul></div>
</div>

/* CSS */
.notes { display: grid; grid-template-columns: 1.4fr 1fr; gap: 1.2rem; }
.nl { list-style: none; padding: 0; margin: .6rem 0 0; }
.nl li { padding: .55rem .7rem; border: 1px solid #d5dbe6; border-radius: 8px; margin-bottom: .4rem; cursor: pointer; background: #fff; }
.nl li.active { border-color: #2b59c3; background: #eef3ff; } .nl li small { display: block; color: #6b7a94; }
@media (max-width: 680px) { .notes { grid-template-columns: 1fr; } }

// JavaScript
const $ = (id) => document.getElementById(id);
const KEY = 'storage-notes';
function load() { try { return JSON.parse(localStorage.getItem(KEY)) || []; } catch (e) { return []; } }
function save() { localStorage.setItem(KEY, JSON.stringify(notes)); }
let notes = load();
let currentId = null;

function render() {
  const q = $('search').value.toLowerCase();
  $('list').innerHTML = '';
  notes.filter((n) => (n.title + n.text).toLowerCase().includes(q)).sort((a, b) => b.updated - a.updated).forEach((n) => {
    const li = document.createElement('li');
    if (n.id === currentId) li.className = 'active';
    li.innerHTML = '<strong></strong><small></small>';
    li.querySelector('strong').textContent = n.title || 'Untitled';
    li.querySelector('small').textContent = new Date(n.updated).toLocaleString();
    li.addEventListener('click', () => open(n.id));
    $('list').appendChild(li);
  });
  if (!$('list').children.length) $('list').innerHTML = '<li class="muted">No notes found.</li>';
}
function open(id) {
  const n = notes.find((x) => x.id === id);
  currentId = id;
  $('title').value = n.title;
  $('text').value = n.text;
  $('status').textContent = 'Editing saved note.';
  render();
}
function clearEditor() { currentId = null; $('title').value = ''; $('text').value = ''; $('status').textContent = 'New note.'; render(); }

$('save').addEventListener('click', () => {
  const title = $('title').value.trim(), text = $('text').value.trim();
  if (!title && !text) { $('status').textContent = 'Write something first.'; return; }
  if (currentId) {
    const n = notes.find((x) => x.id === currentId);
    Object.assign(n, { title, text, updated: Date.now() });
  } else {
    currentId = Date.now();
    notes.push({ id: currentId, title, text, updated: Date.now() });
  }
  save();
  $('status').textContent = 'Saved to localStorage.';
  render();
});
$('new').addEventListener('click', clearEditor);
$('del').addEventListener('click', () => {
  if (!currentId) return;
  notes = notes.filter((n) => n.id !== currentId);
  save();
  clearEditor();
});
$('search').addEventListener('input', render);
render();
