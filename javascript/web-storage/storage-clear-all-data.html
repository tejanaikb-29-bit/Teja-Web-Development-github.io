Clear all localStorage data

localStorage.clear() removes every item stored by this website. This demo first lists everything, then clears it after confirmation.

Working demonstration

Warning: clear() deletes ALL keys for this site, including data saved by other pages of this project.





View source code of this program

<!-- HTML -->
<div class="row"><button id="fill">Add 3 sample items</button><button id="clear" class="danger">localStorage.clear()</button></div><pre class="out" id="out"></pre><p class="muted">Warning: clear() deletes ALL keys for this site, including data saved by other pages of this project.</p>

// JavaScript
const $ = (id) => document.getElementById(id);
function show() {
  const rows = [];
  for (let i = 0; i < localStorage.length; i++) { const k = localStorage.key(i); rows.push(`${k} = ${localStorage.getItem(k).slice(0, 40)}`); }
  $('out').textContent = `localStorage.length = ${localStorage.length}\n` + rows.join('\n');
}
$('fill').addEventListener('click', () => { localStorage.setItem('a', '1'); localStorage.setItem('b', '2'); localStorage.setItem('c', '3'); show(); });
$('clear').addEventListener('click', () => { if (confirm('Delete ALL localStorage data for this site?')) { localStorage.clear(); show(); } });
show();
