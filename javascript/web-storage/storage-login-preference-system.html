Login preference system using localStorage

The "Remember me" option stores the username, language and theme so the next visit starts pre-filled. Logging out can forget them.

<!-- HTML -->
<div id="loginView" class="lp"><h3>Sign in</h3><label class="field">Username<input id="user" type="text"></label><label class="field">Password<input id="pass" type="password" placeholder="(demo - any value)"></label>
<label class="field">Preferred language<select id="lang"><option>English</option><option>Tamil</option><option>Hindi</option></select></label><label><input type="checkbox" id="remember"> Remember me on this device</label><p id="err" class="err"></p><button id="login">Log in</button></div>
<div id="welcome" hidden><h3 id="hi"></h3><p id="prefs"></p><button id="logout" class="secondary">Log out</button> <button id="forget" class="danger">Log out and forget me</button></div>

/* CSS */
.demo .lp { max-width: 380px; } .demo .lp .field { margin-bottom: .7rem; }

// JavaScript
const $ = (id) => document.getElementById(id);
const prefs = JSON.parse(localStorage.getItem('login-prefs') || 'null');
if (prefs) { $('user').value = prefs.user; $('lang').value = prefs.lang; $('remember').checked = true; }
$('login').addEventListener('click', () => {
  const user = $('user').value.trim();
  if (!user || !$('pass').value) { $('err').textContent = 'Enter a username and a password.'; return; }
  $('err').textContent = '';
  if ($('remember').checked) localStorage.setItem('login-prefs', JSON.stringify({ user, lang: $('lang').value }));
  else localStorage.removeItem('login-prefs');
  $('hi').textContent = `Hello, ${user}!`;
  $('prefs').textContent = `Language: ${$('lang').value}. ${$('remember').checked ? 'Your username and language will be remembered.' : 'Nothing will be remembered.'}`;
  $('loginView').hidden = true; $('welcome').hidden = false;
});
$('logout').addEventListener('click', () => { $('welcome').hidden = true; $('loginView').hidden = false; $('pass').value = ''; });
$('forget').addEventListener('click', () => { localStorage.removeItem('login-prefs'); $('user').value = ''; $('remember').checked = false; $('lang').value = 'English'; $('logout').click(); });
