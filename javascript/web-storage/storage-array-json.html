Store an array in localStorage using JSON

localStorage stores only strings, so arrays are converted with JSON.stringify() and restored with JSON.parse().

    <!-- HTML -->
<div class="row"><input id="item" type="text" placeholder="Add a colour..."><button id="add">Add to array</button><button id="clear" class="danger">Clear</button></div><p>Array in memory: <code id="arr"></code></p><p>String stored in localStorage: <code id="str"></code></p><ol id="list"></ol>

// JavaScript
const $ = (id) => document.getElementById(id);
let colours = JSON.parse(localStorage.getItem('colours') || '["red", "green"]');    // string -> array
function render() {
  localStorage.setItem('colours', JSON.stringify(colours));                        // array -> string
  $('arr').textContent = `[${colours.map((c) => '"' + c + '"').join(', ')}]  (${typeof colours}, isArray: ${Array.isArray(colours)})`;
  $('str').textContent = localStorage.getItem('colours');
  $('list').innerHTML = '';
  colours.forEach((c, i) => {
    const li = document.createElement('li');
    li.textContent = c + ' ';
    const b = document.createElement('button'); b.textContent = 'x'; b.className = 'danger small';
    b.addEventListener('click', () => { colours.splice(i, 1); render(); });
    li.appendChild(b); $('list').appendChild(li);
  });
}
$('add').addEventListener('click', () => { const v = $('item').value.trim(); if (v) { colours.push(v); $('item').value = ''; render(); } });
$('clear').addEventListener('click', () => { colours = []; render(); });
render();
