Store temporary form data using sessionStorage

Everything you type is saved in sessionStorage. Refresh the page and it returns; close the tab and it is gone.

Working demonstration

View source code of this program 
<!-- HTML -->
<form id="f" style="max-width:480px"><div class="row"><label class="field">Name<input id="name" type="text"></label><label class="field">Email<input id="email" type="email"></label></div><label class="field">Message<textarea id="msg" rows="4"></textarea></label><div class="row"><button type="button" id="reload" class="secondary">Reload page</button><button type="button" id="clear" class="danger">Clear saved data</button></div></form><p class="muted" id="status"></p>

// JavaScript
const $ = (id) => document.getElementById(id);
const ids = ['name', 'email', 'msg'];
const saved = JSON.parse(sessionStorage.getItem('temp-form') || '{}');
ids.forEach((id) => { if (saved[id]) $(id).value = saved[id]; });
if (Object.keys(saved).length) $('status').textContent = 'Restored your text from sessionStorage.';
ids.forEach((id) => $(id).addEventListener('input', () => {
  const data = {};
  ids.forEach((k) => { data[k] = $(k).value; });
  sessionStorage.setItem('temp-form', JSON.stringify(data));
  $('status').textContent = 'Saved temporarily (' + new Date().toLocaleTimeString() + ').';
}));
$('reload').addEventListener('click', () => location.reload());
$('clear').addEventListener('click', () => { sessionStorage.removeItem('temp-form'); ids.forEach((id) => { $(id).value = ''; }); $('status').textContent = 'Cleared.'; });
