Recently viewed items feature using localStorage

Click "View" on a product and it is added to the top of the recently viewed list (maximum 5, no duplicates), stored in localStorage.

    <!-- HTML -->
<div class="split"><div><h4>Products</h4><div id="products"></div></div><div><h4>Recently viewed</h4><ol id="recent"></ol><button id="clear" class="danger small">Clear history</button></div></div>

/* CSS */
.demo .split { display: grid; grid-template-columns: 1.2fr 1fr; gap: 1.2rem; } .demo .prod { display: flex; justify-content: space-between; align-items: center; padding: .5rem .7rem; border: 1px solid #d5dbe6; border-radius: 8px; margin-bottom: .4rem; } @media (max-width: 640px) { .demo .split { grid-template-columns: 1fr; } }

// JavaScript
const $ = (id) => document.getElementById(id);
const products = ['Laptop', 'Smartphone', 'Headphones', 'Smart watch', 'Keyboard', 'Monitor', 'Backpack', 'Camera'];
let recent = JSON.parse(localStorage.getItem('recent-items') || '[]');
function render() {
  $('recent').innerHTML = recent.length ? '' : '<li class="muted">Nothing viewed yet.</li>';
  recent.forEach((r) => { const li = document.createElement('li'); li.textContent = r; $('recent').appendChild(li); });
}
products.forEach((p) => {
  const d = document.createElement('div');
  d.className = 'prod';
  d.innerHTML = '<span></span>'; d.firstChild.textContent = p;
  const b = document.createElement('button'); b.textContent = 'View'; b.className = 'small';
  b.addEventListener('click', () => {
    recent = [p, ...recent.filter((x) => x !== p)].slice(0, 5);      // newest first, no duplicates, max 5
    localStorage.setItem('recent-items', JSON.stringify(recent));
    render();
  });
  d.appendChild(b); $('products').appendChild(d);
});
$('clear').addEventListener('click', () => { recent = []; localStorage.removeItem('recent-items'); render(); });
render();
