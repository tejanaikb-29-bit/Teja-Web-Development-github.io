<!-- HTML -->
<p class="muted"><strong>Step 2 of 3</strong> &ndash; Address and course</p><div class="fg"><label>City</label><input id="city" type="text"></div><div class="fg"><label>Course</label><input id="course" type="text"></div><p class="err" id="err"></p><a class="btn secondary" href="storage-multi-page-form-sessionstorage.html">&larr; Back</a> <button id="next">Next: summary &rarr;</button>

// JavaScript
const $ = (id) => document.getElementById(id);
const KEY = 'multi-page-form';
const data = JSON.parse(sessionStorage.getItem(KEY) || '{}');
const fields = ['city', 'course'];
fields.forEach((id) => { if (data[id]) $(id).value = data[id]; });
$('next').addEventListener('click', () => {
  let ok = true;
  fields.forEach((id) => {
    const v = $(id).value.trim();
    const bad = !v || (id === 'email' && !/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(v)) || (id === 'phone' && !/^[6-9]\d{9}$/.test(v));
    $(id).classList.toggle('invalid', bad);
    if (bad) ok = false; else data[id] = v;
  });
  $('err').textContent = ok ? '' : 'Please complete every field correctly.';
  if (!ok) return;
  sessionStorage.setItem(KEY, JSON.stringify(data));
  location.href = 'storage-multi-page-form-summary.html';
});
