Store multiple values using localStorage

Each form field is stored under its own key (name, email, city, age) and all values are read back.

<!-- HTML -->
<form id="f" class="row"><label class="field">Name<input id="name" type="text"></label><label class="field">Email<input id="email" type="email"></label><label class="field">City<input id="city" type="text"></label><label class="field">Age<input id="age" type="number" style="width:90px"></label><button>Save all</button></form><table class="data"><thead><tr><th>Key</th><th>Stored value</th></tr></thead><tbody id="tb"></tbody></table>

// JavaScript
const $ = (id) => document.getElementById(id);
const keys = ['name', 'email', 'city', 'age'];
function show() {
  $('tb').innerHTML = '';
  keys.forEach((k) => { const tr = $('tb').insertRow(); tr.insertCell().textContent = k; tr.insertCell().textContent = localStorage.getItem('profile-' + k) ?? '(not set)'; $(k).value = localStorage.getItem('profile-' + k) || ''; });
}
$('f').addEventListener('submit', (e) => {
  e.preventDefault();
  keys.forEach((k) => localStorage.setItem('profile-' + k, $(k).value.trim()));
  show();
});
show();
