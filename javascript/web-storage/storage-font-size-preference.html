Save the user's preferred font size using localStorage

A slider and A+ / A- buttons change the reading text size. The size is stored and restored on the next visit.

Working demonstration

16 px

Reading sample

The quick brown fox jumps over the lazy dog. Choose a size that is comfortable for your eyes, and this page will remember it.

View source code of this program 
<!-- HTML -->
<div class="row"><button id="minus" class="secondary">A-</button><input id="range" type="range" min="12" max="32" value="16"><button id="plus" class="secondary">A+</button><span><strong id="px">16</strong> px</span><button id="reset" class="danger small">Reset</button></div>
<div id="text" class="card"><h3>Reading sample</h3><p>The quick brown fox jumps over the lazy dog. Choose a size that is comfortable for your eyes, and this page will remember it.</p></div>

// JavaScript
const $ = (id) => document.getElementById(id);
function setSize(px) {
  px = Math.min(32, Math.max(12, px));
  $('text').style.fontSize = px + 'px';
  $('range').value = px; $('px').textContent = px;
  localStorage.setItem('fontSize', px);
}
setSize(parseInt(localStorage.getItem('fontSize')) || 16);
$('range').addEventListener('input', () => setSize(Number($('range').value)));
$('plus').addEventListener('click', () => setSize(Number($('range').value) + 2));
$('minus').addEventListener('click', () => setSize(Number($('range').value) - 2));
$('reset').addEventListener('click', () => setSize(16));
