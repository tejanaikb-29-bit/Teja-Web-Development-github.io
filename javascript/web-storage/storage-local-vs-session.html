Compare localStorage and sessionStorage

Save a value in each. Then reload the page (both survive), or open this page in a new tab (only localStorage is shared) and close the tab (sessionStorage is lost).

// JavaScript
const $ = (id) => document.getElementById(id);
function show() { $('lout').textContent = localStorage.getItem('cmp-value') ?? '(empty)'; $('sout').textContent = sessionStorage.getItem('cmp-value') ?? '(empty)'; }
$('ls').addEventListener('click', () => { localStorage.setItem('cmp-value', $('lv').value); show(); });
$('ss').addEventListener('click', () => { sessionStorage.setItem('cmp-value', $('sv').value); show(); });
$('open').addEventListener('click', () => window.open(location.href, '_blank'));
show();
