<!-- HTML -->
<div class="row">
  <label class="field">Enter some text
    <input id="text" type="text" value="  JavaScript is Fun  ">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  const s = txt('text');
  const t = s.trim();
  show(`Original           : "${s}"
  length             : ${s.length}
  trim()             : "${t}"
  toUpperCase()      : ${t.toUpperCase()}
  toLowerCase()      : ${t.toLowerCase()}
  charAt(0)          : ${t.charAt(0)}
  indexOf('a')       : ${t.indexOf('a')}
  includes('Fun')    : ${t.includes('Fun')}
  slice(0, 4)        : ${t.slice(0, 4)}
  replace('Fun','Great') : ${t.replace('Fun', 'Great')}
  split(' ')         : ${JSON.stringify(t.split(' '))}
  padStart(24, '*')  : ${t.padStart(24, '*')}
  repeat(2)          : ${t.repeat(2)}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
