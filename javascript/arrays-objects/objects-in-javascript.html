Objects in JavaScript

Creates an object with properties, a nested object and a method; then reads, adds and deletes properties.

<!-- HTML -->
<button class="btn" id="run">Run object demo</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };
const student = {
  name: 'Asha',
  age: 20,
  marks: { maths: 88, science: 92 },
  introduce() { return `Hi, I am ${this.name}`; },    // method
};

function run() {
  const lines = [];
  lines.push('Object: ' + JSON.stringify(student));
  lines.push(`student.name = ${student.name}`);
  lines.push(`student['age'] = ${student['age']}`);
  lines.push(`student.marks.maths = ${student.marks.maths}`);
  lines.push(`student.introduce() = ${student.introduce()}`);
  student.city = 'Chennai';                   // add a new property
  delete student.age;                         // remove a property
  lines.push('After add + delete: ' + JSON.stringify(student));
  lines.push('Object.keys   : ' + Object.keys(student).join(', '));
  lines.push('Object.values : ' + JSON.stringify(Object.values(student)));
  for (const [key, value] of Object.entries(student)) {
    if (typeof value !== 'function') lines.push(`  ${key} -> ${JSON.stringify(value)}`);
  }
  show(lines.join('\n'));
}
$('run').addEventListener('click', run);
