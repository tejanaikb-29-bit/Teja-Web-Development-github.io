Arrays and array methods

push, pop, shift, unshift, slice, splice, map, filter, reduce, find, includes, join and reverse on a working copy.

<!-- HTML -->
<div class="row">
  <label class="field">Array elements (comma separated)
    <input id="arr" type="text" value="5, 8, 12, 3, 20" placeholder="e.g. 5, 12, 8, 3">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (list('arr').length === 0 || list('arr').some(Number.isNaN)) { show('Please enter valid input.'); return; }
  const arr = list('arr');
  const a = [...arr];
  const lines = [`Start array        : [${a}]`];
  a.push(99);      lines.push(`push(99)           : [${a}]`);
  a.pop();         lines.push(`pop()              : [${a}]`);
  a.unshift(0);    lines.push(`unshift(0)         : [${a}]`);
  a.shift();       lines.push(`shift()            : [${a}]`);
  lines.push(`slice(1, 3)        : [${a.slice(1, 3)}]`);
  lines.push(`map(x => x * 2)    : [${a.map((x) => x * 2)}]`);
  lines.push(`filter(x => x > 5) : [${a.filter((x) => x > 5)}]`);
  lines.push(`reduce (sum)       : ${a.reduce((s, x) => s + x, 0)}`);
  lines.push(`find(x => x > 10)  : ${a.find((x) => x > 10)}`);
  lines.push(`includes(8)        : ${a.includes(8)}`);
  lines.push(`indexOf(12)        : ${a.indexOf(12)}`);
  lines.push(`reverse()          : [${[...a].reverse()}]`);
  lines.push(`join(' - ')        : ${a.join(' - ')}`);
  show(lines.join('\n'));
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
