Find the second-largest element in an array

Finds the largest and second largest values in a single pass without sorting.

<!-- HTML -->
<div class="row">
  <label class="field">Array elements (comma separated)
    <input id="arr" type="text" value="12, 35, 1, 10, 34, 1" placeholder="e.g. 5, 12, 8, 3">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (list('arr').length === 0 || list('arr').some(Number.isNaN)) { show('Please enter valid input.'); return; }
  const arr = list('arr');
  let largest = -Infinity, second = -Infinity;
  for (const x of arr) {
    if (x > largest) { second = largest; largest = x; }
    else if (x > second && x !== largest) second = x;
  }
  show(second === -Infinity ? 'There is no second-largest element (all values are equal).' : `Array: [${arr}]\nLargest        = ${largest}\nSecond largest = ${second}`);
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
