// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };
class Student {
  static count = 0;
  constructor(name, age, marks) {
    this.name = name;
    this.age = age;
    this.marks = marks;
    Student.count++;
  }
  get average() { return this.marks.reduce((a, b) => a + b, 0) / this.marks.length; }   // getter
  grade() { return this.average >= 75 ? 'A' : this.average >= 60 ? 'B' : 'C'; }
  introduce() { return `Hi, I am ${this.name}, ${this.age} years old.`; }
  static compare(a, b) { return a.average > b.average ? `${a.name} scored higher` : `${b.name} scored higher`; }
}

function run() {
  const s1 = new Student('Asha', 20, [88, 92, 79]);
  const s2 = new Student('Ravi', 21, [65, 70, 58]);
  show(`${s1.introduce()}\nAverage: ${s1.average.toFixed(1)}  Grade: ${s1.grade()}\n
  ${s2.introduce()}\nAverage: ${s2.average.toFixed(1)}  Grade: ${s2.grade()}\n
  Static method: ${Student.compare(s1, s2)}\nStudents created: ${Student.count}`);
}
$('run').addEventListener('click', run);
