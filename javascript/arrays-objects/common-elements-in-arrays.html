Find common elements in two arrays

Uses filter() with includes() to keep values found in both arrays.

<!-- HTML -->
<div class="row">
  <label class="field">First array
    <input id="a" type="text" value="1, 2, 3, 4, 5, 6" placeholder="e.g. 5, 12, 8, 3">
  </label>
  <label class="field">Second array
    <input id="b" type="text" value="4, 5, 6, 7, 8" placeholder="e.g. 5, 12, 8, 3">
  </label>
</div>
<button class="btn" id="run">Run</button>
<div class="out" id="out">The result will appear here.</div>

// JavaScript
const $ = (id) => document.getElementById(id);
const num = (id) => parseFloat($(id).value);
const txt = (id) => $(id).value;
const list = (id) => $(id).value.split(/[\s,]+/).filter(Boolean).map(Number);
const show = (msg) => { $('out').textContent = msg; };
const showHTML = (html) => { $('out').innerHTML = html; };

function run() {
  if (list('a').length === 0 || list('a').some(Number.isNaN) || list('b').length === 0 || list('b').some(Number.isNaN)) { show('Please enter valid input.'); return; }
  const a = list('a'), b = list('b');
  const common = [...new Set(a.filter((x) => b.includes(x)))];
  show(common.length ? `Common elements: [${common}]` : 'The arrays have no common elements.');
}
$('run').addEventListener('click', run);
document.querySelectorAll('input').forEach((el) => el.addEventListener('keydown', (e) => { if (e.key === 'Enter') run(); }));
