<!-- HTML -->
<div class="mf"><span class="label"></span><nav><a href="#">Home</a><a href="#">Courses</a><a href="#">About</a><a href="#">Contact</a></nav>
<div class="hero"><h3>Mobile-first design</h3><p style="margin:.3rem 0 0">Start small, then add layout as the screen grows.</p></div>
<div class="cards"><div>Card 1</div><div>Card 2</div><div>Card 3</div></div></div>

/* CSS */
/* 1. Mobile first: single column, stacked navigation */
.demo .mf nav { display: flex; flex-direction: column; background: #14213d; border-radius: 8px; }
.demo .mf nav a { color: #fff; padding: .7rem 1rem; text-decoration: none; border-bottom: 1px solid rgba(255, 255, 255, .15); }
.demo .mf .hero { background: #f2c200; padding: 1.2rem; margin: .8rem 0; border-radius: 8px; }
.demo .mf .hero h3 { margin: 0; font-size: 1.2rem; }
.demo .mf .cards { display: grid; gap: .8rem; }
.demo .mf .cards div { background: #eaf2ff; padding: 1rem; border-radius: 8px; }
.demo .mf .label::after { content: "Base styles (mobile)"; }
.demo .mf .label { display: inline-block; background: #d9481f; color: #fff; padding: .25rem .8rem; border-radius: 999px; font-weight: 600; }

/* 2. Tablet and up */
@media (min-width: 600px) {
  .demo .mf nav { flex-direction: row; } .demo .mf nav a { border-bottom: 0; }
  .demo .mf .cards { grid-template-columns: repeat(2, 1fr); } .demo .mf .label::after { content: "min-width: 600px (tablet)"; }
}
/* 3. Desktop */
@media (min-width: 900px) {
  .demo .mf .cards { grid-template-columns: repeat(3, 1fr); } .demo .mf .hero h3 { font-size: 1.9rem; } .demo .mf .label::after { content: "min-width: 900px (desktop)"; }
}
