/* CSS */
.demo .an { display: flex; flex-wrap: wrap; gap: 2rem; padding: 1.5rem 1rem; align-items: center; }
.demo .ball { width: 60px; height: 60px; border-radius: 50%; background: #d9481f; animation: bounce 1s ease-in-out infinite alternate; }
.demo .pulse { width: 60px; height: 60px; border-radius: 50%; background: #1769b5; animation: pulse 1.4s infinite; }
.demo .spin { width: 60px; height: 60px; border-radius: 10px; background: #178a4a; animation: spin 2s linear infinite; }
.demo .fade { padding: 1rem; background: #f2c200; border-radius: 8px; animation: fade 2s ease-in-out infinite alternate; font-weight: 700; }
.demo .slide { padding: .8rem 1rem; background: #14213d; color: #fff; border-radius: 8px; animation: slide 2.5s ease-in-out infinite alternate; }
@keyframes bounce { from { transform: translateY(0); } to { transform: translateY(-70px); } }
@keyframes pulse { 0% { box-shadow: 0 0 0 0 rgba(23, 105, 181, .6); } 100% { box-shadow: 0 0 0 30px rgba(23, 105, 181, 0); } }
@keyframes spin { to { transform: rotate(360deg); } }
@keyframes fade { from { opacity: .15; } to { opacity: 1; } }
@keyframes slide { from { transform: translateX(-40px); } to { transform: translateX(40px); } }
