CSS Grid template areas

The layout is drawn as text with grid-template-areas. A media query redraws it as a single column on small screens.

Working demonstration

grid-template-areas:
        "head head head"
        "nav  main ads"
        "foot foot foot";
header
nav
main
ads
footer

<!-- HTML -->
<pre style="background:#f1f4fa;padding:.6rem;border-radius:8px;overflow:auto">grid-template-areas:
  "head head head"
  "nav  main ads"
  "foot foot foot";</pre>
<div class="ga"><div class="h">header</div><div class="n">nav</div><div class="m">main</div><div class="a">ads</div><div class="f">footer</div></div>

/* CSS */
.demo .ga { display: grid; gap: .6rem; grid-template-columns: 140px 1fr 140px; grid-template-areas: "head head head" "nav main ads" "foot foot foot"; }
.demo .ga > * { padding: 1rem; border-radius: 8px; color: #fff; font-weight: 700; }
.demo .ga .h { grid-area: head; background: #14213d; } .demo .ga .n { grid-area: nav; background: #1769b5; }
.demo .ga .m { grid-area: main; background: #178a4a; min-height: 110px; } .demo .ga .a { grid-area: ads; background: #d9481f; }
.demo .ga .f { grid-area: foot; background: #7b2cbf; }
@media (max-width: 600px) { .demo .ga { grid-template-columns: 1fr; grid-template-areas: "head" "main" "nav" "ads" "foot"; } }
