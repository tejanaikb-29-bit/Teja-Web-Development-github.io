<!-- HTML -->
<p><code>width: calc(100% - 160px)</code> for the fluid column:</p>
<div class="cc"><div class="fixed">160px sidebar</div><div class="fluid">Fluid column fills the remaining width</div></div>
<div class="full">calc(100% - 2 * 1.5rem) with auto margins</div>
<p><code>gap: calc(var(--gap) * 2)</code></p><div class="vars"><div>One</div><div>Two</div><div>Three</div></div>
<p class="big">font-size: calc(1rem + 2vw)</p>
<div class="center"><i></i></div>

/* CSS */
.demo .cc { display: flex; gap: 0; margin-bottom: 1rem; }
.demo .cc .fixed { width: 160px; background: #14213d; color: #fff; padding: 1rem; }
.demo .cc .fluid { width: calc(100% - 160px); background: #eaf2ff; padding: 1rem; }      /* fills the rest */
.demo .full { width: calc(100% - 2 * 1.5rem); margin: 0 auto; background: #f2c200; padding: 1rem; font-weight: 700; }
.demo .vars { --gap: 12px; display: flex; gap: calc(var(--gap) * 2); } .demo .vars div { background: #1769b5; color: #fff; padding: 1rem; border-radius: 8px; }
.demo .big { font-size: calc(1rem + 2vw); font-weight: 700; color: #d9481f; }
.demo .center { position: relative; height: 120px; background: #eef2fb; border-radius: 8px; }
.demo .center i { position: absolute; width: 60px; height: 60px; background: #178a4a; left: calc(50% - 30px); top: calc(50% - 30px); border-radius: 8px; }
