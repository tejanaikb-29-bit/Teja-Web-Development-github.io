Responsive layout using CSS Grid and media queries

A card grid changes from 4 columns to 2 to 1 as the window narrows, and the featured card stops spanning two columns.

<!-- HTML -->
<span class="now"></span>
<div class="rg"><div class="wide">Featured (spans 2)</div><div>Card 2</div><div>Card 3</div><div>Card 4</div><div>Card 5</div><div>Card 6</div><div>Card 7</div></div>

/* CSS */
.demo .rg { display: grid; gap: .8rem; grid-template-columns: repeat(4, 1fr); }
.demo .rg > div { background: #1769b5; color: #fff; padding: 1.4rem 1rem; border-radius: 8px; text-align: center; font-weight: 700; }
.demo .rg .wide { grid-column: span 2; background: #d9481f; }
.demo .now { display: inline-block; padding: .3rem .9rem; border-radius: 999px; background: #f2c200; font-weight: 700; margin-bottom: .8rem; }
.demo .now::after { content: "4 columns (900px and above)"; }
@media (max-width: 899px) { .demo .rg { grid-template-columns: repeat(2, 1fr); } .demo .now::after { content: "2 columns (600px - 899px)"; } }
@media (max-width: 599px) { .demo .rg { grid-template-columns: 1fr; } .demo .rg .wide { grid-column: auto; } .demo .now::after { content: "1 column (below 600px)"; } }
