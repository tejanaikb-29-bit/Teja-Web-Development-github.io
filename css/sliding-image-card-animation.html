/* CSS */
.demo .marq { overflow: hidden; border-radius: 10px; background: #14213d; padding: .6rem 0; }
.demo .strip { display: flex; gap: .6rem; width: max-content; animation: slide 30s linear infinite; }
.demo .marq:hover .strip { animation-play-state: paused; }
.demo .strip img { width: 180px; height: 110px; object-fit: cover; border-radius: 8px; }
@keyframes slide { to { transform: translateX(calc(-50% - .3rem)); } }
.demo .cs { display: flex; flex-wrap: wrap; gap: 1rem; margin-top: 1.2rem; }
.demo .cs div { flex: 1 1 150px; padding: 1.1rem; border-radius: 10px; color: #fff; font-weight: 700; animation: cardin 5s ease-in-out infinite both; }
.demo .cs div:nth-child(1) { background: #d9481f; } .demo .cs div:nth-child(2) { background: #1769b5; animation-delay: .35s; } .demo .cs div:nth-child(3) { background: #178a4a; animation-delay: .7s; }
@keyframes cardin { 0% { opacity: 0; transform: translateX(-70px); } 15%, 80% { opacity: 1; transform: translateX(0); } 100% { opacity: 0; transform: translateX(70px); } }
