/* CSS */
.demo .f1 { font-family: Georgia, serif; font-size: 1.4rem; }
.demo .f2 { font-family: "Courier New", monospace; font-weight: 700; }
.demo .f3 { font-style: italic; font-variant: small-caps; font-size: 1.2rem; }
.demo .t1 { text-align: center; text-decoration: underline wavy #d9481f; }
.demo .t2 { text-transform: uppercase; letter-spacing: .25em; }
.demo .t3 { text-indent: 2rem; line-height: 1.9; word-spacing: .3em; text-align: justify; }
.demo .t4 { text-decoration: line-through; color: #888; }
.demo .t5 { font: italic 700 1.3rem/1.4 Verdana, sans-serif; color: #1769b5; }
