<div class="legend"><span style="background:#fbd9a6">margin</span><span style="background:#f0a35e">border</span><span style="background:#b9e2b1">padding</span><span style="background:#8fb8e8">content</span></div>

/* CSS */
.demo .bm-outer { background: #fbd9a6; padding: 0; display: inline-block; }
.demo .bm-margin { margin: 30px; }
.demo .bm-box { background: #b9e2b1; border: 14px solid #f0a35e; padding: 24px; }
.demo .bm-content { background: #8fb8e8; width: 160px; height: 60px; display: flex; align-items: center; justify-content: center; font-weight: 700; }
.demo .legend span { display: inline-block; padding: .2rem .6rem; margin: 2px; border-radius: 4px; font-size: .85rem; }
.demo .cmp div { background: #d9481f; color: #fff; width: 200px; padding: 20px; border: 10px solid #14213d; margin-bottom: 10px; }
.demo .cb { box-sizing: content-box; } .demo .bb { box-sizing: border-box; }
