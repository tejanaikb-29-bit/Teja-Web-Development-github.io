/* CSS */
.demo p { color: #333; }                              /* element selector */
.demo .intro { font-size: 1.2rem; font-weight: 700; }   /* class selector */
.demo #unique { background: #fff2c2; padding: .5rem; }  /* id selector */
.demo h4, .demo h5 { color: #1769b5; }                  /* grouping selector */
.demo .box p { border-left: 4px solid #1769b5; padding-left: .6rem; }  /* descendant selector */
.demo a[target="_blank"] { color: #c62828; font-weight: 700; }         /* attribute selector */
.demo input[type="text"] { border: 2px solid #178a4a; }
