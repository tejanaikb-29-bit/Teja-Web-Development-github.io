The sidebar sits beside this text on large screens and moves above it on smaller screens.

View source code of this program 
<!-- HTML -->
<span class="info"></span>
<div class="page"><aside class="side"><h4 style="margin-top:0">Menu</h4><p>Home<br>Courses<br>Contact</p></aside>
<section class="main"><h3>Responsive layout</h3><p>The sidebar sits beside this text on large screens and moves above it on smaller screens.</p><div class="promo">Promo panel (hidden below 600px)</div></section></div>

/* CSS */
.demo .page { display: grid; grid-template-columns: 220px 1fr; gap: 1rem; }
.demo .side { background: #14213d; color: #fff; padding: 1rem; border-radius: 8px; }
.demo .main { background: #eef2fb; padding: 1rem; border-radius: 8px; }
.demo .main h3 { font-size: 1.5rem; margin-top: 0; }
.demo .promo { background: #f2c200; padding: .8rem; border-radius: 8px; margin-top: 1rem; }
.demo .info::after { content: "Screen: wider than 900px (desktop)"; }
.demo .info { display: inline-block; padding: .3rem .8rem; background: #d9481f; color: #fff; border-radius: 999px; margin-bottom: .8rem; font-weight: 600; }

@media (max-width: 900px) {
  .demo .page { grid-template-columns: 1fr; }          /* stack sidebar above content */
  .demo .info::after { content: "Screen: 600px - 900px (tablet)"; }
}
@media (max-width: 600px) {
  .demo .promo { display: none; }                     /* hide extra panel on phones */
  .demo .main h3 { font-size: 1.15rem; }
  .demo .info::after { content: "Screen: below 600px (phone)"; }
}
