/* CSS */
.demo .ka { display: flex; flex-wrap: wrap; gap: 2rem; align-items: center; margin-bottom: 1.4rem; }
.demo .morph { width: 70px; height: 70px; background: #d9481f; animation: morph 4s ease-in-out infinite; }
@keyframes morph { 0% { border-radius: 0; background: #d9481f; transform: rotate(0); } 33% { border-radius: 50%; background: #1769b5; transform: rotate(120deg); } 66% { border-radius: 50% 0; background: #178a4a; transform: rotate(240deg) scale(1.2); } 100% { border-radius: 0; background: #d9481f; transform: rotate(360deg); } }
.demo .type { font: 700 1.3rem "Courier New", monospace; width: 25ch; white-space: nowrap; overflow: hidden; border-right: 3px solid #14213d; animation: typing 3s steps(25) infinite alternate, caret .6s step-end infinite; }
@keyframes typing { from { width: 0; } to { width: 25ch; } } @keyframes caret { 50% { border-color: transparent; } }
.demo .bar { width: 100%; max-width: 320px; height: 16px; background: #e2e8f4; border-radius: 8px; overflow: hidden; }
.demo .bar i { display: block; height: 100%; background: linear-gradient(90deg, #178a4a, #5eead4); animation: fill 3s ease-in-out infinite; }
@keyframes fill { 0% { width: 0; } 70%, 100% { width: 100%; } }
.demo .dots { display: flex; gap: .5rem; } .demo .dots i { width: 18px; height: 18px; border-radius: 50%; background: #7b2cbf; animation: jump .9s ease-in-out infinite; }
.demo .dots i:nth-child(2) { animation-delay: .15s; } .demo .dots i:nth-child(3) { animation-delay: .3s; } .demo .dots i:nth-child(4) { animation-delay: .45s; }
@keyframes jump { 0%, 100% { transform: translateY(0); } 50% { transform: translateY(-18px); } }
