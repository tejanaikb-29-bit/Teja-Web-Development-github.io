/* CSS */
.demo .gal { display: grid; grid-template-columns: repeat(auto-fill, minmax(190px, 1fr)); gap: .8rem; }
.demo .gal figure { margin: 0; position: relative; overflow: hidden; border-radius: 10px; aspect-ratio: 4 / 3; }
.demo .gal img { width: 100%; height: 100%; object-fit: cover; display: block; transition: transform .4s; }
.demo .gal figure:hover img { transform: scale(1.12); }
.demo .gal figcaption { position: absolute; left: 0; right: 0; bottom: 0; padding: .4rem .7rem; color: #fff; background: linear-gradient(transparent, rgba(0, 0, 0, .7)); font-weight: 600; }
