<!-- HTML -->
<h3 class="title">Class selector styled heading</h3>
<p id="special">This paragraph is styled with an <strong>id selector</strong>: colours, padding, margin and a thick left border.</p>
<div class="note">Named colour, hex, rgb() and hsl() values are all used on this page.</div>
<p class="serif">Serif font, 18px size and generous line height improve readability.</p>
<p class="mono">monospace bold uppercase text</p>
<div class="b1">Solid border</div><div class="b2">Dotted border</div><div class="b3">Double border with rounded corners</div>

/* CSS */
.demo h3.title { font-family: Georgia, serif; font-size: 1.6rem; color: #1769b5; letter-spacing: .03em; }
.demo #special { color: #fff; background-color: #14213d; padding: 20px; margin: 10px 0 10px 30px; border-left: 10px solid #f2c200; }
.demo .note { background: rgb(255, 243, 205); color: hsl(35, 90%, 25%); padding: 12px 16px; margin: 15px 0; border: 2px dashed #d98a00; }
.demo p.serif { font-family: "Times New Roman", serif; font-size: 18px; line-height: 1.8; }
.demo p.mono { font-family: "Courier New", monospace; font-weight: bold; text-transform: uppercase; }
.demo .b1 { border: 3px solid #178a4a; padding: 8px; margin-bottom: 8px; }
.demo .b2 { border: 3px dotted #c62828; padding: 8px; margin-bottom: 8px; }
.demo .b3 { border: 3px double #2b59c3; padding: 8px; margin-bottom: 8px; border-radius: 12px; }
