HTML5 input types (email, date, number, password and more)

Shows the new HTML5 input types. Browsers give built-in pickers, keyboards and validation for them.

Working demonstration

View source code of this program 
<!-- HTML -->
<form onsubmit="alert('Valid! Form would be submitted.'); return false;">
  <div class="frow"><label class="l">email</label><input type="email" placeholder="name@example.com"></div>
  <div class="frow"><label class="l">password</label><input type="password" placeholder="hidden characters"></div>
  <div class="frow"><label class="l">number (1-100)</label><input type="number" min="1" max="100" step="1" value="10"></div>
  <div class="frow"><label class="l">date</label><input type="date"></div>
  <div class="frow"><label class="l">time</label><input type="time"></div>
  <div class="frow"><label class="l">month</label><input type="month"></div>
  <div class="frow"><label class="l">tel</label><input type="tel" placeholder="9876543210" pattern="[0-9]{10}"></div>
  <div class="frow"><label class="l">url</label><input type="url" placeholder="https://example.com"></div>
  <div class="frow"><label class="l">search</label><input type="search" placeholder="Search..."></div>
  <div class="frow"><label class="l">color</label><input type="color" value="#2b59c3"></div>
  <div class="frow"><label class="l">range</label><input type="range" min="0" max="10" value="4"></div>
  <button type="submit">Check validation</button>
</form>

/* CSS */
.demo .frow{margin:.6rem 0}.demo label.l{display:inline-block;min-width:150px;font-weight:600}
