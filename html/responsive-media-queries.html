Responsive webpage using CSS media queries

Resize the browser window: the layout switches between 1, 2 and 4 columns and the label shows the active breakpoint.

View source code of this program 
<!-- HTML -->
<span class="badge"></span>
<div class="layout"><div class="box">Box 1</div><div class="box">Box 2</div><div class="box">Box 3</div><div class="box">Box 4</div></div>

/* CSS */
.demo .layout { display: grid; gap: 1rem; grid-template-columns: 1fr; }
.demo .box { background: #d9481f; color: #fff; padding: 1.4rem; border-radius: 8px; text-align: center; font-weight: 700; }
.demo .badge::after { content: "Mobile layout (below 600px): 1 column"; }
.demo .badge { display: block; padding: .6rem 1rem; background: #fff2c2; border-radius: 8px; margin-bottom: 1rem; font-weight: 600; }

@media (min-width: 600px) {
  .demo .layout { grid-template-columns: repeat(2, 1fr); }
  .demo .badge::after { content: "Tablet layout (600px - 899px): 2 columns"; }
}
@media (min-width: 900px) {
  .demo .layout { grid-template-columns: repeat(4, 1fr); }
  .demo .badge::after { content: "Desktop layout (900px and above): 4 columns"; }
  .demo .box { background: #1769b5; }
}
