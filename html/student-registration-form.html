<!-- HTML -->
<form action="#" method="post" onsubmit="alert('Form submitted (demo)'); return false;">
  <fieldset>
    <legend>Personal details</legend>
    <div class="frow"><label class="l" for="fname">Full name</label><input type="text" id="fname" name="fname" placeholder="Your name" required></div>
    <div class="frow"><label class="l" for="email">Email</label><input type="email" id="email" name="email" required></div>
    <div class="frow"><label class="l" for="pwd">Password</label><input type="password" id="pwd" name="pwd" minlength="6" required></div>
    <div class="frow"><span class="l">Gender</span>
      <label><input type="radio" name="gender" value="male"> Male</label>
      <label><input type="radio" name="gender" value="female"> Female</label>
      <label><input type="radio" name="gender" value="other"> Other</label></div>

  <fieldset>
    <legend>Course details</legend>
    <div class="frow"><label class="l" for="course">Course</label>
      <select id="course" name="course"><option value="">-- Select --</option><option>B.Sc Computer Science</option><option>BCA</option><option>B.Com</option></select></div>
    <div class="frow"><span class="l">Hobbies</span>
      <label><input type="checkbox" name="hobby" value="reading"> Reading</label>
      <label><input type="checkbox" name="hobby" value="sports"> Sports</label>
      <label><input type="checkbox" name="hobby" value="music"> Music</label></div>
    <div class="frow"><label class="l" for="photo">Photo</label><input type="file" id="photo" name="photo"></div>
    <div class="frow"><label class="l" for="addr">Address</label><textarea id="addr" name="addr" rows="3" cols="30"></textarea></div>
  </fieldset>
  <button type="submit">Register</button>
  <button type="reset" class="secondary">Reset</button>
</form>
